    80→function DeliveryTick({ status, channel }: { status?: string | null; channel?: string | null }) {
    81→  const label =
    82→    status === "read" ? "Dibaca"
    83→    : status === "delivered" ? "Terkirim"
    84→    : status === "sent" ? "Terkirim"
    85→    : status === "failed" ? "Gagal terkirim"
    86→    : channel === "whatsapp" ? "Menunggu status"
    87→    : null;
    88→  if (!label) return null;
    89→  const Icon =
    90→    status === "read" || status === "delivered" ? CheckCheck
    91→    : status === "sent" ? Check
    92→    : status === "failed" ? AlertTriangle
    93→    : Clock;
    94→  const tone =
    95→    status === "read" ? "text-emerald-600"
    96→    : status === "failed" ? "text-red-600"
    97→    : status === "sent" || status === "delivered" ? "text-zinc-400"
    98→    : "text-zinc-300";
    99→  return (
   100→    <span role="img" aria-label={label} title={label} className="inline-flex shrink-0 items-center">
   101→      <Icon className={cn("size-3.5", tone)} aria-hidden="true" />
   102→    </span>
   103→  );
   104→}

   107→/** Fase 3 — SLA countdown per brand: sisa = SLA brand − jam tunggu (slaHours dari API). */
   108→function slaBadgeInfo(brandSlaHours: number, waitHours: number): {
   109→  label: string; className: string; tone: SlaTone;
   110→} {
   111→  const split = (hours: number) => {
   112→    let h = Math.floor(hours);
   113→    let m = Math.round((hours - h) * 60);
   114→    if (m >= 60) { h += 1; m = 0; }
   115→    return { h, m };
   116→  };
   117→  const fmt = (hours: number, prefix: string) => {
   118→    const { h, m } = split(hours);
   119→    if (h < 1) return `${prefix}${m}m`;
   120→    if (m === 0) return `${prefix}${h}j`;
   121→    return `${prefix}${h}j ${m}m`;
   122→  };
   123→
   124→  const remaining = brandSlaHours - waitHours;
   125→  if (remaining <= 0) {
   126→    return {
   127→      label: fmt(Math.abs(remaining), "Terlambat "),
   128→      className: "border-rose-200 bg-rose-50 text-rose-700",
   129→      tone: "breach" as const,
   130→    };
   131→  }
   132→  if (remaining <= brandSlaHours * 0.5) {
   133→    return {
   134→      label: fmt(remaining, "Segera: "),
   135→      className: "border-amber-200 bg-amber-50 text-amber-700",
   136→      tone: "warning" as const,
   137→    };
   138→  }
   139→  return {
   140→    label: fmt(remaining, "Sisa "),
   141→    className: "border-emerald-200 bg-emerald-50 text-emerald-700",
   142→    tone: "ok" as const,
   143→  };
   144→}

   164→  const Icon = SLA_TONE_ICON[sla.tone];
   165→  return (
   166→    <Badge variant="outline" className={cn("border", sla.className)} aria-label={`Status SLA: ${sla.label}`}>

   172→function scoreTone(score: number): { bar: string; text: string } {
   173→  if (score >= 75) return { bar: "bg-emerald-500", text: "text-emerald-600" };
   174→  if (score >= 50) return { bar: "bg-amber-500", text: "text-amber-600" };
   175→  return { bar: "bg-zinc-400", text: "text-zinc-500" };
   176→}
