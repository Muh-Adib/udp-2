   289→function hashColor(name: string): string {
   290→  let h = 0;
   291→  for (let i = 0; i < name.length; i++) h = (h * 31 + name.charCodeAt(i)) >>> 0;
   292→  return AVATAR_PALETTE[h % AVATAR_PALETTE.length];
   293→}

   295→function useDebounced<T>(value: T, delay = 300): T {
   296→  const [debounced, setDebounced] = useState(value);
   297→  useEffect(() => {
   298→    const t = setTimeout(() => setDebounced(value), delay);
   299→    return () => clearTimeout(t);
   300→  }, [value, delay]);
   301→  return debounced;
   302→}

   312→/**
   313→ * Ronde 40-B — gabung dial + nomor nasional utk payload WhatsApp.
   314→ * Tanpa dial → raw dipertahankan (server tetap normalisasi legacy 0xx → 62xx).
   315→ */
   316→function buildWhatsappPayload(dial: string, national: string): string | null {
   317→  const raw = (national ?? "").trim();
   318→  if (!raw) return null;
   319→  if (!dial) return raw;
   320→  return normalizePhone(raw, dial) ?? raw;
   321→}

   323→/** Ronde 42 — sama dgn WhatsApp, tapi utk field Telepon (dial terpisah dari negara). */
   324→function buildPhonePayload(dial: string, national: string): string | null {
   325→  return buildWhatsappPayload(dial, national);
   326→}

   328→/**
   329→ * Ronde 40-B — pisah nomor tersimpan menjadi { dial, national } utk prefill form.
   330→ * "+628123456789" → { dial: "+62", national: "8123456789" }; nomor lokal (08…)
   331→ * yang tidak cocok dial mana pun tetap utuh tanpa kode.
   332→ */
   333→function splitPhoneParts(full: string): { dial: string; national: string } {
   334→  const trimmed = (full ?? "").trim();
   335→  if (!trimmed) return { dial: "", national: "" };
   336→  const digits = trimmed.replace(/\D/g, "");
   337→  if (!digits) return { dial: "", national: trimmed };
   338→  const candidates = COUNTRIES
   339→    .filter((c) => digits.startsWith(c.dial.slice(1)))
   340→    .sort((a, b) => b.dial.length - a.dial.length);
   341→  const best = candidates[0];
   342→  // sisa nomor harus cukup panjang agar nomor lokal pendek tidak salah potong
   343→  if (best && digits.length - (best.dial.length - 1) >= 6) {
   344→    return { dial: best.dial, national: digits.slice(best.dial.length - 1) };
   345→  }
   346→  return { dial: "", national: trimmed };
   347→}

   349→/** Ronde 40-B — preview E.164 di bawah input WhatsApp (mis. "+6281234567890"). */
   350→function whatsappE164Preview(dial: string, national: string): string | null {
   351→  const raw = (national ?? "").trim();
   352→  if (!raw) return null;
   353→  if (dial) {
   354→    const n = normalizePhone(raw, dial);
   355→    return n ? `+${n}` : null;
   356→  }
   357→  // tanpa kode: hanya bila tampak nomor lokal Indonesia (diawali 0 → server normalkan ke +62…)
   358→  if (raw.replace(/\D/g, "").startsWith("0")) {
   359→    const n = normalizePhone(raw);
   360→    return n ? `+${n}` : null;
   361→  }
   362→  return null;
   363→}

   940→/** Tick status kirim pesan outbound: read/delivered emerald, sent zinc, failed rose. */
   941→function DeliveryTick({ status }: { status?: string | null }) {
   942→  if (!status) return null;
   943→  const s = status.toLowerCase();
   944→  const meta =
   945→    s === "read" || s === "delivered"
   946→      ? { Icon: CheckCheck, cls: "text-emerald-400", label: "Pesan dibaca" }
   947→      : s === "sent"
   948→        ? { Icon: Check, cls: "text-zinc-400", label: "Pesan terkirim" }
   949→        : s === "failed"
   950→          ? { Icon: AlertTriangle, cls: "text-rose-400", label: "Pesan gagal terkirim" }
   951→          : null;
   952→  if (!meta) return null;
   953→  const { Icon, cls, label } = meta;
   954→  return <Icon className={cn("size-3.5 shrink-0", cls)} role="img" aria-label={label} />;
   955→}
