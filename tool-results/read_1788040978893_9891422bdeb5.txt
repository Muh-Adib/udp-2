    70→function escapeCsvField(value: string): string {
    71→  if (/[;"\r\n]/.test(value)) return `"${value.replace(/"/g, '""')}"`;
    72→  return value;

    75→function buildProjectsCsv(projects: ProjectDTO[]): string {
    76→  const lines: string[] = [PROJECT_CSV_HEADERS.join(";")];
    77→  for (const p of projects) {
    78→    const ms = p.milestones ?? [];
    79→    const fields: string[] = [
    80→      p.code,
    81→      p.name,
    82→      p.brand?.name ?? "",
    83→      p.company?.name ?? "",
    84→      p.serviceCategory ?? "",
    85→      statusMeta(p.status).label,
    86→      String(p.progress),
    87→      p.pmName ?? "",
    88→      p.startDate ? formatDate(p.startDate) : "",
    89→      p.dueDate ? formatDate(p.dueDate) : "",
    90→      p.contractValue != null ? String(p.contractValue) : "",
    91→      p.budgetInternal != null ? String(p.budgetInternal) : "",
    92→      String(ms.filter((m) => m.status === "done").length),
    93→      String(ms.length),
    94→    ];
    95→    lines.push(fields.map(escapeCsvField).join(";"));
    96→  }
    97→  return "\uFEFF" + lines.join("\r\n");
    98→}

   320→function TimelineView({ projects, onOpen }: { projects: ProjectDTO[]; onOpen: (p: ProjectDTO) => void }) {
   321→  const range = useMemo(() => {
   322→    const now = Date.now();
   323→    let min: number | null = null;
   324→    let max: number | null = null;
   325→    for (const p of projects) {
   326→      let contributed = false;
   327→      const track = (raw?: string | Date | number | null) => {
   328→        if (!raw) return;
   329→        const ts = new Date(raw).getTime();
   330→        if (!Number.isFinite(ts)) return;
   331→        min = min === null ? ts : Math.min(min, ts);
   332→        max = max === null ? ts : Math.max(max, ts);
   333→        contributed = true;
   334→      };
   335→      track(p.startDate);
   336→      track(p.dueDate);
   337→      for (const m of p.milestones ?? []) track(m.dueDate);
   338→      if (!contributed) track(now); // fallback: project tanpa tanggal → acuan hari ini
   339→    }
   340→    if (min === null || max === null) return { min: now - 15 * DAY_MS, max: now + 15 * DAY_MS };
   341→    const pad = 7 * DAY_MS;
   342→    return { min: min - pad, max: max + pad };
   343→  }, [projects]);
   344→
   345→  const span = Math.max(range.max - range.min, DAY_MS);
   346→  const pct = useCallback((ts: number) => Math.max(0, Math.min(100, ((ts - range.min) / span) * 100)), [range.min, span]);
   347→  const nowPct = pct(Date.now());
   348→
   349→  const gridlines = useMemo(() => {
   350→    const n = 6;
   351→    return Array.from({ length: n }, (_, i) => {
   352→      const ratio = i / (n - 1);
   353→      return { pct: ratio * 100, label: shortDate(range.min + ratio * span) };
   354→    });
   355→  }, [range.min, span]);
   356→
   357→  const todayTransform = nowPct <= 5 ? "translateX(0)" : nowPct >= 95 ? "translateX(-100%)" : "translateX(-50%)";
   358→
   359→  return (
   360→    <div className="overflow-x-auto crm-scroll rounded-xl border bg-white p-4 shadow-sm">
   361→      <div className="min-w-[760px]">
   362→        {/* Sumbu waktu (header) */}
   363→        <div className="flex">
   364→          <div className="w-[200px] shrink-0" />
   365→          <div className="relative h-7 flex-1">
   366→            {gridlines.map((g, i) => (
   367→              <span
   368→                key={i}
   369→                className="absolute top-0 whitespace-nowrap text-[10px] tabular-nums text-zinc-400"
   370→                style={{
   371→                  left: `${g.pct}%`,
   372→                  transform: i === 0 ? "translateX(0)" : i === gridlines.length - 1 ? "translateX(-100%)" : "translateX(-50%)",
   373→                }}
   374→              >
   375→                {g.label}
   376→              </span>
   377→            ))}
   378→            <span
   379→              className="absolute top-0 whitespace-nowrap text-[10px] font-semibold text-rose-500"
   380→              style={{ left: `${nowPct}%`, transform: todayTransform }}
   381→            >
   382→              Hari ini
   383→            </span>
   384→          </div>
   385→        </div>
   386→
   387→        {/* Baris project + gridline vertikal */}
   388→        <div className="relative">
   389→          <div className="pointer-events-none absolute inset-y-0 left-[200px] right-0" aria-hidden>
   390→            {gridlines.map((g, i) => (
   391→              <div key={i} className="absolute inset-y-0 border-l border-dashed border-zinc-200" style={{ left: `${g.pct}%` }} />
   392→            ))}
   393→            <div className="absolute inset-y-0 border-l-2 border-rose-400" style={{ left: `${nowPct}%` }} />
   394→          </div>
   395→          {projects.map((p) => (
   396→            <TimelineRow key={p.id} project={p} pct={pct} onOpen={onOpen} />
   397→          ))}
   398→        </div>
   399→
   400→        {/* Legenda */}
   401→        <div className="mt-3 flex flex-wrap items-center gap-x-4 gap-y-1.5 border-t border-zinc-100 pt-3 text-[11px] text-zinc-500">
   402→          <span className="flex items-center gap-1.5"><span className="h-2.5 w-5 rounded-full bg-zinc-400" aria-hidden /> Perencanaan</span>
   403→          <span className="flex items-center gap-1.5"><span className="h-2.5 w-5 rounded-full bg-zinc-900/90" aria-hidden /> Berjalan</span>
   404→          <span className="flex items-center gap-1.5"><span className="h-2.5 w-5 rounded-full bg-violet-500" aria-hidden /> Review</span>
   405→          <span className="flex items-center gap-1.5"><span className="h-2.5 w-5 rounded-full bg-emerald-500" aria-hidden /> Selesai</span>
   406→          <span className="flex items-center gap-1.5"><span className="h-2 w-2 rotate-45 rounded-[2px] bg-emerald-500" aria-hidden /> Milestone selesai</span>
   407→          <span className="flex items-center gap-1.5"><span className="h-2 w-2 rotate-45 rounded-[2px] bg-amber-500" aria-hidden /> Dikerjakan</span>
   408→          <span className="flex items-center gap-1.5"><span className="h-2 w-2 rotate-45 rounded-[2px] bg-zinc-300" aria-hidden /> Menunggu</span>
   409→          <span className="flex items-center gap-1.5"><span className="h-3.5 border-l-2 border-rose-400" aria-hidden /> Hari ini</span>
   410→        </div>

   445→  // Event per tanggal (key "y-m-d" lokal). Project/milestone tanpa dueDate diabaikan.
   446→  // Milestone hanya dari project berstatus bukan completed/cancelled.
   447→  const eventsByDay = useMemo(() => {
   448→    const map = new Map<string, CalendarEvent[]>();
   449→    const push = (d: Date, ev: CalendarEvent) => {
   450→      const key = dateKey(d);
   451→      const arr = map.get(key);
   452→      if (arr) arr.push(ev);
   453→      else map.set(key, [ev]);
   454→    };
   455→    for (const p of projects) {
   456→      if (p.dueDate) {
   457→        const d = new Date(p.dueDate);
   458→        if (!Number.isNaN(d.getTime())) push(d, { kind: "project", project: p });
   459→      }
   460→      if (p.status !== "completed" && p.status !== "cancelled") {
   461→        for (const m of p.milestones ?? []) {
   462→          if (!m.dueDate) continue;
   463→          const d = new Date(m.dueDate);
   464→          if (!Number.isNaN(d.getTime())) push(d, { kind: "milestone", project: p, milestone: m });
   465→        }
   466→      }
   467→    }
   468→    return map;
   469→  }, [projects]);
   470→
   471→  // Grid 6 baris × 7 kolom (Senin-based) — tinggi konsisten antar bulan.
   472→  const cells = useMemo(() => {
   473→    const offset = (new Date(cursor.year, cursor.month, 1).getDay() + 6) % 7;
   474→    const daysInMonth = new Date(cursor.year, cursor.month + 1, 0).getDate();
   475→    return Array.from({ length: 42 }, (_, i) => {
   476→      const day = i - offset + 1;
   477→      const valid = day >= 1 && day <= daysInMonth;
   478→      return { i, day: valid ? day : null, date: valid ? new Date(cursor.year, cursor.month, day) : null };
   479→    });
   480→  }, [cursor]);
   481→
   482→  const now = new Date();
   483→  const todayKey = dateKey(now);
   484→  const isCurrentMonth = cursor.year === now.getFullYear() && cursor.month === now.getMonth();
   485→  const monthLabel = new Intl.DateTimeFormat("id-ID", { month: "long", year: "numeric" })
   486→    .format(new Date(cursor.year, cursor.month, 1));
   487→
   488→  // Warna brand unik (maks 4) untuk legenda "Deadline project".
   489→  const brandColors = useMemo(() => {
   490→    const seen: string[] = [];
   491→    for (const p of projects) {
   492→      if (p.brand?.color && !seen.includes(p.brand.color)) seen.push(p.brand.color);
   493→      if (seen.length >= 4) break;
   494→    }
   495→    return seen;
   496→  }, [projects]);
   497→
   498→  function shiftMonth(delta: number) {
   499→    setCursor((c) => {
   500→      const d = new Date(c.year, c.month + delta, 1);
   501→      return { year: d.getFullYear(), month: d.getMonth() };
   502→    });
   503→  }
   504→
   505→  if (projects.length === 0) {
   506→    return (
   507→      <div className="flex flex-col items-center gap-2 rounded-xl border border-dashed bg-white p-10 text-center shadow-sm">
   508→        <CalendarDays className="h-8 w-8 text-zinc-300" aria-hidden />
   509→        <p className="text-sm text-zinc-400">Tidak ada project untuk filter ini.</p>
   510→      </div>
   511→    );
   512→  }
   513→
   514→  return (
   515→    <div className="overflow-x-auto crm-scroll rounded-xl border bg-white shadow-sm">
   516→      <div className="min-w-[680px] p-4">
   517→        {/* Header bulan */}
   518→        <div className="flex items-center gap-1">
   519→          <Button type="button" variant="ghost" size="icon" className="h-8 w-8" onClick={() => shiftMonth(-1)} aria-label="Bulan sebelumnya">
   520→            <ChevronLeft className="h-4 w-4" aria-hidden />
   521→          </Button>
   522→          <Button type="button" variant="ghost" size="icon" className="h-8 w-8" onClick={() => shiftMonth(1)} aria-label="Bulan berikutnya">
   523→            <ChevronRight className="h-4 w-4" aria-hidden />
   524→          </Button>
   525→          <p className="ml-1 text-sm font-semibold text-zinc-900">{monthLabel}</p>
   526→          <Button
   527→            type="button"
   528→            variant="outline"
   529→            size="sm"
   530→            className="ml-auto h-8"
   531→            onClick={() => setCursor({ year: now.getFullYear(), month: now.getMonth() })}
   532→            disabled={isCurrentMonth}
   533→          >
   534→            Hari ini
   535→          </Button>
   536→        </div>
   537→
   538→        {/* Grid kalender */}
   539→        <div className="mt-3 overflow-hidden rounded-lg border border-zinc-200">
   540→          <div className="grid grid-cols-7 border-b border-zinc-200 bg-zinc-50">
   541→            {WEEKDAY_LABELS.map((w, i) => (
   542→              <div
   543→                key={w}
   544→                className={`px-2 py-1.5 text-[11px] font-semibold text-zinc-500 ${i < 6 ? "border-r border-zinc-200" : ""}`}
   545→              >
   546→                {w}
   547→              </div>
   548→            ))}
   549→          </div>
   550→          <div className="grid grid-cols-7">
   551→            {cells.map((cell) => {

   554→              const rest = dayEvents.slice(3);
   555→              const restTitle = rest
   556→                .map((ev) => (ev.kind === "project" ? `${ev.project.code} · deadline` : `${ev.project.code} · ${ev.milestone.name}`))

   559→              const isToday = cell.date ? dateKey(cell.date) === todayKey : false;
   560→              const bgCls = !cell.date ? "bg-zinc-50/40" : isToday ? "bg-zinc-100" : isWeekend ? "bg-zinc-50/60" : "";
   561→              const cellKey = cell.date ? dateKey(cell.date) : null;
   562→              const isDropTarget = Boolean(dragging) && cellKey !== null && dropDayKey === cellKey;
   563→              return (

   565→                  key={cell.i}
   566→                  className={`min-h-[92px] p-1.5 ${bgCls} ${cell.i % 7 !== 6 ? "border-r border-zinc-100" : ""} ${cell.i < 35 ? "border-b border-zinc-100" : ""} ${
   567→                    isDropTarget ? "ring-2 ring-inset ring-zinc-900 bg-zinc-100" : ""
   568→                  }`}
   569→                  onDragOver={(e) => {
   570→                    if (!dragging || !cell.date) return;
   571→                    e.preventDefault();
   572→                    e.dataTransfer.dropEffect = "move";
   573→                    if (dropDayKey !== cellKey) setDropDayKey(cellKey);
   574→                  }}
   575→                  onDragLeave={() => {
   576→                    if (cellKey && dropDayKey === cellKey) setDropDayKey(null);
   577→                  }}
   578→                  onDrop={(e) => {
   579→                    e.preventDefault();
   580→                    const target = dragging;
   581→                    setDragging(null);
   582→                    setDropDayKey(null);
   583→                    if (!target || !cell.date || !onRescheduleMilestone) return;
   584→                    const project = projects.find((p) => p.id === target.projectId);
   585→                    const ms = project?.milestones?.find((m) => m.id === target.milestoneId);
   586→                    if (project && ms) onRescheduleMilestone(project, ms, cell.date);
   587→                  }}
   588→                >
   589→                  <div className="flex items-center">
   590→                    {cell.day !== null ? (
   591→                      isToday ? (
   592→                        <span className="flex h-5 min-w-5 items-center justify-center rounded-full bg-zinc-900 px-1 text-[11px] font-bold leading-none text-white tabular-nums">
   593→                          {cell.day}
   594→                        </span>
   595→                      ) : (
   596→                        <span className="text-[11px] leading-none text-zinc-500 tabular-nums">{cell.day}</span>
   597→                      )
   598→                    ) : null}
   599→                  </div>
   600→                  {visible.length > 0 ? (
   601→                    <div className="mt-1 space-y-1">
   602→                      {visible.map((ev) =>
   603→                        ev.kind === "project" ? (
   604→                          <button
   605→                            key={`p-${ev.project.id}`}
   606→                            type="button"
   607→                            onClick={() => onOpenDetail(ev.project)}
   608→                            title={`${ev.project.code} · ${ev.project.name} · deadline · ${ev.project.progress}%`}
   609→                            aria-label={`Buka detail project ${ev.project.name} (deadline)`}
   610→                            className="block w-full truncate rounded px-1.5 py-0.5 text-left text-[10px] font-medium text-white transition-opacity hover:opacity-80"
   611→                            style={{ backgroundColor: ev.project.brand?.color ?? "#3f3f46" }}
   612→                          >
   613→                            {ev.project.code} · deadline
   614→                          </button>
   615→                        ) : (
   616→                          <button
   617→                            key={`m-${ev.milestone.id}`}
   618→                            type="button"
   619→                            onClick={() => onOpenDetail(ev.project)}
   620→                            draggable={ev.milestone.status !== "done" && Boolean(onRescheduleMilestone)}
   621→                            onDragStart={(e) => {
   622→                              if (ev.milestone.status === "done") return;
   623→                              e.dataTransfer.setData("text/plain", ev.milestone.id);
   624→                              e.dataTransfer.effectAllowed = "move";
   625→                              setDragging({ milestoneId: ev.milestone.id, projectId: ev.project.id });
   626→                            }}
   627→                            onDragEnd={() => {
   628→                              setDragging(null);
   629→                              setDropDayKey(null);
   630→                            }}
   631→                            title={`${ev.project.code} · ${ev.milestone.name} · ${msMeta(ev.milestone.status).label}${ev.milestone.status !== "done" && onRescheduleMilestone ? " · seret untuk menjadwalkan ulang" : ""}`}
   632→                            aria-label={`Buka detail project ${ev.project.name} — milestone ${ev.milestone.name}${ev.milestone.status !== "done" && onRescheduleMilestone ? " (dapat diseret ke tanggal lain)" : ""}`}
   633→                            className={cn(
   634→                              "flex w-full items-center gap-1 truncate rounded border border-zinc-200 bg-white px-1 py-0.5 text-left text-[10px] text-zinc-700 transition-colors hover:bg-zinc-50",
   635→                              ev.milestone.status !== "done" && onRescheduleMilestone && "cursor-grab active:cursor-grabbing hover:border-zinc-400",
   636→                              dragging?.milestoneId === ev.milestone.id && "opacity-40"
   637→                            )}
   638→                            style={{ borderLeftWidth: 2, borderLeftColor: ev.project.brand?.color ?? "#a1a1aa" }}
   639→                          >
   640→                            <span
   641→                              className={`h-1.5 w-1.5 shrink-0 rotate-45 rounded-[1px] ${
   642→                                ev.milestone.status === "done" ? "bg-emerald-500" : ev.milestone.status === "in_progress" ? "bg-amber-500" : "bg-zinc-300"
   643→                              }`}
   644→                              aria-hidden
   645→                            />
   646→                            <span className="truncate">{ev.milestone.name}</span>
   647→                            {ev.milestone.status !== "done" && onRescheduleMilestone ? (
   648→                              <GripVertical className="ml-auto h-2.5 w-2.5 shrink-0 text-zinc-300" aria-hidden />
   649→                            ) : null}
   650→                          </button>
   651→                        )
   652→                      )}
   653→                    </div>
   654→                  ) : null}
   655→                  {rest.length > 0 ? (
   656→                    <span className="mt-1 block truncate text-[10px] font-medium text-zinc-500" title={restTitle}>
   657→                      +{rest.length} lagi
   658→                    </span>
   659→                  ) : null}
   660→                </div>
   661→              );
   662→            })}
   663→          </div>
   664→        </div>
   665→
   666→        {/* Legenda */}
   667→        <div className="mt-4 flex flex-wrap items-center gap-x-4 gap-y-1.5 border-t border-zinc-100 pt-3 text-[11px] text-zinc-500">
   668→          <span className="flex items-center gap-1.5">
   669→            <span className="flex -space-x-1" aria-hidden>
   670→              {(brandColors.length > 0 ? brandColors : [null]).map((c, idx) => (
   671→                <span
   672→                  key={idx}
   673→                  className="h-2.5 w-3 rounded-full ring-1 ring-white"
   674→                  style={{ backgroundColor: c ?? "#a1a1aa" }}
   675→                />
   676→              ))}
   677→            </span>
   678→            Deadline project
   679→          </span>
   680→          <span className="flex items-center gap-1.5"><span className="h-2 w-2 rotate-45 rounded-[2px] bg-emerald-500" aria-hidden /> Milestone selesai</span>
   681→          <span className="flex items-center gap-1.5"><span className="h-2 w-2 rotate-45 rounded-[2px] bg-amber-500" aria-hidden /> Dikerjakan</span>
   682→          <span className="flex items-center gap-1.5"><span className="h-2 w-2 rotate-45 rounded-[2px] bg-zinc-300" aria-hidden /> Menunggu</span>
   683→          {onRescheduleMilestone ? (
   684→            <span className="flex items-center gap-1.5 text-zinc-400">
   685→              <GripVertical className="h-3 w-3" aria-hidden />
   686→              Seret milestone (bukan yang selesai) ke tanggal lain untuk menjadwalkan ulang
   687→            </span>
   688→          ) : null}
   689→        </div>
   690→      </div>
   691→    </div>
   692→  );
   693→}

   820→  async function handleRescheduleMilestone(project: ProjectDTO, m: MilestoneDTO, newDate: Date) {
   821→    const oldDue = m.dueDate ? new Date(m.dueDate) : null;
   822→    const sameDay = oldDue && oldDue.getFullYear() === newDate.getFullYear()
   823→      && oldDue.getMonth() === newDate.getMonth() && oldDue.getDate() === newDate.getDate();
   824→    if (sameDay) return;
   825→    if (m.status === "done") {

   827→      return;
   828→    }
   829→
   830→    const applyDue = (iso: string | null) => {
   831→      setProjects((prev) => (prev ?? []).map((p) =>
   832→        p.id !== project.id ? p : {
   833→          ...p,
   834→          milestones: (p.milestones ?? []).map((x) => (x.id === m.id ? { ...x, dueDate: iso } : x)),
   835→        }
   836→      ));
   837→      setDetail((d) => (d && d.id === project.id
   838→        ? { ...d, milestones: (d.milestones ?? []).map((x) => (x.id === m.id ? { ...x, dueDate: iso } : x)) }
   839→        : d));
   840→    };
   841→
   842→    applyDue(newDate.toISOString()); // optimis
   843→    try {
   844→      await api.updateMilestone({
   845→        milestoneId: m.id,
   846→        dueDate: newDate.toISOString(),
   847→        actorName: user?.name ?? "Produksi",
   848→        actorRole: user?.role ?? "production",
   849→      });
   850→      toast.success(`Milestone "${m.name}" dijadwalkan ulang`, {
   851→        description: oldDue ? `${formatDate(oldDue)} → ${formatDate(newDate)}` : `Deadline baru: ${formatDate(newDate)}`,
   852→      });
   853→    } catch (err) {
   854→      applyDue(oldDue ? oldDue.toISOString() : null); // revert
   855→      toast.error(err instanceof Error ? err.message : "Gagal menjadwalkan ulang milestone");
   856→    }
   857→  }
