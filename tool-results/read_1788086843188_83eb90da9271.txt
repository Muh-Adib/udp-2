    75→/** Parse item quotation defensif — items bisa JSON string atau array. */
    76→function parseQuoteItems(items: string | QuotationItemDTO[]): QuotationItemDTO[] {
    77→  if (Array.isArray(items)) return items;
    78→  try {
    79→    const parsed: unknown = JSON.parse(items);
    80→    return Array.isArray(parsed) ? (parsed as QuotationItemDTO[]) : [];
    81→  } catch {
    82→    return [];
    83→  }
    84→}

    86→/** Domain dari URL/email sederhana, untuk mencocokkan akun client dengan perusahaannya. */
    87→function domainOf(raw?: string | null): string | null {
    88→  if (!raw) return null;
    89→  const s = raw.trim().toLowerCase().replace(/^https?:\/\//, "").replace(/^www\./, "").split("/")[0].split("@").pop() ?? "";
    90→  return s.includes(".") ? s : null;
    91→}

   249→export default function PortalModule() {
   250→  const user = useCrmStore((s) => s.user);
   251→  const isClient = user?.role === "client";
   252→  const canPreview = user?.role === "super_admin" || user?.role === "director";
   253→
   254→  const [companies, setCompanies] = useState<CompanyRef[] | null>(null);
   255→  const [selectedCompanyId, setSelectedCompanyId] = useState("");
   256→  const [linkInfo, setLinkInfo] = useState<{ company: CompanyRef | null; resolved: boolean } | null>(null);
   257→
   258→  const [projects, setProjects] = useState<ProjectDTO[] | null>(null);
   259→  const [invoices, setInvoices] = useState<InvoiceDTO[] | null>(null);
   260→  const [changeRequests, setChangeRequests] = useState<ChangeRequestDTO[] | null>(null);
   261→  const [quotations, setQuotations] = useState<QuotationDTO[] | null>(null);
   262→  const [loading, setLoading] = useState(true);
   263→  const [error, setError] = useState<string | null>(null);
   264→
   265→  const [detail, setDetail] = useState<InvoiceDTO | null>(null);
   266→  const [quoteDetail, setQuoteDetail] = useState<QuotationDTO | null>(null);
   267→  const [crDecision, setCrDecision] = useState<{ cr: ChangeRequestDTO; decision: "approve" | "reject" } | null>(null);
   268→  const [decisionNote, setDecisionNote] = useState("");
   269→  const [savingDecide, setSavingDecide] = useState<string | null>(null);
   270→
   271→  /** Untuk role client: cari perusahaan miliknya ( companyName → fallback domain email vs website perusahaan). */
   272→  const resolveClientCompany = useCallback(async (): Promise<CompanyRef | null> => {
   273→    const { companies: list } = await api.companies();
   274→    setCompanies(list);
   275→    const byName = user?.companyName
   276→      ? list.find((c) => c.name.toLowerCase() === user.companyName?.toLowerCase())
   277→      : undefined;
   278→    if (byName) return byName;
   279→    // Fallback demo: cocokkan domain email user dengan domain website perusahaan
   280→    const emailDomain = domainOf(user?.email);
   281→    if (emailDomain) {
   282→      const byDomain = list.find((c) => domainOf(c.website) === emailDomain);
   283→      if (byDomain) return byDomain;
   284→    }
   285→    return null;
   286→  }, [user?.companyName, user?.email]);
   287→
   288→  const loadCompanyData = useCallback(async (companyId: string, silent = false) => {
   289→    if (!silent) setLoading(true);
   290→    setError(null);
   291→    try {
   292→      const [projRes, invRes, crRes, quoteRes] = await Promise.all([
   293→        api.projects({ companyId }),
   294→        api.invoices({ companyId }),
   295→        api.changeRequests({ companyId }),
   296→        api.quotations({ companyId }),
   297→      ]);
   298→      setProjects(projRes.projects);
   299→      setInvoices(invRes.invoices);
   300→      setChangeRequests(crRes.changeRequests);
   301→      setQuotations(quoteRes.quotations);
   302→    } catch (err) {
   303→      setError(err instanceof Error ? err.message : "Gagal memuat data portal");
   304→      if (!silent) toast.error("Gagal memuat data portal klien");
   305→    } finally {
   306→      setLoading(false);
   307→    }
   308→  }, []);
   309→
   310→  useEffect(() => {
   311→    (async () => {
   312→      setLoading(true);
   313→      setError(null);
   314→      try {
   315→        if (isClient) {
   316→          const company = await resolveClientCompany();
   317→          setLinkInfo({ company, resolved: company !== null });
   318→          if (company) await loadCompanyData(company.id, true);
   319→          else setLoading(false);
   320→        } else if (canPreview) {
   321→          const { companies: list } = await api.companies();
   322→          setCompanies(list);
   323→          // Jangan pilih otomatis — tunggu admin memilih perusahaan
   324→          setLoading(false);
   325→        } else {
   326→          setLoading(false);
   327→        }
   328→      } catch (err) {
   329→        setError(err instanceof Error ? err.message : "Gagal memuat portal");
   330→        setLoading(false);
   331→      }
   332→    })();
   333→  }, [isClient, canPreview, resolveClientCompany, loadCompanyData]);
   334→
   335→  const activeCompany = useMemo(() => {
   336→    if (isClient) return linkInfo?.company ?? null;
   337→    return companies?.find((c) => c.id === selectedCompanyId) ?? null;
   338→  }, [isClient, linkInfo, companies, selectedCompanyId]);
   339→
   340→  const invoiceSummary = useMemo(() => {
   341→    const list = invoices ?? [];
   342→    const outstanding = list
   343→      .filter((i) => ["sent", "partial", "overdue"].includes(i.status))
   344→      .reduce((s, i) => s + Math.max(0, i.total - (i.payments ?? []).reduce((ps, p) => ps + p.amount, 0)), 0);
   345→    return { count: list.length, outstanding };
   346→  }, [invoices]);
   347→
   348→  function pickCompany(id: string) {
   349→    setSelectedCompanyId(id);
   350→    setProjects(null);
   351→    setInvoices(null);
   352→    setChangeRequests(null);
   353→    setQuotations(null);
   354→    setDetail(null);
   355→    setQuoteDetail(null);
   356→    void loadCompanyData(id);
   357→  }
   358→
   359→  const canDecide = isClient || canPreview; // client / director / super_admin
   360→
   361→  const crSummary = useMemo(() => {
   362→    const list = changeRequests ?? [];
   363→    const approved = list.filter((c) => c.status === "approved");
   364→    return {
   365→      pendingCount: list.filter((c) => c.status === "pending").length,
   366→      approvedCount: approved.length,
   367→      approvedValue: approved.reduce((s, c) => s + (c.additionalCost ?? 0), 0),
   368→    };
   369→  }, [changeRequests]);
   370→
   371→  /** Brand unik dari project yang dimuat — strip identitas agency di header konten. */
   372→  const brandStrip = useMemo(() => {
   373→    const map = new Map<string, { id: string; name: string; color: string; count: number }>();
   374→    for (const p of projects ?? []) {
   375→      if (!p.brand) continue;
   376→      const cur = map.get(p.brand.id) ?? { id: p.brand.id, name: p.brand.name, color: p.brand.color, count: 0 };
   377→      cur.count += 1;
   378→      map.set(p.brand.id, cur);
   379→    }
   380→    return [...map.values()];
   381→  }, [projects]);
   382→
   383→  async function confirmDecision() {
   384→    const target = crDecision;
   385→    if (!target || !user) return;
   386→    const { cr, decision } = target;
   387→    setSavingDecide(cr.id);
   388→    try {
   389→      const res = await api.decideChangeRequest({
   390→        id: cr.id,
   391→        decision,
   392→        decisionNote: decisionNote.trim() || undefined,
   393→        actorName: user.name,
   394→        actorRole: user.role,
   395→      });
   396→      if (decision === "approve") {
   397→        toast.success(res.invoice?.number ? `Perubahan disetujui — invoice ${res.invoice.number} diterbitkan` : "Perubahan disetujui");
   398→      } else {
   399→        toast.info("Perubahan ditolak — tim kami akan mengikuti scope awal");
   400→      }
   401→      setCrDecision(null);
   402→      setDecisionNote("");
   403→      if (activeCompany) void loadCompanyData(activeCompany.id, true);
   404→    } catch (err) {
   405→      toast.error(err instanceof Error ? err.message : "Gagal memutuskan permintaan perubahan");
   406→    } finally {
   407→      setSavingDecide(null);
   408→    }
   409→  }
   410→
   411→  if (loading && projects === null && !isClient) return <PortalSkeleton />;
   412→  if (loading && isClient && linkInfo === null) return <PortalSkeleton />;
   413→
   414→  return (
   415→    <div className="space-y-6">
   416→      {/* Header */}
   417→      <div className="flex flex-col gap-3 sm:flex-row sm:items-start sm:justify-between">
   418→        <div>
   419→          <h1 className="flex flex-wrap items-center gap-2 text-xl font-bold tracking-tight text-zinc-900">
   420→            Client Portal
   421→            {!isClient && activeCompany ? (
   422→              <Badge className="border-transparent bg-amber-100 text-amber-700">Mode Pratinjau Klien</Badge>
   423→            ) : null}
   424→          </h1>
   425→          <p className="text-sm text-zinc-500">Tampilan terbatas untuk klien — tanpa data internal</p>
   426→        </div>
   427→        {activeCompany ? (
   428→          <Button variant="outline" size="sm" onClick={() => void loadCompanyData(activeCompany.id)} disabled={loading} aria-label="Muat ulang data portal">
   429→            <RefreshCw className={`h-4 w-4 ${loading ? "animate-spin" : ""}`} aria-hidden /> Muat ulang
   430→          </Button>
   431→        ) : null}
   432→      </div>
   433→
   434→      {error ? (
   435→        <div className="rounded-xl border border-rose-200 bg-rose-50 p-4 text-sm text-rose-700">{error}</div>
   436→      ) : null}
   437→
   438→      {/* Role client: penanganan akun belum terkait */}
   439→      {isClient && linkInfo && !linkInfo.company ? (
