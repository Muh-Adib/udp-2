    56→    56→function startOfToday(): Date {

    62→    62→function endOfToday(): Date {
    63→    63→  const d = new Date();
    64→    64→  d.setHours(23, 59, 59, 999);
    65→    65→  return d;

    68→    68→function isOverdue(t: TaskDTO): boolean {
    69→    69→  return t.status === "open" && !!t.dueDate && new Date(t.dueDate).getTime() < startOfToday().getTime();
    70→    70→}
    71→    71→
    72→    72→function isToday(t: TaskDTO): boolean {
    73→    73→  if (!t.dueDate) return false;
    74→    74→  const due = new Date(t.dueDate).getTime();
    75→    75→  return due >= startOfToday().getTime() && due <= endOfToday().getTime();
    76→    76→}

   272→   272→function SendOutboundDialog({ task, onClose, onSent }: {
   273→   273→  task: TaskDTO | null;
   274→   274→  onClose: () => void;
   275→   275→  onSent: (task: TaskDTO, updated?: TaskDTO) => void;
   276→   276→}) {
   277→   277→  const user = useCrmStore((s) => s.user);
   278→   278→  const [ctx, setCtx] = useState<SendContext | null>(null);
   279→   279→  const [templates, setTemplates] = useState<FollowUpTemplateDTO[]>([]);
   280→   280→  const [loadingCtx, setLoadingCtx] = useState(false);
   281→   281→  const [channel, setChannel] = useState("whatsapp");
   282→   282→  const [templateId, setTemplateId] = useState<string>("blank");
   283→   283→  const [body, setBody] = useState("");
   284→   284→  const [markDone, setMarkDone] = useState(true);
   285→   285→  const [sending, setSending] = useState(false);
   286→   286→
   287→   287→  // Muat konteks opportunity + template saat dialog dibuka
   288→   288→  useEffect(() => {
   289→   289→    if (!task?.opportunityId) return;
   290→   290→    let alive = true;
   291→   291→    setLoadingCtx(true);
   292→   292→    setCtx(null);
   293→   293→    setTemplates([]);
   294→   294→    setTemplateId("blank");
   295→   295→    setBody("");
   296→   296→    setChannel("whatsapp");
   297→   297→    setMarkDone(task.status !== "done");
   298→   298→    (async () => {
   299→   299→      try {
   300→   300→        const [detail, tplRes] = await Promise.all([
   301→   301→          api.opportunity(task.opportunityId as string),
   302→   302→          api.followUpTemplates("all"),
   303→   303→        ]);
   304→   304→        if (!alive) return;
   305→   305→        const o = detail.opportunity;
   306→   306→        const nextCtx: SendContext = {
   307→   307→          id: o.id,
   308→   308→          title: o.title,
   309→   309→          serviceName: o.serviceName,
   310→   310→          brand: o.brand ? { id: o.brand.id, name: o.brand.name, color: o.brand.color } : null,
   311→   311→          contact: o.contact
   312→   312→            ? { id: o.contact.id, fullName: o.contact.fullName, email: o.contact.email, whatsapp: o.contact.whatsapp }
   313→   313→            : null,
   314→   314→          company: o.company ? { id: o.company.id, name: o.company.name } : null,
   315→   315→          ownerName: o.ownerName,
   316→   316→        };
   317→   317→        setCtx(nextCtx);
   318→   318→        // Template global + milik brand opportunity, urut delay
   319→   319→        const relevant = tplRes.templates
   320→   320→          .filter((t) => !t.brandId || t.brandId === o.brandId)
   321→   321→          .sort((a, b) => a.delayDays - b.delayDays);
   322→   322→        setTemplates(relevant);
   323→   323→      } catch {
   324→   324→        if (alive) toast.error("Gagal memuat konteks opportunity");
   325→   325→      } finally {
   326→   326→        if (alive) setLoadingCtx(false);
   327→   327→      }
   328→   328→    })();
   329→   329→    return () => { alive = false; };
   330→   330→  }, [task]);
   331→   331→
   332→   332→  const senderName = user?.name ?? ctx?.ownerName ?? "Tim Sales";
   333→   333→
   334→   334→  function pickTemplate(id: string) {
   335→   335→    setTemplateId(id);
   336→   336→    if (!ctx) return;
   337→   337→    if (id === "blank") {
   338→   338→      setBody("");
   339→   339→      return;
   340→   340→    }
   341→   341→    const tpl = templates.find((t) => t.id === id);
   342→   342→    if (!tpl) return;
   343→   343→    setChannel(tpl.channel);
   344→   344→    setBody(renderTemplate(tpl.body, ctx, senderName));
   345→   345→  }
   346→   346→
   347→   347→  const unknownPlaceholders = useMemo(() => {
   348→   348→    if (!ctx) return [] as string[];
   349→   349→    const found = body.match(/{{\s*(\w+)\s*}}/g) ?? [];
   350→   350→    return Array.from(new Set(found));
   351→   351→  }, [body, ctx]);
   352→   352→
   353→   353→  async function handleSend() {
   354→   354→    if (!task || !ctx) return;
   355→   355→    const content = body.trim();
   356→   356→    if (!content) {
   357→   357→      toast.error("Isi pesan tidak boleh kosong");
   358→   358→      return;
   359→   359→    }
   360→   360→    const ch = channelMeta(channel);
   361→   361→    if (ch.key === "email" && !ctx.contact?.email) {
   362→   362→      toast.error("Kontak tidak memiliki alamat email");
   363→   363→      return;
   364→   364→    }
   365→   365→    if (ch.key === "whatsapp" && !ctx.contact?.whatsapp) {
   366→   366→      toast.error("Kontak tidak memiliki nomor WhatsApp");
   367→   367→      return;
   368→   368→    }
   369→   369→    setSending(true);
   370→   370→    try {
   371→   371→      await api.createInteraction({
   372→   372→        channel: ch.key,
   373→   373→        direction: "outbound",
   374→   374→        brandId: ctx.brand?.id,
   375→   375→        senderName,
   376→   376→        recipientName: ctx.contact?.fullName,
   377→   377→        subject: ch.key === "email" ? `Follow-up: ${ctx.title}` : undefined,
   378→   378→        content,
   379→   379→        opportunityId: ctx.id,
   380→   380→        contactId: ctx.contact?.id,
   381→   381→        companyId: ctx.company?.id,
   382→   382→        externalId: `task:${task.id}`,
   383→   383→        actorName: user?.name ?? "System",
   384→   384→        actorRole: user?.role ?? "system",
   385→   385→      });
   386→   386→      let updated: TaskDTO | undefined;
   387→   387→      if (markDone && task.status !== "done") {
   388→   388→        const res = await api.updateTask(task.id, { status: "done" });
   389→   389→        updated = res.task;
   390→   390→      }
   391→   391→      toast.success(`Pesan ${ch.label} tercatat terkirim`, {
   392→   392→        description: `Tercatat sebagai Interaction outbound pada timeline ${ctx.title}.`,
   393→   393→      });
   394→   394→      onSent(task, updated);
   395→   395→      onClose();
   396→   396→    } catch (err) {
   397→   397→      toast.error(err instanceof Error ? err.message : "Gagal mencatat pesan");
   398→   398→    } finally {
   399→   399→      setSending(false);
   400→   400→    }
   401→   401→  }
   402→   402→
   403→   403→  const ChIcon = channelMeta(channel).icon;
   404→   404→
   405→   405→  return (
   406→   406→    <Dialog open={task !== null} onOpenChange={(open) => { if (!open && !sending) onClose(); }}>
   407→   407→      <DialogContent className="max-h-[92vh] overflow-y-auto crm-scroll sm:max-w-xl" aria-label="Kirim pesan follow-up">
   408→   408→        <DialogHeader>
