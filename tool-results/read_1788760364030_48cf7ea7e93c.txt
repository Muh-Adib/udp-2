    87→// Keluhan user: "untuk keproduksian floenya belum jelas recordnya". Sheet detail dulu

    91→const PRODUCTION_STEPS = ["Deal & Kontrak", "Perencanaan", "Produksi", "Review", "Serah Terima"] as const;

    93→function ProductionFlow({ project, deliverables, crs, invoices }: {
    94→  project: ProjectDTO;
    95→  deliverables: ProjectDeliverableDTO[] | null;
    96→  crs: ChangeRequestDTO[];
    97→  invoices: InvoiceDTO[] | null;
    98→}) {
    99→  const milestones = project.milestones ?? [];
   100→  const doneMs = milestones.filter((m) => m.status === "done").length;
   101→  const dlv = deliverables ?? [];
   102→  const dlvApproved = dlv.filter((d) => d.status === "approved").length;
   103→  const dlvRevision = dlv.filter((d) => d.status === "revision").length;
   104→  const dlvPending = dlv.filter((d) => d.status === "pending").length;
   105→  const approvedCr = crs.filter((c) => c.status === "approved");
   106→  const invTotal = (invoices ?? []).reduce((s, i) => s + i.total, 0);
   107→  const paidTotal = (invoices ?? []).reduce((s, i) => s + (i.payments ?? []).reduce((ps, p) => ps + p.amount, 0), 0);
   108→
   109→  // Posisi tahap saat ini (0-4): planning→1, in_progress→2, review→3, completed→4 (semua selesai)
   110→  const currentIdx = project.status === "planning" ? 1
   111→    : project.status === "in_progress" ? 2
   112→    : project.status === "review" ? 3
   113→    : project.status === "completed" ? 5 // 5 = seluruh alur selesai
   114→    : 0;
   115→
   116→  /** Record yang tampil di bawah tiap nama tahap — dari data nyata, bukan teks statis. */
   117→  const stepRecords: string[] = [
   118→    // 0 — Deal & Kontrak
   119→    project.opportunityId
   120→      ? `Dari opportunity won · kontrak ${formatCurrency(project.contractValue ?? 0)}`
   121→      : `Project manual · kontrak ${formatCurrency(project.contractValue ?? 0)}`,
   122→    // 1 — Perencanaan
   123→    milestones.length > 0
   124→      ? `${milestones.length} milestone direncanakan`
   125→      : "Belum ada milestone — tambahkan rencana kerja",
   126→    // 2 — Produksi
   127→    milestones.length > 0
   128→      ? `${doneMs}/${milestones.length} milestone selesai · progress ${project.progress}%`
   129→      : `Progress ${project.progress}%`,
   130→    // 3 — Review
   131→    deliverables === null
   132→      ? "Memuat deliverable…"
   133→      : dlv.length > 0
   134→        ? `${dlv.length} deliverable · ${dlvApproved} disetujui${dlvPending ? ` · ${dlvPending} menunggu` : ""}${dlvRevision ? ` · ${dlvRevision} revisi` : ""}`
   135→        : "Belum ada deliverable dikirim",
   136→    // 4 — Serah Terima & Penagihan
   137→    invoices === null
   138→      ? "Memuat invoice…"
   139→      : invoices.length > 0
   140→        ? `${invoices.length} invoice · terbit ${formatCurrency(invTotal)}${paidTotal > 0 ? ` · dibayar ${formatCurrency(paidTotal)}` : ""}`
   141→        : approvedCr.length > 0
   142→          ? `${approvedCr.length} CR disetujui — invoice tambahan terbit`
   143→          : "Belum ada invoice project",
   144→  ];
   145→
   146→  return (
   147→    <div className="rounded-xl border bg-white p-3 shadow-sm">
   148→      <p className="flex items-center gap-1.5 text-xs font-medium uppercase tracking-wide text-zinc-500">
   149→        <Route className="h-3.5 w-3.5" aria-hidden /> Alur Produksi
   150→        {currentIdx === 5 ? (
   151→          <Badge variant="outline" className="ml-auto border-transparent bg-emerald-100 text-[11px] text-emerald-700">Alur selesai</Badge>
   152→        ) : (
   153→          <span className="ml-auto text-[11px] font-normal normal-case text-zinc-400">
   154→            Tahap {currentIdx + 1}/5 · {PRODUCTION_STEPS[currentIdx]}
   155→          </span>
   156→        )}
   157→      </p>
   158→      <ol className="mt-2.5 flex items-start gap-0" aria-label="Alur produksi project">
   159→        {PRODUCTION_STEPS.map((label, idx) => {
   160→          const isDone = idx < currentIdx;
   161→          const isCurrent = idx === currentIdx;
   162→          return (
   163→            <li key={label} className="flex min-w-0 flex-1 flex-col items-center text-center" aria-current={isCurrent ? "step" : undefined}>
   164→              <div className="flex w-full items-center" aria-hidden>
   165→                <span className={cn("h-0.5 flex-1", idx === 0 ? "bg-transparent" : isDone || isCurrent ? "bg-emerald-400" : "bg-zinc-200")} />
   166→                <span
   167→                  className={cn(
   168→                    "flex size-5 shrink-0 items-center justify-center rounded-full border-2 text-[9px] font-bold",
   169→                    isDone ? "border-emerald-500 bg-emerald-500 text-white"
   170→                      : isCurrent ? "border-emerald-500 bg-white text-emerald-600"
   171→                      : "border-zinc-200 bg-white text-zinc-300",
   172→                  )}
   173→                >
   174→                  {isDone ? <Check className="h-3 w-3" /> : idx + 1}
   175→                </span>
   176→                <span className={cn("h-0.5 flex-1", idx === PRODUCTION_STEPS.length - 1 ? "bg-transparent" : isDone ? "bg-emerald-400" : "bg-zinc-200")} />
   177→              </div>
   178→              <p className={cn("mt-1 px-0.5 text-[10px] font-semibold leading-tight", isDone || isCurrent ? "text-zinc-800" : "text-zinc-400")}>
   179→                {label}
   180→              </p>
   181→              <p className={cn("mt-0.5 px-0.5 text-[9px] leading-tight", isDone || isCurrent ? "text-zinc-500" : "text-zinc-300")}>
   182→                {stepRecords[idx]}
   183→              </p>
   184→            </li>
   185→          );
   186→        })}
   187→      </ol>
   188→    </div>
   189→  );
   190→}

   216→const PROJECT_CSV_HEADERS = [
   217→  "kode", "nama", "brand", "klien", "kategori_layanan", "status", "progress_pct",
   218→  "pm", "mulai", "deadline", "nilai_kontrak", "budget_internal", "milestone_selesai", "milestone_total",

   221→function escapeCsvField(value: string): string {
   222→  if (/[;"\r\n]/.test(value)) return `"${value.replace(/"/g, '""')}"`;
   223→  return value;
   224→}
   225→
   226→function buildProjectsCsv(projects: ProjectDTO[]): string {
   227→  const lines: string[] = [PROJECT_CSV_HEADERS.join(";")];
   228→  for (const p of projects) {
   229→    const ms = p.milestones ?? [];
   230→    const fields: string[] = [
   231→      p.code,
   232→      p.name,
   233→      p.brand?.name ?? "",
   234→      p.company?.name ?? "",
   235→      p.serviceCategory ?? "",
   236→      statusMeta(p.status).label,
   237→      String(p.progress),
   238→      p.pmName ?? "",
   239→      p.startDate ? formatDate(p.startDate) : "",
   240→      p.dueDate ? formatDate(p.dueDate) : "",
   241→      p.contractValue != null ? String(p.contractValue) : "",
   242→      p.budgetInternal != null ? String(p.budgetInternal) : "",
   243→      String(ms.filter((m) => m.status === "done").length),
   244→      String(ms.length),
   245→    ];
   246→    lines.push(fields.map(escapeCsvField).join(";"));
   247→  }
   248→  return "\uFEFF" + lines.join("\r\n");
   249→}

   259→function dueSoon(p: ProjectDTO): boolean {
   260→  if (p.status === "completed" || !p.dueDate) return false;
   261→  const diffDays = (new Date(p.dueDate).getTime() - Date.now()) / (24 * 60 * 60 * 1000);
   262→  return diffDays < 7;
   263→}

   374→function TimelineRow({ project, pct, onOpen }: {
   375→  project: ProjectDTO;
   376→  pct: (ts: number) => number;
   377→  onOpen: (p: ProjectDTO) => void;
   378→}) {
   379→  const barCls =
   380→    project.status === "in_progress" ? "bg-zinc-900/90"
   381→    : project.status === "review" ? "bg-violet-500"
   382→    : project.status === "completed" ? "bg-emerald-500"
   383→    : "bg-zinc-400";
   384→  const startTs = project.startDate ? new Date(project.startDate).getTime() : NaN;
   385→  const endTs = project.dueDate ? new Date(project.dueDate).getTime() : NaN;
   386→  const hasStart = Number.isFinite(startTs);
   387→  const hasEnd = Number.isFinite(endTs);
   388→  const leftPct = hasStart ? pct(startTs) : null;
   389→  const widthPct = hasStart && hasEnd ? Math.max(pct(endTs) - pct(startTs), 2) : null;
   390→  const barWidth = widthPct !== null ? Math.min(widthPct, 100 - (leftPct ?? 0)) : null;
   391→  const barTitle = `${project.name} · ${formatDate(project.startDate)} → ${formatDate(project.dueDate)} · ${project.progress}%`;
   392→
   393→  return (
   394→    <button
   395→      type="button"
   396→      onClick={() => onOpen(project)}
   397→      aria-label={`Buka detail project ${project.name}`}
   398→      className="flex h-11 w-full border-t border-zinc-100 text-left transition-colors hover:bg-zinc-50"
   399→    >
   400→      <div className="flex w-[200px] shrink-0 items-center gap-2 overflow-hidden pr-3">
   401→        <span
   402→          className="h-2 w-2 shrink-0 rounded-full"
   403→          style={{ backgroundColor: project.brand?.color ?? "#a1a1aa" }}
   404→          aria-hidden
   405→        />
   406→        <div className="min-w-0">
   407→          <p className="truncate font-mono text-[11px] leading-tight text-zinc-500">{project.code}</p>
   408→          <p className="truncate text-sm leading-tight text-zinc-800">{project.name}</p>
   409→        </div>
   410→      </div>
   411→      <div className="relative flex-1">
   412→        {leftPct !== null && barWidth !== null ? (
   413→          <div
   414→            className={`absolute top-1/2 h-5 -translate-y-1/2 rounded-full ${barCls}`}
   415→            style={{ left: `${leftPct}%`, width: `${barWidth}%` }}
   416→            title={barTitle}
   417→          >
   418→            {widthPct !== null && widthPct >= 14 ? (
   419→              <span className="absolute inset-y-0 right-1.5 flex items-center text-[10px] font-semibold text-white">
   420→                {project.progress}%
   421→              </span>
   422→            ) : null}
   423→          </div>
   424→        ) : null}
   425→        {leftPct !== null && barWidth !== null && widthPct !== null && widthPct < 14 ? (
   426→          <span
   427→            className="absolute top-1/2 -translate-y-1/2 whitespace-nowrap text-[10px] font-medium text-zinc-500"
   428→            style={{ left: `calc(${leftPct + barWidth}% + 6px)` }}
   429→          >
   430→            {project.progress}%
   431→          </span>
   432→        ) : null}

   471→function TimelineView({ projects, onOpen }: { projects: ProjectDTO[]; onOpen: (p: ProjectDTO) => void }) {
   472→  const range = useMemo(() => {
   473→    const now = Date.now();
   474→    let min: number | null = null;
   475→    let max: number | null = null;
   476→    for (const p of projects) {
   477→      let contributed = false;
   478→      const track = (raw?: string | Date | number | null) => {
   479→        if (!raw) return;
   480→        const ts = new Date(raw).getTime();
   481→        if (!Number.isFinite(ts)) return;
   482→        min = min === null ? ts : Math.min(min, ts);
   483→        max = max === null ? ts : Math.max(max, ts);
   484→        contributed = true;
   485→      };
   486→      track(p.startDate);
   487→      track(p.dueDate);
   488→      for (const m of p.milestones ?? []) track(m.dueDate);
   489→      if (!contributed) track(now); // fallback: project tanpa tanggal → acuan hari ini
   490→    }
   491→    if (min === null || max === null) return { min: now - 15 * DAY_MS, max: now + 15 * DAY_MS };
   492→    const pad = 7 * DAY_MS;
   493→    return { min: min - pad, max: max + pad };
   494→  }, [projects]);
   495→
   496→  const span = Math.max(range.max - range.min, DAY_MS);
   497→  const pct = useCallback((ts: number) => Math.max(0, Math.min(100, ((ts - range.min) / span) * 100)), [range.min, span]);
   498→  const nowPct = pct(Date.now());
   499→
   500→  const gridlines = useMemo(() => {
   501→    const n = 6;
   502→    return Array.from({ length: n }, (_, i) => {
   503→      const ratio = i / (n - 1);
   504→      return { pct: ratio * 100, label: shortDate(range.min + ratio * span) };
   505→    });
   506→  }, [range.min, span]);
   507→
   508→  const todayTransform = nowPct <= 5 ? "translateX(0)" : nowPct >= 95 ? "translateX(-100%)" : "translateX(-50%)";

   510→  return (

   518→              <span
   519→                key={i}
   520→                className="absolute top-0 whitespace-nowrap text-[10px] tabular-nums text-zinc-400"
   521→                style={{
   522→                  left: `${g.pct}%`,
   523→                  transform: i === 0 ? "translateX(0)" : i === gridlines.length - 1 ? "translateX(-100%)" : "translateX(-50%)",
   524→                }}
   525→              >
   526→                {g.label}
   527→              </span>
   528→            ))}
   529→            <span
   530→              className="absolute top-0 whitespace-nowrap text-[10px] font-semibold text-rose-500"
   531→              style={{ left: `${nowPct}%`, transform: todayTransform }}
   532→            >
   533→              Hari ini
   534→            </span>
   535→          </div>
   536→        </div>
   537→
   538→        {/* Baris project + gridline vertikal */}
   539→        <div className="relative">
   540→          <div className="pointer-events-none absolute inset-y-0 left-[200px] right-0" aria-hidden>
   541→            {gridlines.map((g, i) => (

   554→          <span className="flex items-center gap-1.5"><span className="h-2.5 w-5 rounded-full bg-zinc-900/90" aria-hidden /> Berjalan</span>

   556→          <span className="flex items-center gap-1.5"><span className="h-2.5 w-5 rounded-full bg-emerald-500" aria-hidden /> Selesai</span>

   578→function dateKey(d: Date): string {

   584→/** Format Date → nilai untuk <input type="date"> (yyyy-mm-dd lokal). */
   585→function toDateInputValue(d: Date | null): string {
   586→  if (!d || Number.isNaN(d.getTime())) return "";
   587→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
   588→}

   590→/** Parse nilai <input type="date"> → Date lokal pukul 12:00 (aman dari geser tanggal). */
   591→function parseDateInputValue(v: string): Date | null {
   592→  const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(v.trim());
   593→  if (!m) return null;
   594→  const d = new Date(Number(m[1]), Number(m[2]) - 1, Number(m[3]), 12, 0, 0, 0);
   595→  return Number.isNaN(d.getTime()) ? null : d;
   596→}

   598→function isSameLocalDay(a: Date, b: Date): boolean {
   599→  return a.getFullYear() === b.getFullYear() && a.getMonth() === b.getMonth() && a.getDate() === b.getDate();
