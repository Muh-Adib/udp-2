   185→function TimelineRow({ project, pct, onOpen }: {
   186→  project: ProjectDTO;
   187→  pct: (ts: number) => number;
   188→  onOpen: (p: ProjectDTO) => void;
   189→}) {
   190→  const barCls =
   191→    project.status === "in_progress" ? "bg-zinc-900/90"
   192→    : project.status === "review" ? "bg-violet-500"
   193→    : project.status === "completed" ? "bg-emerald-500"
   194→    : "bg-zinc-400";
   195→  const startTs = project.startDate ? new Date(project.startDate).getTime() : NaN;
   196→  const endTs = project.dueDate ? new Date(project.dueDate).getTime() : NaN;
   197→  const hasStart = Number.isFinite(startTs);
   198→  const hasEnd = Number.isFinite(endTs);
   199→  const leftPct = hasStart ? pct(startTs) : null;
   200→  const widthPct = hasStart && hasEnd ? Math.max(pct(endTs) - pct(startTs), 2) : null;
   201→  const barWidth = widthPct !== null ? Math.min(widthPct, 100 - (leftPct ?? 0)) : null;
   202→  const barTitle = `${project.name} · ${formatDate(project.startDate)} → ${formatDate(project.dueDate)} · ${project.progress}%`;
   203→
   204→  return (
   205→    <button
   206→      type="button"
   207→      onClick={() => onOpen(project)}
   208→      aria-label={`Buka detail project ${project.name}`}
   209→      className="flex h-11 w-full border-t border-zinc-100 text-left transition-colors hover:bg-zinc-50"
   210→    >
   211→      <div className="flex w-[200px] shrink-0 items-center gap-2 overflow-hidden pr-3">
   212→        <span
   213→          className="h-2 w-2 shrink-0 rounded-full"
   214→          style={{ backgroundColor: project.brand?.color ?? "#a1a1aa" }}
   215→          aria-hidden
   216→        />
   217→        <div className="min-w-0">
   218→          <p className="truncate font-mono text-[11px] leading-tight text-zinc-500">{project.code}</p>
   219→          <p className="truncate text-sm leading-tight text-zinc-800">{project.name}</p>
   220→        </div>
   221→      </div>
   222→      <div className="relative flex-1">
   223→        {leftPct !== null && barWidth !== null ? (
   224→          <div
   225→            className={`absolute top-1/2 h-5 -translate-y-1/2 rounded-full ${barCls}`}
   226→            style={{ left: `${leftPct}%`, width: `${barWidth}%` }}
   227→            title={barTitle}
   228→          >
   229→            {widthPct !== null && widthPct >= 14 ? (
   230→              <span className="absolute inset-y-0 right-1.5 flex items-center text-[10px] font-semibold text-white">
   231→                {project.progress}%
   232→              </span>
   233→            ) : null}
   234→          </div>
   235→        ) : null}
   236→        {leftPct !== null && barWidth !== null && widthPct !== null && widthPct < 14 ? (
   237→          <span
   238→            className="absolute top-1/2 -translate-y-1/2 whitespace-nowrap text-[10px] font-medium text-zinc-500"
   239→            style={{ left: `calc(${leftPct + barWidth}% + 6px)` }}
   240→          >
   241→            {project.progress}%
   242→          </span>
   243→        ) : null}
   244→        {leftPct !== null && barWidth === null ? (
   245→          <span
   246→            className={`absolute top-1/2 h-2.5 w-2.5 -translate-y-1/2 rounded-full ${barCls}`}
   247→            style={{ left: `${leftPct}%` }}
   248→            title={barTitle}
   249→          />
   250→        ) : null}
   251→        {!hasStart && !hasEnd ? (
   252→          <span className="absolute left-2 top-1/2 -translate-y-1/2 text-[11px] text-zinc-400">
   253→            Tanpa tanggal
   254→          </span>
   255→        ) : null}
   256→        {(project.milestones ?? []).map((m) => {
   257→          if (!m.dueDate) return null;
   258→          const ts = new Date(m.dueDate).getTime();
   259→          if (!Number.isFinite(ts)) return null;
   260→          const dCls =
   261→            m.status === "done" ? "bg-emerald-500"
   262→            : m.status === "in_progress" ? "bg-amber-500"
   263→            : "bg-zinc-300";

   266→              key={m.id}
   267→              className={`absolute h-2.5 w-2.5 rounded-[2px] ${dCls}`}
   268→              style={{
   269→                left: `${pct(ts)}%`,

   271→                transform: "translate(-50%, -50%) rotate(45deg)",
   272→              }}
   273→              title={`${m.name} · ${msMeta(m.status).label}`}
   274→            />
   275→          );
   276→        })}
   277→      </div>
   278→    </button>

   282→function TimelineView({ projects, onOpen }: { projects: ProjectDTO[]; onOpen: (p: ProjectDTO) => void }) {
   283→  const range = useMemo(() => {
   284→    const now = Date.now();
   285→    let min: number | null = null;
   286→    let max: number | null = null;
   287→    for (const p of projects) {
   288→      let contributed = false;
   289→      const track = (raw?: string | Date | number | null) => {
   290→        if (!raw) return;
   291→        const ts = new Date(raw).getTime();
   292→        if (!Number.isFinite(ts)) return;
   293→        min = min === null ? ts : Math.min(min, ts);
   294→        max = max === null ? ts : Math.max(max, ts);
   295→        contributed = true;
   296→      };
   297→      track(p.startDate);
   298→      track(p.dueDate);
   299→      for (const m of p.milestones ?? []) track(m.dueDate);
   300→      if (!contributed) track(now); // fallback: project tanpa tanggal → acuan hari ini
   301→    }
   302→    if (min === null || max === null) return { min: now - 15 * DAY_MS, max: now + 15 * DAY_MS };
   303→    const pad = 7 * DAY_MS;
   304→    return { min: min - pad, max: max + pad };
   305→  }, [projects]);
   306→
   307→  const span = Math.max(range.max - range.min, DAY_MS);
   308→  const pct = useCallback((ts: number) => Math.max(0, Math.min(100, ((ts - range.min) / span) * 100)), [range.min, span]);
   309→  const nowPct = pct(Date.now());
   310→
   311→  const gridlines = useMemo(() => {
   312→    const n = 6;
   313→    return Array.from({ length: n }, (_, i) => {
   314→      const ratio = i / (n - 1);
   315→      return { pct: ratio * 100, label: shortDate(range.min + ratio * span) };
   316→    });
   317→  }, [range.min, span]);
   318→
   319→  const todayTransform = nowPct <= 5 ? "translateX(0)" : nowPct >= 95 ? "translateX(-100%)" : "translateX(-50%)";
   320→
   321→  return (
   322→    <div className="overflow-x-auto crm-scroll rounded-xl border bg-white p-4 shadow-sm">
   323→      <div className="min-w-[760px]">
   324→        {/* Sumbu waktu (header) */}
   325→        <div className="flex">
   326→          <div className="w-[200px] shrink-0" />
   327→          <div className="relative h-7 flex-1">
   328→            {gridlines.map((g, i) => (
   329→              <span
   330→                key={i}
   331→                className="absolute top-0 whitespace-nowrap text-[10px] tabular-nums text-zinc-400"
   332→                style={{
   333→                  left: `${g.pct}%`,
   334→                  transform: i === 0 ? "translateX(0)" : i === gridlines.length - 1 ? "translateX(-100%)" : "translateX(-50%)",
   335→                }}
   336→              >
   337→                {g.label}
   338→              </span>
   339→            ))}
   340→            <span
   341→              className="absolute top-0 whitespace-nowrap text-[10px] font-semibold text-rose-500"
   342→              style={{ left: `${nowPct}%`, transform: todayTransform }}
   343→            >
   344→              Hari ini
   345→            </span>
   346→          </div>
   347→        </div>
   348→
   349→        {/* Baris project + gridline vertikal */}
   350→        <div className="relative">
   351→          <div className="pointer-events-none absolute inset-y-0 left-[200px] right-0" aria-hidden>
   352→            {gridlines.map((g, i) => (
   353→              <div key={i} className="absolute inset-y-0 border-l border-dashed border-zinc-200" style={{ left: `${g.pct}%` }} />
   354→            ))}
   355→            <div className="absolute inset-y-0 border-l-2 border-rose-400" style={{ left: `${nowPct}%` }} />
   356→          </div>
   357→          {projects.map((p) => (
   358→            <TimelineRow key={p.id} project={p} pct={pct} onOpen={onOpen} />
   359→          ))}
   360→        </div>
   361→
   362→        {/* Legenda */}
   363→        <div className="mt-3 flex flex-wrap items-center gap-x-4 gap-y-1.5 border-t border-zinc-100 pt-3 text-[11px] text-zinc-500">
   364→          <span className="flex items-center gap-1.5"><span className="h-2.5 w-5 rounded-full bg-zinc-400" aria-hidden /> Perencanaan</span>
   365→          <span className="flex items-center gap-1.5"><span className="h-2.5 w-5 rounded-full bg-zinc-900/90" aria-hidden /> Berjalan</span>
   366→          <span className="flex items-center gap-1.5"><span className="h-2.5 w-5 rounded-full bg-violet-500" aria-hidden /> Review</span>
   367→          <span className="flex items-center gap-1.5"><span className="h-2.5 w-5 rounded-full bg-emerald-500" aria-hidden /> Selesai</span>
   368→          <span className="flex items-center gap-1.5"><span className="h-2 w-2 rotate-45 rounded-[2px] bg-emerald-500" aria-hidden /> Milestone selesai</span>
   369→          <span className="flex items-center gap-1.5"><span className="h-2 w-2 rotate-45 rounded-[2px] bg-amber-500" aria-hidden /> Dikerjakan</span>
   370→          <span className="flex items-center gap-1.5"><span className="h-2 w-2 rotate-45 rounded-[2px] bg-zinc-300" aria-hidden /> Menunggu</span>
   371→          <span className="flex items-center gap-1.5"><span className="h-3.5 border-l-2 border-rose-400" aria-hidden /> Hari ini</span>
   372→        </div>
   373→      </div>
   374→    </div>
   375→  );
   376→}
