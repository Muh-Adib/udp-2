    41→function channelMeta(channel: string): { label: string; icon: LucideIcon } {
    42→  if (CHANNEL_META[channel]) return CHANNEL_META[channel];
    43→  return {
    44→    label: channel ? channel.charAt(0).toUpperCase() + channel.slice(1) : "Lainnya",
    45→    icon: BellRing,
    46→  };

    60→function slugify(name: string): string {
    61→  return name
    62→    .toLowerCase()
    63→    .replace(/[^a-z0-9]+/g, "_")
    64→    .replace(/^_+|_+$/g, "");
    65→}
    66→
    67→function quotePrefixFrom(prefix: string): string {
    68→  const clean = prefix.trim().replace(/[^a-z0-9]/gi, "").slice(0, 3).toUpperCase();
    69→  return clean ? `Q${clean}` : "";
    70→}

   267→export default function BrandsModule() {
   268→  const user = useCrmStore((s) => s.user);
   269→
   270→  const [brands, setBrands] = useState<Brand[] | null>(null);
   271→  const [loading, setLoading] = useState(true);
   272→  const [error, setError] = useState<string | null>(null);
   273→  const [serviceFilter, setServiceFilter] = useState("");
   274→
   275→  const [createOpen, setCreateOpen] = useState(false);
   276→  const [draft, setDraft] = useState<BrandDraft>(EMPTY_DRAFT);
   277→  const [submitting, setSubmitting] = useState(false);
   278→
   279→  const [templates, setTemplates] = useState<FollowUpTemplateDTO[] | null>(null);
   280→  const [templatesLoading, setTemplatesLoading] = useState(true);
   281→  const [templatesError, setTemplatesError] = useState<string | null>(null);
   282→  const [channelFilter, setChannelFilter] = useState("");
   283→
   284→  const load = useCallback(async (silent = false) => {
   285→    if (!silent) setLoading(true);
   286→    setError(null);
   287→    try {
   288→      const res = await api.brands();
   289→      setBrands(res.brands);
   290→    } catch (err) {
   291→      setError(err instanceof Error ? err.message : "Gagal memuat brand");
   292→      if (!silent) toast.error("Gagal memuat konfigurasi brand");
   293→    } finally {
   294→      setLoading(false);
   295→    }
   296→  }, []);
   297→
   298→  const loadTemplates = useCallback(async () => {
   299→    setTemplatesLoading(true);
   300→    setTemplatesError(null);
   301→    try {
   302→      const res = await api.followUpTemplates();
   303→      setTemplates(res.templates);
   304→    } catch (err) {
   305→      setTemplatesError(err instanceof Error ? err.message : "Gagal memuat template follow-up");
   306→    } finally {
   307→      setTemplatesLoading(false);
   308→    }
   309→  }, []);
   310→
   311→  useEffect(() => {
   312→    void load();
   313→    void loadTemplates();
   314→  }, [load, loadTemplates]);
   315→
   316→  const allServices = useMemo(() => {
   317→    const set = new Set<string>();
   318→    Object.values(BRAND_SERVICES).forEach((arr) => arr.forEach((s) => set.add(s)));
   319→    return Array.from(set);
   320→  }, []);
   321→
   322→  const filteredBrands = useMemo(() => {
   323→    if (!serviceFilter || serviceFilter === "all") return brands ?? [];
   324→    return (brands ?? []).filter((b) => (BRAND_SERVICES[b.slug] ?? []).includes(serviceFilter));
   325→  }, [brands, serviceFilter]);
   326→
   327→  const templateChannels = useMemo(() => {
   328→    const set = new Set<string>();
   329→    (templates ?? []).forEach((t) => set.add(t.channel));
   330→    return Array.from(set).sort().map((c) => ({ key: c, label: channelMeta(c).label }));
   331→  }, [templates]);
   332→
   333→  const filteredTemplates = useMemo(() => {
   334→    if (!templates) return [];
   335→    if (!channelFilter || channelFilter === "all") return templates;
   336→    return templates.filter((t) => t.channel === channelFilter);
   337→  }, [templates, channelFilter]);
   338→
   339→  const brandNameById = useMemo(() => {
   340→    const map = new Map<string, string>();
   341→    (brands ?? []).forEach((b) => map.set(b.id, b.name));
   342→    return map;
   343→  }, [brands]);
   344→
   345→  function handleNameChange(name: string) {
   346→    setDraft((d) => ({ ...d, name, slug: slugify(name) }));
   347→  }
   348→
   349→  async function submitDraft(e: React.FormEvent<HTMLFormElement>) {
   350→    e.preventDefault();
   351→    if (!draft.name.trim()) { toast.error("Nama brand wajib diisi"); return; }
   352→    if (!draft.slug.trim()) { toast.error("Slug brand wajib diisi"); return; }
   353→    const sla = Number(draft.slaHours);
   354→    if (!Number.isFinite(sla) || sla < 1 || sla > 72) { toast.error("SLA harus antara 1 dan 72 jam"); return; }
   355→    if (!user) { toast.error("Sesi berakhir — silakan login ulang"); return; }
   356→    setSubmitting(true);
   357→    try {
   358→      const res = await api.createBrand({
   359→        name: draft.name.trim(),
   360→        slug: draft.slug.trim(),
   361→        color: draft.color,
   362→        logoEmoji: draft.logoEmoji.trim() || "✨",
   363→        description: draft.description.trim(),
   364→        website: "",
   365→        primaryCurrency: draft.primaryCurrency,
   366→        invoicePrefix: draft.invoicePrefix.trim(),
   367→        quotePrefix: quotePrefixFrom(draft.invoicePrefix),
   368→        slaHours: sla,
   369→        portalDomain: "",
   370→        actorName: user.name,
   371→        actorRole: user.role,
   372→      });
   373→      toast.success(`Brand ${res.brand.name} dikonfigurasi`);
   374→      setDraft(EMPTY_DRAFT);
   375→      setCreateOpen(false);
   376→      await load(true);
   377→    } catch (err) {
   378→      toast.error(err instanceof Error ? err.message : "Gagal menyimpan brand");
   379→    } finally {
   380→      setSubmitting(false);
   381→    }
   382→  }
   383→
   384→  if (loading && brands === null) return <BrandsSkeleton />;
   385→
   386→  return (
   387→    <div className="space-y-6">
   388→      {/* Header */}
   389→      <div className="flex flex-col gap-3 sm:flex-row sm:items-start sm:justify-between">
   390→        <div>
   391→          <h1 className="text-xl font-bold tracking-tight text-zinc-900">Konfigurasi Brand</h1>
   392→          <p className="text-sm text-zinc-500">Tambah brand baru tanpa mengubah source code</p>
   393→        </div>
   394→        <div className="flex items-center gap-2">
   395→          <Button variant="outline" size="sm" onClick={() => void load()} disabled={loading} aria-label="Muat ulang daftar brand">
   396→            <RefreshCw className={`h-4 w-4 ${loading ? "animate-spin" : ""}`} aria-hidden /> Muat ulang
   397→          </Button>
   398→          <Button size="sm" onClick={() => setCreateOpen(true)} aria-label="Tambah brand baru">
   399→            <Plus className="h-4 w-4" aria-hidden /> Tambah Brand
   400→          </Button>
   401→        </div>
   402→      </div>
   403→
   404→      {error ? (
   405→        <div className="rounded-xl border border-rose-200 bg-rose-50 p-4 text-sm text-rose-700">{error}</div>
   406→      ) : null}
   407→
   408→      {/* Filter layanan */}
   409→      <div className="flex flex-col gap-3 rounded-xl border bg-white p-4 shadow-sm sm:flex-row sm:items-center">
   410→        <div className="flex items-center gap-2">
   411→          <Layers className="h-4 w-4 shrink-0 text-zinc-400" aria-hidden />
   412→          <Select value={serviceFilter} onValueChange={setServiceFilter}>
   413→            <SelectTrigger className="w-full sm:w-[260px]" aria-label="Filter brand berdasarkan layanan">
   414→              <SelectValue placeholder="Semua layanan" />
   415→            </SelectTrigger>
   416→            <SelectContent>
   417→              <SelectItem value="all">Semua Layanan</SelectItem>
   418→              {allServices.map((s) => <SelectItem key={s} value={s}>{s}</SelectItem>)}
   419→            </SelectContent>
   420→          </Select>
   421→        </div>
   422→        <span className="text-xs text-zinc-500 sm:ml-auto">{filteredBrands.length} dari {brands?.length ?? 0} brand</span>
   423→      </div>
   424→
   425→      {/* Grid brand */}
   426→      <div className="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
   427→        {filteredBrands.map((b) => <BrandCard key={b.id} brand={b} />)}
   428→
   429→        {/* Kartu tambah */}
   430→        <button
   431→          type="button"
   432→          onClick={() => setCreateOpen(true)}
   433→          aria-label="Tambah brand baru"
   434→          className="flex min-h-[220px] flex-col items-center justify-center gap-2 rounded-xl border-2 border-dashed border-zinc-300 p-5 text-zinc-400 transition-colors hover:border-zinc-400 hover:text-zinc-600"
   435→        >
   436→          <Plus className="h-8 w-8" aria-hidden />
   437→          <span className="text-sm font-semibold">Tambah Brand</span>
   438→          <span className="max-w-[220px] text-center text-xs">Definisikan brand, warna, SLA, dan prefix invoice</span>
   439→        </button>
   440→      </div>
   441→
   442→      {/* Konfigurasi pipeline standar */}
   443→      <section aria-label="Konfigurasi pipeline standar" className="rounded-xl border bg-white p-4 shadow-sm sm:p-6">
   444→        <div className="mb-4">
   445→          <p className="flex items-center gap-1.5 text-xs font-semibold uppercase tracking-wider text-zinc-500">
   446→            <Wand2 className="h-3.5 w-3.5" aria-hidden /> Standar Tim Sales
   447→          </p>
   448→          <h2 className="mt-1 text-sm font-semibold text-zinc-900">Konfigurasi Pipeline Standar (12 Stage)</h2>
   449→          <p className="mt-0.5 text-xs text-zinc-500">
   450→            Acuan baku dari lead baru hingga deal — dipakai sebagai acuan aktivitas follow-up di semua brand.
   451→          </p>
   452→        </div>
   453→        <div className="grid grid-cols-1 gap-3 md:grid-cols-2 xl:grid-cols-3">
   454→          {PIPELINE_STAGES.map((s, i) => <PipelineStageCard key={s.key} stage={s} index={i} />)}
   455→        </div>
   456→      </section>
   457→
   458→      {/* Template follow-up */}
   459→      <section aria-label="Template follow-up" className="rounded-xl border bg-white p-4 shadow-sm sm:p-6">
   460→        <div className="mb-4 flex flex-col gap-3 sm:flex-row sm:items-end sm:justify-between">
   461→          <div>

   579→                <Label htmlFor="brand-currency">Mata Uang</Label>

   581→                  <SelectTrigger id="brand-currency"><SelectValue /></SelectTrigger>
   582→                  <SelectContent>
   583→                    {CURRENCIES.map((c) => <SelectItem key={c} value={c}>{c}</SelectItem>)}
   584→                  </SelectContent>
   585→                </Select>
