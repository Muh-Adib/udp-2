   393→        setError("Waktu muat habis — periksa koneksi lalu coba lagi");

   395→        setError(err instanceof Error ? err.message : "Gagal memuat peta layanan");

   398→      clearTimeout(timer);
   399→      setLoading(false);

   435→            <MapIcon className="h-5 w-5" />
   436→          </span>
