   234→function hashColor(name: string): string {
   235→  let h = 0;
   236→  for (let i = 0; i < name.length; i++) h = (h * 31 + name.charCodeAt(i)) >>> 0;
   237→  return AVATAR_PALETTE[h % AVATAR_PALETTE.length];
   238→}

   240→function useDebounced<T>(value: T, delay = 300): T {
   241→  const [debounced, setDebounced] = useState(value);
   242→  useEffect(() => {
   243→    const t = setTimeout(() => setDebounced(value), delay);
   244→    return () => clearTimeout(t);
   245→  }, [value, delay]);
   246→  return debounced;
   247→}

   257→function consentMeta(status: string): { label: string; className: string } {
   258→  if (status === "granted") return { label: "Consent: Disetujui", className: "bg-emerald-100 text-emerald-700" };
   259→  if (status === "pending") return { label: "Consent: Menunggu", className: "bg-amber-100 text-amber-700" };
   260→  if (status === "declined") return { label: "Consent: Ditolak", className: "bg-rose-100 text-rose-700" };
   261→  return { label: status, className: "bg-zinc-100 text-zinc-600" };
   262→}

   264→function sizeBadgeClass(size?: string | null): string {
   265→  switch (size) {
   266→    case "enterprise":
   267→      return "bg-violet-100 text-violet-700";
   268→    case "government":
   269→      return "bg-zinc-200 text-zinc-800";
   270→    case "sme":
   271→      return "bg-emerald-100 text-emerald-700";
   272→    case "startup":
   273→      return "bg-orange-100 text-orange-700";
   274→    default:
   275→      return "bg-zinc-100 text-zinc-600";
   276→  }

   647→function ContactDetailBody({

   655→  companies: CompanyRecord[];

   690→  async function handleSave(e: FormEvent<HTMLFormElement>) {

   710→        actorRole: user?.role ?? "system",
   711→      });

   714→      onSaved(res.contact);

   719→    }

   722→  async function patchTags(next: string[]) {

   725→      const res = await api.updateContact(contact.id, {
   726→        tags: next,
   727→        actorName: user?.name ?? "System",
   728→        actorRole: user?.role ?? "system",
   729→      });
   730→      setTags(next);
   731→      onSaved(res.contact);
   732→    } catch (err) {
   733→      toast.error(err instanceof Error ? err.message : "Gagal memperbarui tag");
   734→    } finally {
   735→      setTagBusy(false);
   736→    }

   739→  function addTag() {

   742→      setTagInput("");
   743→      return;
   744→    }
   745→    setTagInput("");

   747→  }

   749→  function removeTag(t: string) {

   753→  async function handleDelete() {

   763→      }
   764→      toast.success(`Contact "${contact.fullName}" dihapus`);
   765→      onDeleted();
   766→    } catch (err) {
   767→      toast.error(err instanceof Error ? err.message : "Gagal menghapus contact");
   768→      setDeleting(false);
   769→    }
   770→  }

   772→  return (
   773→    <div className="flex flex-1 flex-col">
   774→      <SheetHeader className="border-b border-zinc-100">
   775→        <div className="flex items-start gap-3 pr-8">
   776→          <AvatarBubble name={contact.fullName} size="lg" />

   790→            <Pencil className="size-3.5" /> {editing ? "Batal" : "Edit"}
   791→          </Button>
   792→        </div>
   793→      </SheetHeader>
   794→
   795→      <div className="flex-1 space-y-5 p-4">
   796→        {editing ? (
   797→          <form onSubmit={handleSave} className="space-y-4 rounded-xl border bg-white p-4 shadow-sm">
   798→            <p className="text-sm font-semibold text-zinc-900">Edit Contact</p>
   799→            <ContactFormFields
   800→              mode="edit"
   801→              values={form}
   802→              onChange={patchForm}
   803→              disabled={saving}
   804→              companies={companies}
   805→            />
   806→            <div className="flex justify-end gap-2">
   807→              <Button type="button" variant="outline" onClick={() => setEditing(false)} disabled={saving}>
   808→                Batal
   809→              </Button>
   810→              <Button type="submit" className="bg-zinc-900 text-white hover:bg-zinc-800" disabled={saving}>
   811→                {saving && <Loader2 className="size-4 animate-spin" />} Simpan Perubahan
   812→              </Button>
   813→            </div>
   814→          </form>
   815→        ) : (
   816→          <section aria-label="Informasi contact">
   817→            <SectionTitle>Informasi</SectionTitle>
   818→            <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   819→              <InfoRow icon={Mail} label="Email" value={contact.email} />
   820→              {contact.emailAlt ? <InfoRow icon={MailPlus} label="Email alternatif" value={contact.emailAlt} /> : null}
   821→              <InfoRow icon={MessageCircle} label="WhatsApp" value={contact.whatsapp} />

   828→              <div className="flex items-start gap-3">
   829→                <span className="mt-0.5 flex size-8 shrink-0 items-center justify-center rounded-lg bg-zinc-100 text-zinc-500">
   830→                  <ShieldCheck className="size-4" />
   831→                </span>
   832→                <div>

  1484→/** Deteksi separator dari baris pertama: titik koma (hasil ekspor Excel locale Indonesia) atau koma. */
  1485→function detectCsvSeparator(firstLine: string): string {
  1486→  const semis = (firstLine.match(/;/g) ?? []).length;
  1487→  const commas = (firstLine.match(/,/g) ?? []).length;
  1488→  return semis > commas ? ";" : ",";
  1489→}

  1491→/** Parser CSV murni (tanpa dep): quoted field dengan escape "", \r\n/\n, separator koma atau titik koma. */
  1492→function parseCsv(text: string): string[][] {
  1493→  const raw = text.replace(/^\uFEFF/, "").trim();
  1494→  if (!raw) return [];
  1495→  const firstNewline = raw.search(/\r?\n/);
  1496→  const sep = detectCsvSeparator(firstNewline === -1 ? raw : raw.slice(0, firstNewline));
  1497→
  1498→  const rows: string[][] = [];
  1499→  let row: string[] = [];
  1500→  let field = "";
  1501→  let inQuotes = false;
  1502→
  1503→  for (let i = 0; i < raw.length; i++) {
  1504→    const ch = raw[i];
  1505→    if (inQuotes) {
  1506→      if (ch === '"') {
  1507→        if (raw[i + 1] === '"') {
  1508→          field += '"';
  1509→          i++;
  1510→        } else {
  1511→          inQuotes = false;
  1512→        }
  1513→      } else {
  1514→        field += ch;
  1515→      }
  1516→    } else if (ch === '"') {
  1517→      inQuotes = true;
  1518→    } else if (ch === sep) {
  1519→      row.push(field);
  1520→      field = "";
  1521→    } else if (ch === "\n" || ch === "\r") {
  1522→      if (ch === "\r" && raw[i + 1] === "\n") i++;
  1523→      row.push(field);
  1524→      field = "";
  1525→      rows.push(row);
  1526→      row = [];
  1527→    } else {
  1528→      field += ch;
  1529→    }
  1530→  }
  1531→  row.push(field);
  1532→  if (row.length > 1 || row[0] !== "") rows.push(row);
  1533→
  1534→  // Lewati baris yang seluruh selnya kosong.
  1535→  return rows.filter((r) => r.some((c) => c.trim() !== ""));
  1536→}

  1538→/** Petakan baris hasil parse ke bentuk API: via header sinonim, atau posisi kolom bila tanpa header. */
  1539→function mapCsvRows(rows: string[][]): ImportRowValues[] {
  1540→  if (rows.length === 0) return [];
  1541→  const header = isHeaderRow(rows[0]);
  1542→  const cols = header
  1543→    ? rows[0].map((c) => IMPORT_HEADER_SYNONYMS[c.trim().toLowerCase()] ?? null)
  1544→    : IMPORT_FIELD_ORDER;
  1545→  const body = header ? rows.slice(1) : rows;
  1546→  return body.map((cells) => {
  1547→    const out = emptyImportRow();
  1548→    cols.forEach((key, colIdx) => {
  1549→      if (key) out[key] = (cells[colIdx] ?? "").trim();
  1550→    });
  1551→    return out;
  1552→  });
  1553→}
