    46→/** Baca file gambar kecil → data URL (dipakai logo & kop surat). */
    47→function readImageFile(file: File, maxChars = 1_500_000): Promise<string> {
    48→  return new Promise((resolve, reject) => {
    49→    if (!file.type.startsWith("image/")) {
    50→      reject(new Error("File harus berupa gambar (PNG/JPG/SVG/WebP)"));
    51→      return;
    52→    }
    53→    if (file.size > 1_100_000) {
    54→      reject(new Error("Ukuran gambar terlalu besar (maksimal ±1MB)"));
    55→      return;
    56→    }
    57→    const reader = new FileReader();
    58→    reader.onload = () => {
    59→      const result = String(reader.result ?? "");
    60→      if (result.length > maxChars) {
    61→        reject(new Error("Gambar terlalu besar setelah dibaca — pakai file lebih kecil"));
    62→        return;
    63→      }
    64→      resolve(result);
    65→    };
    66→    reader.onerror = () => reject(new Error("Gagal membaca file"));
    67→    reader.readAsDataURL(file);
    68→  });
    69→}

    73→const IDR_FMT = new Intl.NumberFormat("id-ID", { style: "currency", currency: "IDR", maximumFractionDigits: 0 });

    75→/** Format uang rupiah tanpa desimal (mis. Rp28.000.000). */
    76→function fmtIDR(value: number): string {
    77→  return IDR_FMT.format(value);

    80→/** Tanggal-waktu ringkas id-ID (mis. 02 Sep, 13.01) — "" bila tidak valid. */
    81→function fmtWhen(iso?: string | null): string {
    82→  if (!iso) return "";
    83→  const d = new Date(iso);
    84→  if (Number.isNaN(d.getTime())) return "";
    85→  return new Intl.DateTimeFormat("id-ID", { day: "2-digit", month: "short", hour: "2-digit", minute: "2-digit" }).format(d);
    86→}

  1247→/** Parse nominal "28.000.000" / "28000000" → number (≥0); kosong/non-angka → 0. */
  1248→function parseAmount(raw: string): number {
  1249→  const digits = raw.replace(/[^\d]/g, "");
  1250→  return digits === "" ? 0 : Number(digits);
  1251→}

  1253→/** Saran harga = total biaya × (1 + margin%) dibulatkan ke 100 ribu terdekat. */
  1254→function suggestedPriceOf(total: number, marginPct: number): number {
  1255→  return Math.round((total * (1 + marginPct / 100)) / 100_000) * 100_000;
  1256→}

  1261→ */

  1272→  const [saving, setSaving] = useState(false);
  1273→
  1274→  // Sinkronkan draf saat data server berubah (setelah simpan / katalog dimuat ulang)
  1275→  useEffect(() => {
  1276→    setItems(costDraftFromService(service));
  1277→    setMargin(service.targetMarginPct != null ? String(service.targetMarginPct) : "30");
  1278→  }, [service]);
  1279→
  1280→  const hasItems = (service.costItems ?? []).length > 0;
  1281→  const serverTotal = service.costTotal ?? (service.costItems ?? []).reduce((sum, it) => sum + (Number(it.amount) || 0), 0);
  1282→  const serverSaran = service.suggestedPrice ?? suggestedPriceOf(serverTotal, service.targetMarginPct ?? 30);
  1283→
  1284→  const parsed = useMemo(
  1285→    () => items.map((it) => ({ name: it.name.trim(), amount: parseAmount(it.amount), note: it.note ?? null })),
  1286→    [items]
  1287→  );
  1288→  const invalid = parsed.some((it) => it.name === "" || !Number.isFinite(it.amount) || it.amount < 0);
  1289→  const total = parsed.reduce((sum, it) => sum + it.amount, 0);
  1290→  const marginNum = margin.trim() === "" ? 30 : Number(margin.replace(",", "."));
  1291→  const marginPct = Number.isFinite(marginNum) ? Math.min(95, Math.max(0, marginNum)) : 30;

  1299→  function resetDraft() {

  1309→      await onSave(service.id, payload, margin.trim() === "" ? null : marginPct);
  1310→      // Sukses → katalog dimuat ulang, draf direset otomatis via useEffect [service]
  1311→    } finally {
  1312→      setSaving(false);
  1313→    }
  1314→  }
  1315→
  1316→  return (
  1317→    <div className="mt-2 rounded-lg border border-zinc-200 bg-zinc-50/60">
  1318→      <div className="flex items-center gap-1.5 px-2.5 py-2">
  1319→        <button
  1320→          type="button"
  1321→          className="flex min-w-0 flex-1 items-center gap-1.5 text-left"
  1322→          aria-label={`Rincian biaya ${service.name}`}
  1323→          aria-expanded={open}
  1324→          onClick={() => setOpen((o) => !o)}
  1325→        >
  1326→          {open ? (
  1327→            <ChevronUp className="h-3.5 w-3.5 shrink-0 text-zinc-400" aria-hidden />
  1328→          ) : (
  1329→            <ChevronDown className="h-3.5 w-3.5 shrink-0 text-zinc-400" aria-hidden />
  1330→          )}
  1331→          <span className="shrink-0 text-xs font-semibold text-zinc-700">Rincian Biaya</span>
  1332→          {!open ? (
  1333→            hasItems ? (
  1334→              <span className="min-w-0 truncate text-[11px] text-zinc-500">
  1335→                {service.costItems?.length} butir · Total biaya {fmtIDR(serverTotal)} · Saran harga {fmtIDR(serverSaran)}
  1336→              </span>
  1337→            ) : (
  1338→              <span className="text-[11px] italic text-zinc-400">Belum ada rincian biaya</span>
  1339→            )
  1340→          ) : null}
  1341→        </button>
  1342→        {!open && !hasItems ? (
  1343→          <Button
  1344→            type="button" variant="outline" className="h-7 shrink-0 px-2 text-[11px]"
  1345→            onClick={() => { setOpen(true); if (items.length === 0) addItem(); }}
  1346→          >
  1347→            <Plus className="h-3 w-3" aria-hidden /> Mulai isi
  1348→          </Button>
  1349→        ) : null}
  1350→      </div>
  1351→
  1352→      {open ? (
  1353→        <div className="space-y-2.5 border-t px-3 py-3">
  1354→          {items.length > 0 ? (
  1355→            <div className="max-h-56 space-y-1.5 overflow-y-auto pr-1">
  1356→              {items.map((it, idx) => (

  1360→                      aria-label={`Nama butir biaya ${idx + 1}`}

  1362→                      placeholder="mis. Talenta / Sewa alat"

  1364→                      onChange={(e) => setItems((m) => m.map((row, i) => (i === idx ? { ...row, name: e.target.value } : row)))}
  1365→                    />
  1366→                    <Input

  1384→              ))}
  1385→            </div>
  1386→          ) : (

  1388→          )}
  1389→
  1390→          <Button type="button" variant="outline" className="h-8" onClick={addItem}>
  1391→            <Plus className="h-3.5 w-3.5" aria-hidden /> Tambah butir

  1394→          <div className="flex flex-wrap items-end justify-between gap-x-4 gap-y-2 rounded-lg border border-dashed bg-white px-3 py-2.5">

  1396→              <p className="text-[11px] text-zinc-500">Total biaya</p>

  1398→            </div>
  1399→            <div className="space-y-1">
  1400→              <Label htmlFor={`cost-margin-${service.id}`} className="text-[11px]">Margin target %</Label>
  1401→              <Input
