  1236→interface CostDraftItem {
  1237→  name: string;
  1238→  /** String agar input bebas mengetik angka (separator diabaikan saat parse). */
  1239→  amount: string;
  1240→  note?: string | null;
  1241→}
  1242→
  1243→function costDraftFromService(svc: ServiceDTO): CostDraftItem[] {
  1244→  return (svc.costItems ?? []).map((it) => ({ name: it.name, amount: String(it.amount ?? 0), note: it.note ?? null }));
  1245→}
  1246→
  1247→/** Parse nominal "28.000.000" / "28000000" → number (≥0); kosong/non-angka → 0. */
  1248→function parseAmount(raw: string): number {
  1249→  const digits = raw.replace(/[^\d]/g, "");
  1250→  return digits === "" ? 0 : Number(digits);
  1251→}
  1252→
  1253→/** Saran harga = total biaya × (1 + margin%) dibulatkan ke 100 ribu terdekat. */
  1254→function suggestedPriceOf(total: number, marginPct: number): number {
  1255→  return Math.round((total * (1 + marginPct / 100)) / 100_000) * 100_000;
  1256→}
  1257→
  1258→/**
  1259→ * Blok "Rincian Biaya" per layanan — expand/collapse dgn state LOKAL
  1260→ * (mengetik di sini tidak me-render ulang seluruh dialog).
  1261→ */
  1262→function ServiceCostEditor({
  1263→  service,
  1264→  onSave,

  1276→    setItems(costDraftFromService(service));
  1277→    setMargin(service.targetMarginPct != null ? String(service.targetMarginPct) : "30");
  1278→  }, [service]);
  1279→
  1280→  const hasItems = (service.costItems ?? []).length > 0;
  1281→  const serverTotal = service.costTotal ?? (service.costItems ?? []).reduce((sum, it) => sum + (Number(it.amount) || 0), 0);
  1282→  const serverSaran = service.suggestedPrice ?? suggestedPriceOf(serverTotal, service.targetMarginPct ?? 30);

  1311→    } finally {
  1312→      setSaving(false);
  1313→    }
