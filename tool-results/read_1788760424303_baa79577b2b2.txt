   933→/** Satu baris deliverable: ikon jenis, meta, status, aksi review (manajemen/produksi) & hapus. */
   934→function DeliverableRow({ d, user, milestoneName, onChanged }: {
   935→  d: DeliverableRowData;
   936→  user: { name: string; role: string } | null;
   937→  /** Ronde 35 — nama milestone terkait (bila deliverable ditautkan). */
   938→  milestoneName?: string | null;
   939→  onChanged: () => Promise<void>;
   940→}) {
   941→  const meta = dlvMeta(d.status);
   942→  const [review, setReview] = useState<null | "approved" | "revision">(null);
   943→  const [comment, setComment] = useState("");
   944→  const [busy, setBusy] = useState(false);
   945→
   946→  const canReview = user?.role === "super_admin" || user?.role === "director" || user?.role === "production";
   947→  const canDelete = canReview || (!!user?.name && d.createdBy === user.name);
   948→  const KindIcon = d.kind === "file" ? Paperclip : Link2;
   949→
   950→  async function submitDecision() {
   951→    if (!user || !review) return;
   952→    setBusy(true);
   953→    try {
   954→      await api.decideDeliverable({
   955→        id: d.id,
   956→        decision: review,
   957→        reviewComment: comment.trim() || undefined,
   958→        reviewedBy: user.name,
   959→        reviewedRole: user.role,
   960→      });
   961→      toast.success(review === "approved" ? `“${d.name}” disetujui` : `Revisi diminta untuk “${d.name}”`);
   962→      setReview(null);
   963→      setComment("");
   964→      await onChanged();
   965→    } catch (err) {
   966→      toast.error(err instanceof Error ? err.message : "Gagal menyimpan keputusan review");
   967→    } finally {
   968→      setBusy(false);
   969→    }
   970→  }
   971→
   972→  /** Hapus via fetch langsung: api.deleteDeliverable (kontrak api-client tidak boleh diubah)
   973→   *  tidak mengirim identitas aktor, sedangkan aturan hapus butuh actorName/actorRole. */
   974→  async function remove() {
   975→    if (!user) return;
   976→    if (!window.confirm(`Hapus deliverable “${d.name}”? Tindakan ini tidak dapat dibatalkan.`)) return;
   977→    setBusy(true);
   978→    try {
   979→      const res = await fetch("/api/projects/deliverables", {
   980→        method: "DELETE",
   981→        headers: { "Content-Type": "application/json" },
   982→        body: JSON.stringify({ id: d.id, actorName: user.name, actorRole: user.role }),
   983→      });
   984→      const json: { error?: string } = await res.json().catch(() => ({}));
   985→      if (!res.ok) throw new Error(json.error ?? "Gagal menghapus deliverable");
   986→      toast.success("Deliverable dihapus");
   987→      await onChanged();
   988→    } catch (err) {
   989→      toast.error(err instanceof Error ? err.message : "Gagal menghapus deliverable");
   990→    } finally {
   991→      setBusy(false);
   992→    }
   993→  }
   994→
   995→  return (
   996→    <div className="rounded-lg border bg-white p-3">
   997→      <div className="flex items-start justify-between gap-2">
   998→        <div className="flex min-w-0 items-start gap-2.5">
   999→          <span className="mt-0.5 flex h-7 w-7 shrink-0 items-center justify-center rounded-full bg-zinc-100" aria-hidden>
  1000→            <KindIcon className="h-3.5 w-3.5 text-zinc-500" />
  1001→          </span>
  1002→          <div className="min-w-0">
  1003→            <p className="truncate text-sm font-semibold text-zinc-900">{d.name}</p>
  1004→            {milestoneName ? (
  1005→              <span className="mt-1 inline-flex max-w-full items-center gap-1 rounded-full border border-zinc-200 bg-zinc-50 px-1.5 py-0.5 text-[10px] font-medium text-zinc-600" title={`Deliverable untuk milestone ${milestoneName}`}>
  1006→                <Route className="h-2.5 w-2.5 shrink-0 text-zinc-400" aria-hidden />
  1007→                <span className="truncate">Milestone: {milestoneName}</span>
  1008→              </span>
  1009→            ) : null}
  1010→            <p className="truncate text-[11px] text-zinc-500">
  1011→              {d.kind === "file"
  1012→                ? [d.fileName, d.sizeBytes ? formatSizeKb(d.sizeBytes) : null].filter(Boolean).join(" · ")
  1013→                : d.url}
