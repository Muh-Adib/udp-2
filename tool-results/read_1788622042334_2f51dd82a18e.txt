    19→    19→async function request<T>(url: string, init?: RequestInit): Promise<T> {
    20→    20→  const res = await fetch(url, {
    21→    21→    ...init,
    22→    22→    credentials: "same-origin", // Ronde 27: cookie sesi selalu ikut utk request same-origin
    23→    23→    headers: { "Content-Type": "application/json", ...(init?.headers ?? {}) },
    24→    24→  });
    25→    25→  const data = await res.json().catch(() => ({}));
    26→    26→  if (!res.ok) throw new Error((data as { error?: string }).error ?? `HTTP ${res.status}`);
    27→    27→  return data as T;
    28→    28→}
