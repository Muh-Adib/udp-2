    44→/** Baca file gambar kecil → data URL (dipakai logo & kop surat). */
    45→function readImageFile(file: File, maxChars = 1_500_000): Promise<string> {
    46→  return new Promise((resolve, reject) => {
    47→    if (!file.type.startsWith("image/")) {
    48→      reject(new Error("File harus berupa gambar (PNG/JPG/SVG/WebP)"));
    49→      return;
    50→    }
    51→    if (file.size > 1_100_000) {
    52→      reject(new Error("Ukuran gambar terlalu besar (maksimal ±1MB)"));
    53→      return;
    54→    }
    55→    const reader = new FileReader();
    56→    reader.onload = () => {
    57→      const result = String(reader.result ?? "");
    58→      if (result.length > maxChars) {
    59→        reject(new Error("Gambar terlalu besar setelah dibaca — pakai file lebih kecil"));
    60→        return;
    61→      }
    62→      resolve(result);
    63→    };
    64→    reader.onerror = () => reject(new Error("Gagal membaca file"));
    65→    reader.readAsDataURL(file);
    66→  });
    67→}

   865→                        </Button>
   866→                      </div>
   867→
   868→                      {svcOpen ? (
   869→                        <div className="mt-3 space-y-2.5 pl-6">
   870→                          {/* Workflow */}
   871→                          {svc.workflow.length > 0 ? (
   872→                            <ol className="space-y-1.5">
   873→                              {svc.workflow.map((w, idx) => (
   874→                                <li key={w.id} className="flex items-center gap-2 rounded-lg border bg-zinc-50/60 px-3 py-2">
   875→                                  <span className="flex h-5 w-5 shrink-0 items-center justify-center rounded-full bg-zinc-900 text-[10px] font-bold text-white" aria-hidden>{idx + 1}</span>
   876→                                  <div className="min-w-0 flex-1">
   877→                                    <p className="truncate text-xs font-medium text-zinc-800">
   878→                                      {w.name}
   879→                                      {w.isMilestone ? (
   880→                                        <Badge variant="outline" className="ml-1.5 border-amber-200 bg-amber-50 px-1 text-[10px] text-amber-700">Milestone</Badge>
   881→                                      ) : null}
   882→                                    </p>
   883→                                    <p className="truncate text-[11px] text-zinc-500">{w.phase}</p>
   884→                                  </div>
   885→                                  <Button type="button" variant="ghost" size="sm" className="h-7 px-2 text-[11px]"
   886→                                    disabled={busyKey === `stg:${svc.id}`}
   887→                                    onClick={() => void onToggleMilestone(w, svc.id)}>
   888→                                    {w.isMilestone ? "Lepas" : "Jadikan milestone"}
   889→                                  </Button>
   890→                                  <Button type="button" variant="ghost" size="icon" className="h-7 w-7 text-zinc-400 hover:text-rose-600"
   891→                                    aria-label={`Hapus langkah ${w.name}`} onClick={() => void onRemove("stage", w.id, `${w.phase} · ${w.name}`)}>
   892→                                    <Trash2 className="h-3.5 w-3.5" aria-hidden />
   893→                                  </Button>
   894→                                </li>
   895→                              ))}
   896→                            </ol>
   897→                          ) : (
   898→                            <p className="text-xs italic text-zinc-400">Belum ada workflow — tambahkan fase produksi di bawah.</p>
   899→                          )}
   900→
   901→                          {/* Form tambah langkah */}
   902→                          <div className="flex flex-wrap items-end gap-2 rounded-lg border border-dashed p-2.5">
   903→                            <div className="w-32 space-y-1">
   904→                              <Label className="text-[11px]">Fase</Label>
   905→                              <Input className="h-8 text-xs" value={form.phase} placeholder="mis. Production"
   906→                                onChange={(e) => setStageForms((m) => ({ ...m, [svc.id]: { ...form, phase: e.target.value } }))} />
   907→                            </div>
   908→                            <div className="min-w-[160px] flex-1 space-y-1">
   909→                              <Label className="text-[11px]">Langkah</Label>
   910→                              <Input className="h-8 text-xs" value={form.name} placeholder="mis. Creative Concept & Story Board"
   911→                                onChange={(e) => setStageForms((m) => ({ ...m, [svc.id]: { ...form, name: e.target.value } }))} />
   912→                            </div>
   913→                            <label className="flex items-center gap-1.5 pb-2 text-xs text-zinc-600">
   914→                              <input type="checkbox" className="accent-zinc-900" checked={form.milestone}
   915→                                onChange={(e) => setStageForms((m) => ({ ...m, [svc.id]: { ...form, milestone: e.target.checked } }))} />
   916→                              Milestone
   917→                            </label>
   918→                            <Button type="button" size="sm" variant="outline" className="h-8"
   919→                              disabled={busyKey === `stg:${svc.id}` || !form.name.trim()}
   920→                              onClick={() => void onAddStage(svc.id, form.phase, form.name, form.milestone)
   921→                                .then(() => setStageForms((m) => ({ ...m, [svc.id]: { phase: "", name: "", milestone: false } })))}>
   922→                              <Plus className="h-3.5 w-3.5" aria-hidden /> Langkah
   923→                            </Button>
   924→                          </div>
   925→                        </div>
   926→                      ) : null}
   927→                    </div>
   928→                  );
   929→                })}
   930→
   931→                {/* Form tambah layanan */}
   932→                <ServiceAddForm
   933→                  busy={busyKey === `svc:${cat.id}`}
   934→                  form={catForms[cat.id] ?? { name: "", unit: "", price: "" }}
   935→                  onChange={(form) => setCatForms((m) => ({ ...m, [cat.id]: form }))}
   936→                  onSubmit={(f) =>
   937→                    onAddService(cat.id, f.name, f.unit, f.price)
   938→                      .then((res) => {
   939→                        if (res) setCatForms((m) => ({ ...m, [cat.id]: { name: "", unit: "", price: "" } }));
   940→                        return res;
   941→                      })
   942→                  }
   943→                />
   944→              </div>
   945→            ) : null}
   946→          </div>
   947→        );
   948→      })}
   949→
   950→      {/* Tanpa kategori */}
   951→      {uncategorized.length > 0 ? (
   952→        <div className="rounded-xl border border-dashed p-3">
   953→          <p className="mb-2 text-xs font-semibold uppercase tracking-wider text-zinc-500">Tanpa kategori</p>
   954→          <div className="flex flex-wrap gap-1.5">
   955→            {uncategorized.map((s) => (
   956→              <span key={s.id} className="group inline-flex items-center gap-1 rounded-full border border-zinc-200 bg-zinc-50 px-2.5 py-1 text-xs text-zinc-600">
   957→                {s.name}
   958→                <button type="button" className="text-zinc-300 hover:text-rose-600" aria-label={`Hapus ${s.name}`} onClick={() => void onRemove("service", s.id, s.name)}>
   959→                  <X className="h-3 w-3" aria-hidden />
   960→                </button>
   961→              </span>
   962→            ))}
   963→          </div>
   964→        </div>
   965→      ) : null}
   966→
   967→      {/* Form kategori baru */}
   968→      <div className="flex flex-wrap items-end gap-2 rounded-xl border border-dashed p-3.5">
   969→        <div className="min-w-[180px] flex-1 space-y-1">
   970→          <Label className="text-xs">Kategori layanan baru</Label>
   971→          <Input className="h-9" value={newCatName} placeholder="mis. Animasi 3D"
   972→            onChange={(e) => setNewCatName(e.target.value)} />
   973→        </div>
   974→        <Button type="button" size="sm" className="h-9" disabled={busyKey === "cat:new" || !newCatName.trim()}
   975→          onClick={() => void onAddCategory(newCatName).then((res) => { if (res) setNewCatName(""); return res; })}>
   976→          <Plus className="h-3.5 w-3.5" aria-hidden /> Tambah kategori
   977→        </Button>
   978→      </div>
   979→    </div>
   980→  );
   981→}
