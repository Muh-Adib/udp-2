    88→function isoToDateInput(iso?: string | null): string {
    89→  if (!iso) return "";
    90→  const d = new Date(iso);
    91→  if (Number.isNaN(d.getTime())) return "";
    92→  return d.toISOString().slice(0, 10);
    93→}
    94→
    95→function parseBudget(v: string): number | null {
    96→  const digits = v.replace(/[^\d]/g, "");
    97→  if (!digits) return null;
    98→  const n = Number(digits);
    99→  return Number.isFinite(n) && n > 0 ? n : null;
   100→}
   101→
   102→/** Label durasi hari antara dua tanggal (inklusif kasar). */
   103→function durationLabel(start?: string | null, end?: string | null): string | null {
   104→  if (!start && !end) return null;
   105→  const s = start ? new Date(start).getTime() : null;
   106→  const e = end ? new Date(end).getTime() : null;
   107→  if (s && e && e >= s) {
   108→    const days = Math.round((e - s) / 86_400_000);
   109→    return `± ${days} hari kerja kalender`;
   110→  }
   111→  return null;
   112→}

   143→/** Stepper status: Draft → Review → Disetujui (revision ditandai alert terpisah). */
   144→function StatusStepper({ status }: { status: BriefStatus }) {
   145→  const steps = [
   146→    { key: "draft", label: "Draft", icon: ClipboardList },
   147→    { key: "in_review", label: "Review", icon: Send },
   148→    { key: "approved", label: "Disetujui", icon: BadgeCheck },
   149→  ];
   150→  const reachedIdx = status === "approved" ? 2 : status === "in_review" ? 1 : 0;
   151→  return (
   152→    <ol className="flex items-center gap-1.5" aria-label="Status brief">
   153→      {steps.map((s, i) => {
   154→        const active = i <= reachedIdx && (status !== "revision" || i === 0);
   155→        const current = status === "in_review" ? i === 1 : status === "approved" ? i === 2 : i === 0;
   156→        return (
   157→          <li key={s.key} className="flex items-center gap-1.5">
   158→            {i > 0 ? <span className={cn("h-px w-4 sm:w-6", i <= reachedIdx ? "bg-emerald-300" : "bg-zinc-200")} aria-hidden="true" /> : null}
   159→            <span
   160→              className={cn(
   161→                "flex items-center gap-1 rounded-full border px-2 py-0.5 text-[11px] font-medium",
   162→                status === "revision" && i === 0
   163→                  ? "border-red-200 bg-red-50 text-red-600"
   164→                  : active
   165→                    ? current
   166→                      ? "border-emerald-200 bg-emerald-50 text-emerald-700"
   167→                      : "border-emerald-100 bg-emerald-50/60 text-emerald-600"
   168→                    : "border-zinc-200 bg-white text-zinc-400"
   169→              )}
   170→            >
   171→              <s.icon className="size-3" aria-hidden="true" />
   172→              {s.label}
   173→            </span>
   174→          </li>
   175→        );
   176→      })}
   177→      {status === "revision" ? (
   178→        <li className="ml-1 flex items-center gap-1 rounded-full border border-red-200 bg-red-50 px-2 py-0.5 text-[11px] font-medium text-red-600">
   179→          <CircleAlert className="size-3" aria-hidden="true" />
   180→          Revisi
   181→        </li>
   182→      ) : null}
   183→    </ol>
   184→  );
   185→}

   188→function FlowDiagram({ brandColor }: { brandColor: string }) {
   189→  const items = [
   190→    { label: "Brief", icon: ClipboardList, active: true },
   191→    { label: "Estimasi", icon: Wallet, active: false },

   194→  return (
   195→    <div className="flex items-center justify-center gap-2" aria-hidden="true">
   196→      {items.map((it, i) => (
   197→        <div key={it.label} className="flex items-center gap-2">
   198→          {i > 0 ? <ArrowRight className="size-3.5 text-zinc-300" /> : null}
   199→          <span
   200→            className={cn(
   201→              "flex items-center gap-1.5 rounded-full border px-3 py-1 text-xs font-medium",
   202→              it.active ? "border-transparent text-white shadow-sm" : "border-zinc-200 bg-white text-zinc-400"
   203→            )}
   204→            style={it.active ? { backgroundColor: brandColor } : undefined}
   205→          >
   206→            <it.icon className="size-3.5" />
   207→            {it.label}
   208→          </span>
   209→        </div>
   210→      ))}
   211→    </div>
   212→  );
   213→}
