   172→function parseAiSummary(text: string): { label: string; body: string }[] {
   173→  const sections: { label: string; body: string }[] = [];
   174→  let current: { label: string; body: string } | null = null;
   175→  for (const raw of text.split(/\r?\n/)) {
   176→    const line = raw.replace(/\*\*/g, "").trim();
   177→    if (!line) {
   178→      if (current) current.body += "\n";
   179→      continue;
   180→    }
   181→    const key = Object.keys(AI_LABELS).find((k) => line.toUpperCase().startsWith(k));
   182→    if (key) {
   183→      if (current) sections.push(current);
   184→      const rest = line.slice(key.length).replace(/^[:\-\u2013\s]+/, "");
   185→      current = { label: AI_LABELS[key], body: rest };
   186→    } else if (current) {
   187→      current.body += (current.body ? "\n" : "") + line;
   188→    }
   189→  }
   190→  if (current) sections.push(current);
   191→  return sections;
   192→}

   480→type EstimationCalc = {
   481→  totalCost: number;
   482→  contingency: number;
   483→  managementFee: number;
   484→  costWithFees: number;
   485→  discountAmount: number;
   486→  netRevenue: number;
   487→  taxAmount: number;
   488→  grandTotal: number;
   489→  margin: number;
   490→  marginPct: number;
   491→};
   492→
   493→/** Rumus identik dengan server (src/app/api/opportunities/[id]/estimation). */
   494→function computeEstimation(form: EstimationForm): EstimationCalc {
   495→  const totalCost = COST_CATEGORIES.reduce((s, c) => s + toNum(form[c.key]), 0);
   496→  const contingency = Math.round((totalCost * toNum(form.contingencyPct)) / 100);
   497→  const managementFee = Math.round((totalCost * toNum(form.managementFeePct)) / 100);
   498→  const costWithFees = totalCost + contingency + managementFee;
   499→  const revenue = toNum(form.revenue);
   500→  const discountAmount = Math.round((revenue * toNum(form.discountPct)) / 100);
   501→  const netRevenue = revenue - discountAmount;
   502→  const taxAmount = Math.round((netRevenue * toNum(form.taxPct)) / 100);
   503→  const grandTotal = netRevenue + taxAmount;
   504→  const margin = netRevenue - costWithFees;
   505→  const marginPct = netRevenue > 0 ? Math.round((margin / netRevenue) * 1000) / 10 : 0;
   506→  return {
   507→    totalCost, contingency, managementFee, costWithFees,
   508→    discountAmount, netRevenue, taxAmount, grandTotal, margin, marginPct,
   509→  };
   510→}

   512→function marginTone(marginPct: number, target: number): { label: string; text: string; badge: string; bar: string } {
   513→  if (marginPct >= target) {
   514→    return { label: "Di atas target", text: "text-emerald-700", badge: "bg-emerald-100 text-emerald-700 hover:bg-emerald-100", bar: "[&>div]:bg-emerald-500" };
   515→  }
   516→  if (marginPct >= target - 10) {
   517→    return { label: "Mendekati target", text: "text-amber-600", badge: "bg-amber-100 text-amber-700 hover:bg-amber-100", bar: "[&>div]:bg-amber-500" };
   518→  }
   519→  return { label: "Di bawah target", text: "text-rose-600", badge: "bg-rose-100 text-rose-700 hover:bg-rose-100", bar: "[&>div]:bg-rose-500" };
   520→}
