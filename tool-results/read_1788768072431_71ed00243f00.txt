    46→/** Baca file gambar kecil → data URL (dipakai logo & kop surat). */
    47→function readImageFile(file: File, maxChars = 1_500_000): Promise<string> {
    48→  return new Promise((resolve, reject) => {
    49→    if (!file.type.startsWith("image/")) {
    50→      reject(new Error("File harus berupa gambar (PNG/JPG/SVG/WebP)"));
    51→      return;
    52→    }
    53→    if (file.size > 1_100_000) {
    54→      reject(new Error("Ukuran gambar terlalu besar (maksimal ±1MB)"));
    55→      return;
    56→    }
    57→    const reader = new FileReader();
    58→    reader.onload = () => {
    59→      const result = String(reader.result ?? "");
    60→      if (result.length > maxChars) {
    61→        reject(new Error("Gambar terlalu besar setelah dibaca — pakai file lebih kecil"));
    62→        return;
    63→      }
    64→      resolve(result);
    65→    };
    66→    reader.onerror = () => reject(new Error("Gagal membaca file"));
    67→    reader.readAsDataURL(file);
    68→  });
    69→}

    73→const IDR_FMT = new Intl.NumberFormat("id-ID", { style: "currency", currency: "IDR", maximumFractionDigits: 0 });

    75→/** Format uang rupiah tanpa desimal (mis. Rp28.000.000). */
    76→function fmtIDR(value: number): string {
    77→  return IDR_FMT.format(value);

    80→/** Tanggal-waktu ringkas id-ID (mis. 02 Sep, 13.01) — "" bila tidak valid. */
    81→function fmtWhen(iso?: string | null): string {
    82→  if (!iso) return "";
    83→  const d = new Date(iso);
    84→  if (Number.isNaN(d.getTime())) return "";
    85→  return new Intl.DateTimeFormat("id-ID", { day: "2-digit", month: "short", hour: "2-digit", minute: "2-digit" }).format(d);
    86→}

  1117→// ============ Rincian biaya per layanan (template harga marketing) — Ronde 29-b ============
  1118→
  1119→interface CostDraftItem {
  1120→  name: string;
  1121→  /** String agar input bebas mengetik angka (separator diabaikan saat parse). */
  1122→  amount: string;
  1123→  note?: string | null;
  1124→}
  1125→
  1126→function costDraftFromService(svc: ServiceDTO): CostDraftItem[] {
  1127→  return (svc.costItems ?? []).map((it) => ({ name: it.name, amount: String(it.amount ?? 0), note: it.note ?? null }));
  1128→}
  1129→
  1130→/** Parse nominal "28.000.000" / "28000000" → number (≥0); kosong/non-angka → 0. */
  1131→function parseAmount(raw: string): number {
  1132→  const digits = raw.replace(/[^\d]/g, "");
  1133→  return digits === "" ? 0 : Number(digits);
  1134→}
  1135→
  1136→/** Saran harga = total biaya × (1 + margin%) dibulatkan ke 100 ribu terdekat. */
  1137→function suggestedPriceOf(total: number, marginPct: number): number {
  1138→  return Math.round((total * (1 + marginPct / 100)) / 100_000) * 100_000;
  1139→}
  1140→
  1141→/**
  1142→ * Blok "Rincian Biaya" per layanan — expand/collapse dgn state LOKAL
  1143→ * (mengetik di sini tidak me-render ulang seluruh dialog).
  1144→ */
  1145→function ServiceCostEditor({
  1146→  service,
  1147→  onSave,
  1148→}: {
  1149→  service: ServiceDTO;
  1150→  onSave: (serviceId: string, items: ServiceCostItem[], marginPct: number | null) => Promise<boolean>;
  1151→}) {
  1152→  const [open, setOpen] = useState(false);
  1153→  const [items, setItems] = useState<CostDraftItem[]>(() => costDraftFromService(service));
  1154→  const [margin, setMargin] = useState<string>(() => (service.targetMarginPct != null ? String(service.targetMarginPct) : "30"));
  1155→  const [saving, setSaving] = useState(false);
  1156→
  1157→  // Sinkronkan draf saat data server berubah (setelah simpan / katalog dimuat ulang)
  1158→  useEffect(() => {
  1159→    setItems(costDraftFromService(service));
  1160→    setMargin(service.targetMarginPct != null ? String(service.targetMarginPct) : "30");
  1161→  }, [service]);
  1162→
  1163→  const hasItems = (service.costItems ?? []).length > 0;
  1164→  const serverTotal = service.costTotal ?? (service.costItems ?? []).reduce((sum, it) => sum + (Number(it.amount) || 0), 0);
  1165→  const serverSaran = service.suggestedPrice ?? suggestedPriceOf(serverTotal, service.targetMarginPct ?? 30);
  1166→
  1167→  const parsed = useMemo(
  1168→    () => items.map((it) => ({ name: it.name.trim(), amount: parseAmount(it.amount), note: it.note ?? null })),
  1169→    [items]
  1170→  );
  1171→  const invalid = parsed.some((it) => it.name === "" || !Number.isFinite(it.amount) || it.amount < 0);
  1172→  const total = parsed.reduce((sum, it) => sum + it.amount, 0);
  1173→  const marginNum = margin.trim() === "" ? 30 : Number(margin.replace(",", "."));
  1174→  const marginPct = Number.isFinite(marginNum) ? Math.min(95, Math.max(0, marginNum)) : 30;
  1175→  const saran = suggestedPriceOf(total, marginPct);
  1176→  const basePrice = service.basePrice ?? null;
  1177→
  1178→  function addItem() {
  1179→    setItems((m) => [...m, { name: "", amount: "", note: null }]);
  1180→  }
  1181→
  1182→  function resetDraft() {
  1183→    setItems(costDraftFromService(service));
  1184→    setMargin(service.targetMarginPct != null ? String(service.targetMarginPct) : "30");
  1185→  }
  1186→
  1187→  async function save() {
  1188→    if (invalid || saving) return;
  1189→    setSaving(true);
  1190→    try {
  1191→      const payload: ServiceCostItem[] = parsed.map((it) => ({ name: it.name, amount: it.amount, note: it.note }));
  1192→      await onSave(service.id, payload, margin.trim() === "" ? null : marginPct);
  1193→      // Sukses → katalog dimuat ulang, draf direset otomatis via useEffect [service]
  1194→    } finally {
  1195→      setSaving(false);
  1196→    }
  1197→  }
  1198→
  1199→  return (
  1200→    <div className="mt-2 rounded-lg border border-zinc-200 bg-zinc-50/60">
  1201→      <div className="flex items-center gap-1.5 px-2.5 py-2">
  1202→        <button
  1203→          type="button"
  1204→          className="flex min-w-0 flex-1 items-center gap-1.5 text-left"
  1205→          aria-label={`Rincian biaya ${service.name}`}
  1206→          aria-expanded={open}
  1207→          onClick={() => setOpen((o) => !o)}
  1208→        >
  1209→          {open ? (
  1210→            <ChevronUp className="h-3.5 w-3.5 shrink-0 text-zinc-400" aria-hidden />
  1211→          ) : (
  1212→            <ChevronDown className="h-3.5 w-3.5 shrink-0 text-zinc-400" aria-hidden />

  1222→            )
  1223→          ) : null}
  1224→        </button>
  1225→        {!open && !hasItems ? (
  1226→          <Button
  1227→            type="button" variant="outline" className="h-7 shrink-0 px-2 text-[11px]"
  1228→            onClick={() => { setOpen(true); if (items.length === 0) addItem(); }}
  1229→          >

  1241→                  <div className="flex items-center gap-1.5">

  1245→                      placeholder="mis. Talenta / Sewa alat"

  1247→                      onChange={(e) => setItems((m) => m.map((row, i) => (i === idx ? { ...row, name: e.target.value } : row)))}

  1273→          <Button type="button" variant="outline" className="h-8" onClick={addItem}>

  1277→          <div className="flex flex-wrap items-end justify-between gap-x-4 gap-y-2 rounded-lg border border-dashed bg-white px-3 py-2.5">
