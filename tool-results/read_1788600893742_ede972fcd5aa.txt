    40→function taskTypeMeta(type: string) {
    41→  return TASK_TYPES.find((t) => t.key === type) ?? { key: type, label: type, icon: ClipboardList, cls: "bg-zinc-100 text-zinc-600" };

    44→function priorityMeta(p: string): { label: string; cls: string } {
    45→  switch (p) {
    46→    case "urgent": return { label: "Urgent", cls: "bg-rose-100 text-rose-700" };
    47→    case "high": return { label: "Tinggi", cls: "bg-amber-100 text-amber-700" };
    48→    case "medium": return { label: "Sedang", cls: "bg-zinc-100 text-zinc-600" };
    49→    case "low": return { label: "Rendah", cls: "bg-zinc-100 text-zinc-400" };
    50→    default: return { label: p, cls: "bg-zinc-100 text-zinc-600" };
    51→  }
    52→}

    54→// ============ Helper tanggal ============
    55→
    56→function startOfToday(): Date {
    57→  const d = new Date();
    58→  d.setHours(0, 0, 0, 0);
    59→  return d;
    60→}
    61→
    62→function endOfToday(): Date {
    63→  const d = new Date();
    64→  d.setHours(23, 59, 59, 999);
    65→  return d;
    66→}
    67→
    68→function isOverdue(t: TaskDTO): boolean {
    69→  return t.status === "open" && !!t.dueDate && new Date(t.dueDate).getTime() < startOfToday().getTime();
    70→}
    71→
    72→function isToday(t: TaskDTO): boolean {
    73→  if (!t.dueDate) return false;
    74→  const due = new Date(t.dueDate).getTime();
    75→  return due >= startOfToday().getTime() && due <= endOfToday().getTime();
    76→}

   245→/** Konteks opportunity yang dibutuhkan dialog kirim (subset detail API). */
   246→interface SendContext {

   256→/** Ganti placeholder template dengan data opportunity yang diketahui. */
   257→function renderTemplate(body: string, ctx: SendContext, senderName: string): string {
   258→  const map: Record<string, string> = {
   259→    contact_name: ctx.contact?.fullName ?? "Bapak/Ibu",
   260→    company_name: ctx.company?.name ?? "perusahaan Bapak/Ibu",
   261→    brand_name: ctx.brand?.name ?? "tim kami",
   262→    service_name: ctx.serviceName ?? ctx.title,
   263→    marketing_name: senderName,
   264→  };
   265→  return body.replace(/{{\s*(\w+)\s*}}/g, (full, key: string) => map[key] ?? full);
   266→}

   268→function channelMeta(key: string) {
   269→  return OUTBOUND_CHANNELS.find((c) => c.key === key) ?? OUTBOUND_CHANNELS[0];
   270→}

   272→function SendOutboundDialog({ task, onClose, onSent }: {
   273→  task: TaskDTO | null;
   274→  onClose: () => void;
   275→  onSent: (task: TaskDTO, updated?: TaskDTO) => void;
   276→}) {
   277→  const user = useCrmStore((s) => s.user);
   278→  const [ctx, setCtx] = useState<SendContext | null>(null);
   279→  const [templates, setTemplates] = useState<FollowUpTemplateDTO[]>([]);
   280→  const [loadingCtx, setLoadingCtx] = useState(false);
   281→  const [channel, setChannel] = useState("whatsapp");
   282→  const [templateId, setTemplateId] = useState<string>("blank");
   283→  const [body, setBody] = useState("");
   284→  const [markDone, setMarkDone] = useState(true);
   285→  const [sending, setSending] = useState(false);
   286→
   287→  // Muat konteks opportunity + template saat dialog dibuka
   288→  useEffect(() => {
   289→    if (!task?.opportunityId) return;
   290→    let alive = true;
   291→    setLoadingCtx(true);
   292→    setCtx(null);
   293→    setTemplates([]);
   294→    setTemplateId("blank");
   295→    setBody("");
   296→    setChannel("whatsapp");
   297→    setMarkDone(task.status !== "done");
   298→    (async () => {
   299→      try {
   300→        const [detail, tplRes] = await Promise.all([
   301→          api.opportunity(task.opportunityId as string),
   302→          api.followUpTemplates("all"),
   303→        ]);
   304→        if (!alive) return;
   305→        const o = detail.opportunity;
   306→        const nextCtx: SendContext = {
   307→          id: o.id,
   308→          title: o.title,
   309→          serviceName: o.serviceName,
   310→          brand: o.brand ? { id: o.brand.id, name: o.brand.name, color: o.brand.color } : null,
   311→          contact: o.contact
   312→            ? { id: o.contact.id, fullName: o.contact.fullName, email: o.contact.email, whatsapp: o.contact.whatsapp }
   313→            : null,
   314→          company: o.company ? { id: o.company.id, name: o.company.name } : null,
   315→          ownerName: o.ownerName,
   316→        };
   317→        setCtx(nextCtx);
   318→        // Template global + milik brand opportunity, urut delay
   319→        const relevant = tplRes.templates
   320→          .filter((t) => !t.brandId || t.brandId === o.brandId)
   321→          .sort((a, b) => a.delayDays - b.delayDays);
   322→        setTemplates(relevant);
   323→      } catch {
   324→        if (alive) toast.error("Gagal memuat konteks opportunity");
   325→      } finally {
   326→        if (alive) setLoadingCtx(false);
   327→      }
   328→    })();
   329→    return () => { alive = false; };
   330→  }, [task]);
   331→
   332→  const senderName = user?.name ?? ctx?.ownerName ?? "Tim Sales";
   333→
   334→  function pickTemplate(id: string) {
   335→    setTemplateId(id);
   336→    if (!ctx) return;
   337→    if (id === "blank") {
   338→      setBody("");
   339→      return;
   340→    }
   341→    const tpl = templates.find((t) => t.id === id);
   342→    if (!tpl) return;
   343→    setChannel(tpl.channel);
   344→    setBody(renderTemplate(tpl.body, ctx, senderName));
   345→  }
   346→
   347→  const unknownPlaceholders = useMemo(() => {
   348→    if (!ctx) return [] as string[];
   349→    const found = body.match(/{{\s*(\w+)\s*}}/g) ?? [];
   350→    return Array.from(new Set(found));
   351→  }, [body, ctx]);
   352→
   353→  async function handleSend() {
   354→    if (!task || !ctx) return;
   355→    const content = body.trim();
   356→    if (!content) {
   357→      toast.error("Isi pesan tidak boleh kosong");
   358→      return;
   359→    }
   360→    const ch = channelMeta(channel);
   361→    if (ch.key === "email" && !ctx.contact?.email) {
   362→      toast.error("Kontak tidak memiliki alamat email");
   363→      return;
   364→    }
   365→    if (ch.key === "whatsapp" && !ctx.contact?.whatsapp) {
   366→      toast.error("Kontak tidak memiliki nomor WhatsApp");
   367→      return;
   368→    }
   369→    setSending(true);
   370→    try {
   371→      await api.createInteraction({
   372→        channel: ch.key,
   373→        direction: "outbound",
   374→        brandId: ctx.brand?.id,
   375→        senderName,
   376→        recipientName: ctx.contact?.fullName,
   377→        subject: ch.key === "email" ? `Follow-up: ${ctx.title}` : undefined,
   378→        content,
   379→        opportunityId: ctx.id,
   380→        contactId: ctx.contact?.id,
   381→        companyId: ctx.company?.id,
   382→        externalId: `task:${task.id}`,
   383→        actorName: user?.name ?? "System",
   384→        actorRole: user?.role ?? "system",
   385→      });
   386→      let updated: TaskDTO | undefined;
   387→      if (markDone && task.status !== "done") {
   388→        const res = await api.updateTask(task.id, { status: "done" });
   389→        updated = res.task;
   390→      }
   391→      toast.success(`Pesan ${ch.label} tercatat terkirim`, {
   392→        description: `Tercatat sebagai Interaction outbound pada timeline ${ctx.title}.`,
   393→      });
   394→      onSent(task, updated);
   395→      onClose();
   396→    } catch (err) {
   397→      toast.error(err instanceof Error ? err.message : "Gagal mencatat pesan");
   398→    } finally {
   399→      setSending(false);
   400→    }
   401→  }
   402→
   403→  const ChIcon = channelMeta(channel).icon;
   404→
   405→  return (
   406→    <Dialog open={task !== null} onOpenChange={(open) => { if (!open && !sending) onClose(); }}>
   407→      <DialogContent className="max-h-[92vh] overflow-y-auto crm-scroll sm:max-w-xl" aria-label="Kirim pesan follow-up">
   408→        <DialogHeader>
   409→          <DialogTitle className="flex items-center gap-2">
   410→            <span className="flex h-7 w-7 items-center justify-center rounded-lg bg-zinc-900 text-white" aria-hidden>
   411→              <Send className="h-3.5 w-3.5" />
   412→            </span>
   413→            Kirim &amp; Catat Follow-up
   414→          </DialogTitle>
   415→          <DialogDescription>
   416→            {task?.opportunity ? (
   417→              <>Pesan outbound untuk <span className="font-medium text-zinc-700">{task.opportunity.title}</span> — tercatat otomatis di timeline interaksi.</>
   418→            ) : (
   419→              <>Task ini belum tertaut ke opportunity — kirim pesan tersedia hanya untuk task dengan opportunity.</>
   420→            )}
   421→          </DialogDescription>
   422→        </DialogHeader>
   423→
   424→        {loadingCtx || !ctx ? (
   425→          <div className="space-y-3 py-2" aria-hidden>
   426→            <Skeleton className="h-16 w-full rounded-lg" />
   427→            <Skeleton className="h-20 w-full rounded-lg" />
   428→            <Skeleton className="h-28 w-full rounded-lg" />
   429→          </div>
   430→        ) : (
   431→          <div className="space-y-4 py-1">
   432→            {/* Penerima */}
   433→            <div className="rounded-lg border bg-zinc-50 p-3">
   434→              <div className="flex flex-wrap items-center gap-x-4 gap-y-1.5 text-xs">
   435→                <span className="flex items-center gap-1.5 font-medium text-zinc-900">
   436→                  <span className="flex h-6 w-6 items-center justify-center rounded-full bg-zinc-900 text-[9px] font-bold text-white" aria-hidden>
   437→                    {initials(ctx.contact?.fullName ?? "?")}
   438→                  </span>
   439→                  {ctx.contact?.fullName ?? "Kontak tidak dikenal"}
   440→                </span>
   441→                {ctx.contact?.whatsapp ? (
   442→                  <span className="inline-flex items-center gap-1 text-zinc-600"><MessageCircle className="h-3.5 w-3.5 text-emerald-600" aria-hidden /> {ctx.contact.whatsapp}</span>
   443→                ) : null}
   444→                {ctx.contact?.email ? (
   445→                  <span className="inline-flex items-center gap-1 text-zinc-600"><Mail className="h-3.5 w-3.5 text-zinc-400" aria-hidden /> {ctx.contact.email}</span>
   446→                ) : null}
   447→                {ctx.brand ? (
   448→                  <span className="inline-flex items-center gap-1 text-zinc-600">
   449→                    <span className="h-2 w-2 rounded-full" style={{ backgroundColor: ctx.brand.color }} aria-hidden />
   450→                    {ctx.brand.name}
   451→                  </span>
   452→                ) : null}
   453→              </div>
   454→            </div>
   455→
   456→            {/* Pilih template */}
   457→            <div className="space-y-2">
   458→              <Label className="flex items-center gap-1.5">
   459→                <FileText className="h-3.5 w-3.5 text-zinc-400" aria-hidden /> Template follow-up
   460→              </Label>
   461→              {templates.length === 0 ? (
   462→                <p className="rounded-lg border border-dashed border-zinc-200 p-2.5 text-xs text-zinc-400">
   463→                  Belum ada template untuk brand ini — tulis pesan manual di bawah.
   464→                </p>
   465→              ) : (
   466→                <div className="crm-scroll max-h-40 space-y-1.5 overflow-y-auto pr-1">
   467→                  <label
   468→                    className={`flex cursor-pointer items-center justify-between gap-2 rounded-lg border p-2.5 text-xs transition-colors ${templateId === "blank" ? "border-zinc-900 bg-zinc-50 ring-1 ring-zinc-900" : "border-zinc-200 hover:border-zinc-300"}`}
   469→                  >
   470→                    <span className="flex items-center gap-2 font-medium text-zinc-900">
   471→                      <Sparkles className="h-3.5 w-3.5 text-zinc-400" aria-hidden /> Tulis pesan sendiri
   472→                    </span>
   473→                    <input type="radio" className="sr-only" checked={templateId === "blank"} onChange={() => pickTemplate("blank")} aria-label="Tulis pesan sendiri" />
   474→                  </label>
   475→                  {templates.map((t) => {
   476→                    const TplIcon = channelMeta(t.channel).icon;
   477→                    return (
   478→                      <label
   479→                        key={t.id}
   480→                        className={`flex cursor-pointer items-center justify-between gap-2 rounded-lg border p-2.5 text-xs transition-colors ${templateId === t.id ? "border-zinc-900 bg-zinc-50 ring-1 ring-zinc-900" : "border-zinc-200 hover:border-zinc-300"}`}
   481→                      >
   482→                        <span className="min-w-0">
   483→                          <span className="flex items-center gap-1.5 font-medium text-zinc-900">
   484→                            {t.name}
   485→                            {!t.approved ? <Badge variant="outline" className="border-amber-200 bg-amber-50 px-1 text-[9px] text-amber-700">review</Badge> : null}
   486→                          </span>
   487→                          <span className="mt-0.5 flex items-center gap-2 text-[10px] text-zinc-500">
   488→                            <span className="inline-flex items-center gap-1">
   489→                              <TplIcon className="h-3 w-3" aria-hidden />
   490→                              {channelMeta(t.channel).label}
   491→                            </span>
   492→                            <span aria-hidden>·</span>
   493→                            <span>H+{t.delayDays}</span>
   494→                          </span>
   495→                        </span>
   496→                        <input type="radio" className="sr-only" checked={templateId === t.id} onChange={() => pickTemplate(t.id)} aria-label={`Pakai template ${t.name}`} />
   497→                      </label>
   498→                    );
   499→                  })}
   500→                </div>
   501→              )}
   502→            </div>
   503→
   504→            {/* Kanal + isi */}
   505→            <div className="grid gap-3 sm:grid-cols-[170px_1fr]">
   506→              <div className="space-y-2">
   507→                <Label htmlFor="outbound-channel">Kanal</Label>
   508→                <Select value={channel} onValueChange={setChannel}>
   509→                  <SelectTrigger id="outbound-channel" className="bg-white">
   510→                    <SelectValue />
   511→                  </SelectTrigger>
   512→                  <SelectContent>
   513→                    {OUTBOUND_CHANNELS.map((c) => (
   514→                      <SelectItem key={c.key} value={c.key} disabled={c.key === "email" && !ctx.contact?.email}>
   515→                        <span className="flex items-center gap-2"><c.icon className="h-3.5 w-3.5" aria-hidden /> {c.label}</span>
   516→                      </SelectItem>
   517→                    ))}
   518→                  </SelectContent>
   519→                </Select>
   520→              </div>
   521→              <div className="space-y-2">
   522→                <Label htmlFor="outbound-body">Isi pesan</Label>
   523→                <Textarea
   524→                  id="outbound-body"
   525→                  value={body}
   526→                  onChange={(e) => setBody(e.target.value)}
   527→                  rows={7}
   528→                  placeholder="Tulis pesan follow-up…"
   529→                  className="bg-white text-sm"
   530→                />
   531→                {unknownPlaceholders.length > 0 ? (
   532→                  <p className="text-[11px] text-amber-600">
   533→                    Placeholder belum terisi: {unknownPlaceholders.join(", ")} — lengkapi manual sebelum kirim.
   534→                  </p>
   535→                ) : null}
   536→              </div>
   537→            </div>
   538→
   539→            <label className="flex cursor-pointer items-center gap-2 rounded-lg border bg-zinc-50 p-3 text-sm text-zinc-700">
   540→              <Checkbox checked={markDone} onCheckedChange={(v) => setMarkDone(v === true)} aria-label="Tandai task selesai setelah kirim" />
   541→              Tandai task selesai setelah pesan tercatat
   542→            </label>
   543→          </div>
   544→        )}
   545→
   546→        <DialogFooter>
   547→          <Button variant="outline" onClick={onClose} disabled={sending}>Batal</Button>
   548→          <Button
   549→            onClick={() => void handleSend()}
   550→            disabled={sending || loadingCtx || !ctx || !body.trim()}
   551→            aria-label="Kirim dan catat pesan follow-up"
   552→          >
   553→            {sending ? <RefreshCw className="h-4 w-4 animate-spin" aria-hidden /> : <ChIcon className="h-4 w-4" aria-hidden />}
   554→            {sending ? "Mencatat…" : `Kirim via ${channelMeta(channel).label}`}
   555→          </Button>
   556→        </DialogFooter>
   557→      </DialogContent>
   558→    </Dialog>
   559→  );
   560→}

   562→// ============ Module utama ============
   563→
   564→export default function FollowupsModule() {
   565→  const user = useCrmStore((s) => s.user);
   566→
   567→  const [tasks, setTasks] = useState<TaskDTO[] | null>(null);
   568→  const [loading, setLoading] = useState(true);
   569→  const [error, setError] = useState<string | null>(null);
   570→  const [busyId, setBusyId] = useState<string | null>(null);
   571→
   572→  const [statusFilter, setStatusFilter] = useState("all");
   573→  const [assigneeFilter, setAssigneeFilter] = useState("all");
   574→  const [onlyOverdue, setOnlyOverdue] = useState(false);
   575→
   576→  const [createOpen, setCreateOpen] = useState(false);
   577→  const [saving, setSaving] = useState(false);
   578→  const [form, setForm] = useState({
   579→    title: "", type: "follow_up", priority: "medium", assigneeName: "", dueDate: "", opportunityId: "",
   580→  });
   581→  // Fase 3 — kirim & catat pesan outbound dari task follow-up
   582→  const [sendTarget, setSendTarget] = useState<TaskDTO | null>(null);
   583→
   584→  const load = useCallback(async (silent = false) => {
   585→    if (!silent) setLoading(true);
   586→    setError(null);
   587→    try {
   588→      const res = await api.tasks({
   589→        status: statusFilter !== "all" ? statusFilter : undefined,
   590→        assignee: assigneeFilter !== "all" ? assigneeFilter : undefined,
   591→        overdue: onlyOverdue ? "true" : undefined,
   592→      });
   593→      setTasks(res.tasks);
   594→    } catch (err) {
   595→      setError(err instanceof Error ? err.message : "Gagal memuat task");
   596→      if (!silent) toast.error("Gagal memuat task follow-up");
   597→    } finally {
   598→      setLoading(false);
   599→    }
   600→  }, [statusFilter, assigneeFilter, onlyOverdue]);
   601→
   602→  useEffect(() => { void load(); }, [load]);
   603→
   604→  const assignees = useMemo(() => {
   605→    const set = new Set<string>();
   606→    (tasks ?? []).forEach((t) => { if (t.assigneeName) set.add(t.assigneeName); });
   607→    return Array.from(set).sort((a, b) => a.localeCompare(b));
   608→  }, [tasks]);
   609→
   610→  const stats = useMemo(() => {
   611→    const list = tasks ?? [];
   612→    const todayEnd = endOfToday().getTime();
   613→    const weekAgo = todayEnd - 7 * 24 * 60 * 60 * 1000;
   614→    return {
   615→      open: list.filter((t) => t.status === "open").length,
   616→      overdue: list.filter((t) => isOverdue(t)).length,
   617→      dueToday: list.filter((t) => t.status === "open" && isToday(t)).length,
   618→      doneThisWeek: list.filter((t) => t.status === "done" && t.completedAt
   619→        && new Date(t.completedAt).getTime() >= weekAgo
   620→        && new Date(t.completedAt).getTime() <= todayEnd).length,
   621→    };
   622→  }, [tasks]);
   623→
   624→  const groups = useMemo(() => {
   625→    const list = tasks ?? [];
   626→    return {
   627→      overdue: list.filter((t) => isOverdue(t)),
   628→      today: list.filter((t) => t.status === "open" && isToday(t)),
   629→      upcoming: list.filter((t) => t.status === "open" && !isOverdue(t) && !isToday(t)),
   630→      done: list.filter((t) => t.status === "done"),
   631→    };
   632→  }, [tasks]);
   633→
   634→  async function handleToggle(t: TaskDTO) {
   635→    const nextStatus = t.status === "done" ? "open" : "done";
   636→    setBusyId(t.id);
   637→    try {
   638→      const res = await api.updateTask(t.id, { status: nextStatus });
   639→      setTasks((prev) => (prev ?? []).map((x) => (x.id === t.id ? res.task : x)));
   640→      toast.success(nextStatus === "done" ? "Task ditandai selesai" : "Task dibuka kembali");
   641→    } catch (err) {
   642→      toast.error(err instanceof Error ? err.message : "Gagal memperbarui task");
   643→    } finally {
   644→      setBusyId(null);
   645→    }
   646→  }
   647→
   648→  async function handleCreate(e: React.FormEvent) {
   649→    e?.preventDefault();
   650→    if (!form.title.trim()) { toast.error("Judul task wajib diisi"); return; }
   651→    setSaving(true);
   652→    try {
   653→      await api.createTask({
   654→        title: form.title.trim(),
   655→        type: form.type,
   656→        priority: form.priority,
   657→        assigneeName: form.assigneeName.trim() || undefined,
   658→        dueDate: form.dueDate || undefined,
   659→        opportunityId: form.opportunityId.trim() || undefined,
   660→        actorName: user?.name ?? "System",
   661→        actorRole: user?.role ?? "system",
   662→      });
   663→      toast.success("Task follow-up dibuat");
   664→      setCreateOpen(false);
   665→      setForm({ title: "", type: "follow_up", priority: "medium", assigneeName: "", dueDate: "", opportunityId: "" });
   666→      await load(true);
   667→    } catch (err) {
   668→      toast.error(err instanceof Error ? err.message : "Gagal membuat task");
   669→    } finally {
   670→      setSaving(false);
   671→    }
   672→  }
   673→
   674→  /** Callback setelah pesan outbound tercatat: perbarui status task di list. */
   675→  function handleSent(_task: TaskDTO, updated?: TaskDTO) {
   676→    if (updated) {
   677→      setTasks((prev) => (prev ?? []).map((x) => (x.id === updated.id ? updated : x)));
   678→    }
   679→  }
   680→
   681→  if (loading && tasks === null) return <FollowupsSkeleton />;
   682→
   683→  const sectionBody = (list: TaskDTO[], emptyText: string): ReactNode =>
   684→    list.length === 0 ? (
   685→      <EmptyGroup text={emptyText} />
   686→    ) : (
   687→      <div className="space-y-3">
   688→        {list.map((t) => (
   689→          <TaskCard key={t.id} task={t} onToggle={handleToggle} busy={busyId === t.id} onSend={setSendTarget} />
   690→        ))}
   691→      </div>
   692→    );
   693→
   694→  return (
   695→    <div className="space-y-6">
   696→      {/* Header + filter */}
   697→      <div className="flex flex-col gap-3 sm:flex-row sm:items-start sm:justify-between">
   698→        <div>
   699→          <h1 className="text-xl font-bold tracking-tight text-zinc-900">Follow-up Center</h1>
   700→          <p className="text-sm text-zinc-500">Task &amp; jadwal follow-up per opportunity</p>
   701→        </div>
   702→        <div className="flex items-center gap-2">
   703→          <Button
   704→            variant="outline"
   705→            size="sm"
   706→            onClick={() => void load()}
   707→            disabled={loading}
   708→            aria-label="Muat ulang daftar task"
   709→          >
   710→            <RefreshCw className={`h-4 w-4 ${loading ? "animate-spin" : ""}`} aria-hidden /> Muat ulang
   711→          </Button>
   712→          <Button size="sm" onClick={() => setCreateOpen(true)} aria-label="Tambah task follow-up baru">
   713→            <Plus className="h-4 w-4" aria-hidden /> Tambah Task
   714→          </Button>
   715→        </div>
   716→      </div>
   717→
   718→      <div className="flex flex-col gap-3 rounded-xl border bg-white p-4 shadow-sm lg:flex-row lg:items-center lg:justify-between">
   719→        <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   720→          <div className="flex items-center gap-2">
   721→            <ListFilter className="h-4 w-4 shrink-0 text-zinc-400" aria-hidden />
   722→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   723→              <SelectTrigger className="w-full sm:w-[150px]" aria-label="Filter status task">
   724→                <SelectValue placeholder="Status" />
   725→              </SelectTrigger>
   726→              <SelectContent>
   727→                <SelectItem value="all">Semua Status</SelectItem>
   728→                <SelectItem value="open">Open</SelectItem>
   729→                <SelectItem value="done">Done</SelectItem>
   730→              </SelectContent>
   731→            </Select>
   732→          </div>
   733→          <Select value={assigneeFilter} onValueChange={setAssigneeFilter}>
   734→            <SelectTrigger className="w-full sm:w-[190px]" aria-label="Filter assignee">
   735→              <SelectValue placeholder="Assignee" />
   736→            </SelectTrigger>
   737→            <SelectContent>
   738→              <SelectItem value="all">Semua Assignee</SelectItem>
   739→              {assignees.map((a) => <SelectItem key={a} value={a}>{a}</SelectItem>)}
   740→            </SelectContent>
   741→          </Select>
   742→        </div>
   743→        <label className="flex cursor-pointer items-center gap-2 text-sm text-zinc-700">
   744→          <Switch checked={onlyOverdue} onCheckedChange={setOnlyOverdue} aria-label="Hanya tampilkan task overdue" />
   745→          Hanya overdue
   746→        </label>
   747→      </div>
   748→
   749→      {/* Stat strip */}
   750→      <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   751→        <StatCard label="Total Open" value={stats.open} icon={ListFilter} cls="bg-zinc-100 text-zinc-600" />
   752→        <StatCard label="Overdue" value={stats.overdue} icon={CalendarClock} cls="bg-rose-100 text-rose-600" />
   753→        <StatCard label="Due Hari Ini" value={stats.dueToday} icon={CalendarDays} cls="bg-amber-100 text-amber-600" />
   754→        <StatCard label="Selesai Minggu Ini" value={stats.doneThisWeek} icon={CheckCircle2} cls="bg-emerald-100 text-emerald-600" />
   755→      </div>
   756→
   757→      {error ? (
   758→        <div className="rounded-xl border border-rose-200 bg-rose-50 p-4 text-sm text-rose-700">
   759→          {error}
   760→        </div>
   761→      ) : null}
   762→
   763→      {/* List dikelompokkan */}
   764→      <div className="rounded-xl border bg-white p-4 shadow-sm sm:p-6">
   765→        {assigneeFilter !== "all" && assignees.length === 0 && tasks?.length === 0 ? (
   766→          <EmptyGroup text="Tidak ada task untuk filter ini" />
   767→        ) : null}
   768→        <div className="space-y-6">
   769→          {groups.overdue.length > 0 && (
   770→            <section aria-label="Task overdue" className="space-y-3">
   771→              <SectionHeading title="Overdue" count={groups.overdue.length} />
   772→              {sectionBody(groups.overdue, "Tidak ada task overdue")}
   773→            </section>
   774→          )}
   775→          <section aria-label="Task hari ini" className="space-y-3">
   776→            <SectionHeading title="Hari ini" count={groups.today.length} />
   777→            {sectionBody(groups.today, "Tidak ada task jatuh tempo hari ini")}
   778→          </section>
   779→          <section aria-label="Task mendatang" className="space-y-3">
   780→            <SectionHeading title="Mendatang" count={groups.upcoming.length} />
   781→            {sectionBody(groups.upcoming, "Tidak ada task mendatang")}
   782→          </section>
   783→          <section aria-label="Task selesai" className="space-y-3">
   784→            <SectionHeading title="Selesai" count={groups.done.length} />
   785→            {sectionBody(groups.done, "Belum ada task selesai")}
   786→          </section>
   787→        </div>
   788→      </div>
   789→
   790→      {/* Dialog tambah task */}
   791→      <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   792→        <DialogContent className="sm:max-w-lg">
   793→          <DialogHeader>
   794→            <DialogTitle>Tambah Task Follow-up</DialogTitle>
   795→            <DialogDescription>Buat task baru untuk jadwal follow-up, meeting, atau pekerjaan internal.</DialogDescription>
   796→          </DialogHeader>
   797→          <form onSubmit={handleCreate} className="grid gap-4 py-2">
   798→            <div className="grid gap-2">
   799→              <Label htmlFor="task-title">Judul Task *</Label>
   800→              <Input
   801→                id="task-title" value={form.title} required
   802→                onChange={(e) => setForm((f) => ({ ...f, title: e.target.value }))}
   803→                placeholder="Contoh: Follow-up proposal redesign website"
   804→              />
   805→            </div>
   806→            <div className="grid gap-4 sm:grid-cols-2">
   807→              <div className="grid gap-2">
   808→                <Label htmlFor="task-type">Tipe</Label>
   809→                <Select value={form.type} onValueChange={(v) => setForm((f) => ({ ...f, type: v }))}>
   810→                  <SelectTrigger id="task-type"><SelectValue /></SelectTrigger>
   811→                  <SelectContent>
   812→                    {TASK_TYPES.map((t) => <SelectItem key={t.key} value={t.key}>{t.label}</SelectItem>)}
   813→                  </SelectContent>
   814→                </Select>
   815→              </div>
   816→              <div className="grid gap-2">
   817→                <Label htmlFor="task-priority">Prioritas</Label>
   818→                <Select value={form.priority} onValueChange={(v) => setForm((f) => ({ ...f, priority: v }))}>
   819→                  <SelectTrigger id="task-priority"><SelectValue /></SelectTrigger>
   820→                  <SelectContent>
   821→                    {PRIORITIES.map((p) => (
   822→                      <SelectItem key={p} value={p}>{priorityMeta(p).label}</SelectItem>
   823→                    ))}
   824→                  </SelectContent>
   825→                </Select>
   826→              </div>
   827→            </div>
   828→            <div className="grid gap-4 sm:grid-cols-2">
   829→              <div className="grid gap-2">
   830→                <Label htmlFor="task-assignee">Assignee</Label>
   831→                <Input
   832→                  id="task-assignee" value={form.assigneeName}
   833→                  onChange={(e) => setForm((f) => ({ ...f, assigneeName: e.target.value }))}
   834→                  placeholder="Nama orang yang mengerjakan"
   835→                />
   836→              </div>
   837→              <div className="grid gap-2">
   838→                <Label htmlFor="task-due">Due Date</Label>
   839→                <Input
   840→                  id="task-due" type="date" value={form.dueDate}
   841→                  onChange={(e) => setForm((f) => ({ ...f, dueDate: e.target.value }))}
   842→                />
   843→              </div>
   844→            </div>
   845→            <div className="grid gap-2">
   846→              <Label htmlFor="task-opp">Opportunity ID (opsional)</Label>
   847→              <Input
   848→                id="task-opp" value={form.opportunityId}
   849→                onChange={(e) => setForm((f) => ({ ...f, opportunityId: e.target.value }))}
   850→                placeholder="Tempel ID opportunity bila terkait"
   851→              />
   852→            </div>
   853→            <DialogFooter className="mt-2">
   854→              <Button type="button" variant="outline" onClick={() => setCreateOpen(false)}>Batal</Button>
   855→              <Button type="submit" disabled={saving} aria-label="Simpan task baru">
   856→                {saving ? "Menyimpan…" : "Simpan Task"}
   857→              </Button>
   858→            </DialogFooter>
   859→          </form>
   860→        </DialogContent>
   861→      </Dialog>
   862→
   863→      {/* Dialog kirim & catat pesan outbound (Fase 3) */}
   864→      <SendOutboundDialog task={sendTarget} onClose={() => setSendTarget(null)} onSent={handleSent} />
   865→    </div>
   866→  );
   867→}
