    81→function formatSizeKb(bytes?: number | null): string {

    95→/** Parse item quotation defensif — items bisa JSON string atau array. */
    96→function parseQuoteItems(items: string | QuotationItemDTO[]): QuotationItemDTO[] {
    97→  if (Array.isArray(items)) return items;
    98→  try {
    99→    const parsed: unknown = JSON.parse(items);
   100→    return Array.isArray(parsed) ? (parsed as QuotationItemDTO[]) : [];
   101→  } catch {
   102→    return [];
   103→  }
   104→}

   106→/** Domain dari URL/email sederhana, untuk mencocokkan akun client dengan perusahaannya. */
   107→function domainOf(raw?: string | null): string | null {
   108→  if (!raw) return null;
   109→  const s = raw.trim().toLowerCase().replace(/^https?:\/\//, "").replace(/^www\./, "").split("/")[0].split("@").pop() ?? "";
   110→  return s.includes(".") ? s : null;
   111→}

   160→/** Satu baris deliverable di portal klien: ikon jenis, meta, status, dan aksi review (Setujui / Minta Revisi). */
   161→function PortalDeliverableRow({ d, user, canReview, onChanged }: {
   162→  d: DeliverableRowData;
   163→  user: SessionUser | null;
   164→  canReview: boolean;
   165→  onChanged: () => Promise<void>;
   166→}) {
   167→  const meta = dlvStatus(d.status);
   168→  const [review, setReview] = useState<null | "approved" | "revision">(null);
   169→  const [comment, setComment] = useState("");
   170→  const [busy, setBusy] = useState(false);
   171→  const KindIcon = d.kind === "file" ? Paperclip : Link2;
   172→
   173→  async function submitDecision() {
   174→    if (!user || !review) return;
   175→    setBusy(true);
   176→    try {
   177→      await api.decideDeliverable({
   178→        id: d.id,
   179→        decision: review,
   180→        reviewComment: comment.trim() || undefined,
   181→        reviewedBy: user.name,
   182→        reviewedRole: user.role, // "client" saat klien login; role staf saat mode pratinjau
   183→      });
   184→      toast.success(
   185→        review === "approved"
   186→          ? `“${d.name}” disetujui — terima kasih`
   187→          : `Revisi diminta untuk “${d.name}” — tim kami akan menindaklanjuti`
   188→      );
   189→      setReview(null);
   190→      setComment("");
   191→      await onChanged();
   192→    } catch (err) {
   193→      toast.error(err instanceof Error ? err.message : "Gagal menyimpan keputusan review");
   194→    } finally {
   195→      setBusy(false);
   196→    }
   197→  }
   198→
   199→  return (
   200→    <div className="rounded-lg border bg-white p-3">
   201→      <div className="flex items-start justify-between gap-2">
   202→        <div className="flex min-w-0 items-start gap-2.5">
   203→          <span className="mt-0.5 flex h-7 w-7 shrink-0 items-center justify-center rounded-full bg-zinc-100" aria-hidden>
   204→            <KindIcon className="h-3.5 w-3.5 text-zinc-500" />
   205→          </span>
   206→          <div className="min-w-0">
   207→            <p className="truncate text-sm font-semibold text-zinc-900">{d.name}</p>
   208→            <p className="truncate text-[11px] text-zinc-500">
   209→              {d.kind === "file"
   210→                ? [d.fileName, d.sizeBytes ? formatSizeKb(d.sizeBytes) : null].filter(Boolean).join(" · ")
   211→                : d.url}
   212→            </p>
   213→            {d.note ? <p className="mt-1 text-xs text-zinc-500">{d.note}</p> : null}
   214→            <p className="mt-1 text-[11px] text-zinc-400">
   215→              Dikirim {timeAgo(d.createdAt)}{d.createdBy ? ` oleh ${d.createdBy}` : ""}
   216→            </p>
   217→            {d.status !== "pending" && d.reviewedBy ? (
   218→              <p className="mt-1 text-[11px] text-zinc-500">
   219→                Anda sudah mereview{d.reviewedAt ? ` · ${timeAgo(d.reviewedAt)}` : ""}
   220→              </p>
   221→            ) : null}
   222→            {d.reviewComment ? (
   223→              <p className="mt-0.5 text-[11px] italic text-zinc-500">Catatan review: “{d.reviewComment}”</p>
   224→            ) : null}
   225→          </div>
   226→        </div>
   227→        <Badge variant="outline" className={`shrink-0 border-transparent px-1.5 ${meta.cls}`}>{meta.label}</Badge>
   228→      </div>
   229→
   230→      <div className="mt-2 flex flex-wrap items-center gap-x-3 gap-y-1.5">
   231→        {d.kind === "link" && d.url ? (
   232→          <a
   233→            href={d.url}
   234→            target="_blank"
   235→            rel="noreferrer"
   236→            className="inline-flex items-center gap-1 text-xs font-medium text-orange-600 hover:text-orange-700 hover:underline"
   237→          >
   238→            <ExternalLink className="h-3.5 w-3.5" aria-hidden /> Buka tautan
   239→          </a>
   240→        ) : null}
   241→        {d.kind === "file" && d.fileData ? (
   242→          <a
   243→            href={d.fileData}
   244→            download={d.fileName ?? undefined}
   245→            className="inline-flex items-center gap-1 text-xs font-medium text-orange-600 hover:text-orange-700 hover:underline"
   246→          >
   247→            <ReceiptText className="h-3.5 w-3.5" aria-hidden /> Unduh
   248→          </a>
   249→        ) : null}
   250→        {canReview ? (
   251→          <>
   252→            <Button
   253→              type="button"
   254→              variant="outline"
   255→              size="sm"
   256→              className="h-7 border-emerald-200 px-2 text-xs text-emerald-700 hover:bg-emerald-50 hover:text-emerald-800"
   257→              disabled={busy}
   258→              onClick={() => { setReview("approved"); setComment(d.reviewComment ?? ""); }}
   259→              aria-label={`Setujui deliverable ${d.name}`}
   260→            >
   261→              <Check className="h-3.5 w-3.5" aria-hidden /> Setujui
   262→            </Button>
   263→            <Button
   264→              type="button"
   265→              variant="outline"
   266→              size="sm"
   267→              className="h-7 border-amber-200 px-2 text-xs text-amber-700 hover:bg-amber-50 hover:text-amber-800"
   268→              disabled={busy}
   269→              onClick={() => { setReview("revision"); setComment(d.reviewComment ?? ""); }}
   270→              aria-label={`Minta revisi deliverable ${d.name}`}
   271→            >
   272→              <Pencil className="h-3.5 w-3.5" aria-hidden /> Minta Revisi
   273→            </Button>
   274→          </>
   275→        ) : null}
   276→      </div>
   277→
   278→      {review !== null ? (
   279→        <div className="mt-2.5 space-y-2 rounded-lg border border-zinc-200 bg-zinc-50 p-2.5">
   280→          <Textarea
   281→            rows={2}
   282→            value={comment}
   283→            onChange={(e) => setComment(e.target.value)}
   284→            placeholder={review === "approved" ? "Catatan persetujuan (opsional)…" : "Jelaskan bagian yang perlu direvisi…"}
   285→            aria-label="Catatan review deliverable"
   286→          />
   287→          <div className="flex items-center justify-end gap-2">
   288→            <Button
   289→              type="button"
   290→              variant="outline"
   291→              size="sm"
   292→              disabled={busy}
   293→              onClick={() => { setReview(null); setComment(""); }}
   294→              aria-label="Batal review deliverable"
   295→            >
   296→              Batal
   297→            </Button>
   298→            <Button
   299→              type="button"
   300→              size="sm"
   301→              disabled={busy}
   302→              onClick={() => void submitDecision()}
   303→              className={review === "approved" ? "bg-emerald-600 text-white hover:bg-emerald-700" : "bg-amber-600 text-white hover:bg-amber-700"}
   304→              aria-label="Konfirmasi keputusan review"

   308→            </Button>
   309→          </div>
   310→        </div>
   311→      ) : null}
   312→    </div>
   313→  );
   314→}

   316→/** Blok deliverable per project — lazy fetch saat dirender, refresh mandiri setelah review. */
   317→function PortalProjectDeliverables({ project, user, canReview }: {
   318→  project: ProjectDTO;
   319→  user: SessionUser | null;
   320→  canReview: boolean;
   321→}) {
   322→  const [deliverables, setDeliverables] = useState<ProjectDeliverableDTO[] | null>(null);
   323→  const [loadError, setLoadError] = useState(false);
   324→
   325→  /** Fetch daftar deliverable project ini (dipakai effect awal & refresh setelah review). */
   326→  const reload = useCallback(async () => {
   327→    try {
   328→      const res = await api.projectDeliverables(project.id);
   329→      setDeliverables(res.deliverables);
   330→      setLoadError(false);
   331→    } catch {
   332→      setDeliverables([]);
   333→      setLoadError(true);
   334→    }
   335→  }, [project.id]);
   336→
   337→  // Initial fetch — pola .then() agar setState tidak berjalan sinkron di body effect.
   338→  useEffect(() => {
   339→    let cancelled = false;
   340→    api.projectDeliverables(project.id)
   341→      .then((res) => { if (!cancelled) { setDeliverables(res.deliverables); setLoadError(false); } })
   342→      .catch(() => { if (!cancelled) { setDeliverables([]); setLoadError(true); } });
   343→    return () => { cancelled = true; };
   344→  }, [project.id]);
   345→
   346→  const pendingCount = (deliverables ?? []).filter((d) => d.status === "pending").length;
   347→
   348→  return (
   349→    <div className="rounded-xl border bg-white p-4 shadow-sm">
   350→      <div className="flex flex-wrap items-center justify-between gap-2">
   351→        <div className="flex min-w-0 items-center gap-2">
   352→          <p className="truncate text-sm font-semibold text-zinc-900">{project.name}</p>
   353→          <span className="shrink-0 rounded-md bg-zinc-100 px-1.5 py-0.5 font-mono text-[11px] text-zinc-500">{project.code}</span>
   354→        </div>
   355→        {deliverables !== null && deliverables.length > 0 ? (
   356→          pendingCount > 0 ? (
   357→            <span className="rounded-full bg-amber-100 px-2 py-0.5 text-[11px] font-medium text-amber-700">
   358→              {pendingCount} menunggu review Anda
   359→            </span>
   360→          ) : (
   361→            <span className="rounded-full bg-emerald-100 px-2 py-0.5 text-[11px] font-medium text-emerald-700">
   362→              semua sudah direview
   363→            </span>
   364→          )
   365→        ) : null}
   366→      </div>
   367→      <div className="mt-3">
   368→        {deliverables === null ? (
   369→          <div className="space-y-2" aria-hidden>
   370→            <Skeleton className="h-16 rounded-lg" />
   371→            <Skeleton className="h-16 rounded-lg" />
   372→          </div>
   373→        ) : loadError ? (
   374→          <div className="rounded-lg border border-rose-200 bg-rose-50 p-3 text-xs text-rose-700">
   375→            Gagal memuat deliverable.{" "}
   376→            <button type="button" className="font-medium underline" onClick={() => void reload()}>
   377→              Coba lagi
   378→            </button>
   379→          </div>
   380→        ) : deliverables.length === 0 ? (
   381→          <p className="rounded-lg border border-dashed border-zinc-200 p-3 text-center text-xs text-zinc-400">
   382→            Belum ada deliverable untuk project ini.
   383→          </p>
   384→        ) : (
   385→          <div className="max-h-96 space-y-2 overflow-y-auto crm-scroll">
   386→            {deliverables.map((d) => (
   387→              <PortalDeliverableRow key={d.id} d={d} user={user} canReview={canReview} onChanged={reload} />
   388→            ))}
   389→          </div>
   390→        )}
   391→      </div>
   392→    </div>
   393→  );
   394→}
