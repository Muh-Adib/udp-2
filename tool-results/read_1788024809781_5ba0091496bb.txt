   114→function parseAiSummary(text: string): { label: string; body: string }[] {
   115→  const sections: { label: string; body: string }[] = [];
   116→  let current: { label: string; body: string } | null = null;
   117→  for (const raw of text.split(/\r?\n/)) {
   118→    const line = raw.replace(/\*\*/g, "").trim();
   119→    if (!line) {
   120→      if (current) current.body += "\n";
   121→      continue;
   122→    }
   123→    const key = Object.keys(AI_LABELS).find((k) => line.toUpperCase().startsWith(k));
   124→    if (key) {
   125→      if (current) sections.push(current);
   126→      const rest = line.slice(key.length).replace(/^[:\-\u2013\s]+/, "");
   127→      current = { label: AI_LABELS[key], body: rest };
   128→    } else if (current) {
   129→      current.body += (current.body ? "\n" : "") + line;
   130→    }
   131→  }
   132→  if (current) sections.push(current);
   133→  return sections;
   134→}

   136→function startOfToday(): number {
   137→  const d = new Date();
   138→  d.setHours(0, 0, 0, 0);
   139→  return d.getTime();
   140→}

   142→function isPastDate(value?: string | null): boolean {
   143→  if (!value) return false;
   144→  return new Date(value).getTime() < startOfToday();
   145→}
