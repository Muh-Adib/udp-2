   621→function ContactDetailBody({
   622→  contact,
   623→  companies,
   624→  onSaved,
   625→  onDeleted,
   626→  onOpenCompany,
   627→}: {
   628→  contact: ContactRecord;
   629→  companies: CompanyRecord[];
   630→  onSaved: (contact: ContactRef) => void;
   631→  onDeleted: () => void;
   632→  onOpenCompany: (company: CompanyRef) => void;
   633→}) {
   634→  const user = useCrmStore((s) => s.user);
   635→  const [editing, setEditing] = useState(false);
   636→  const [saving, setSaving] = useState(false);
   637→  const [form, setForm] = useState<ContactFormValues>({
   638→    firstName: contact.firstName ?? "",
   639→    lastName: contact.lastName ?? "",
   640→    position: contact.position ?? "",
   641→    email: contact.email ?? "",
   642→    whatsapp: contact.whatsapp ?? "",
   643→    phone: contact.phone ?? "",
   644→    companyName: contact.company?.name ?? "",
   645→    companyId: contact.companyId || NO_VALUE,
   646→    city: contact.city ?? "",
   647→    country: contact.country ?? "",
   648→    preferredChannel: contact.preferredChannel || "whatsapp",
   649→    tagsText: "",
   650→  });
   651→  const [tags, setTags] = useState<string[]>(() => parseJsonArray(contact.tags));
   652→  const [tagInput, setTagInput] = useState("");
   653→  const [tagBusy, setTagBusy] = useState(false);
   654→  const [deleting, setDeleting] = useState(false);
   655→
   656→  const PreferredIcon = CHANNEL_ICONS[contact.preferredChannel] ?? Globe;

   659→
   660→  function patchForm(p: Partial<ContactFormValues>) {
   661→    setForm((v) => ({ ...v, ...p }));
   662→  }
   663→
   664→  async function handleSave(e: FormEvent<HTMLFormElement>) {
   665→    e.preventDefault();
   666→    if (!form.firstName.trim()) {
   667→      toast.error("Nama depan wajib diisi");
   668→      return;
   669→    }
   670→    setSaving(true);
   671→    try {
   672→      const res = await api.updateContact(contact.id, {
   673→        firstName: form.firstName.trim(),
   674→        lastName: form.lastName.trim() || null,
   675→        position: form.position.trim() || null,
   676→        email: form.email.trim() || null,
   677→        whatsapp: form.whatsapp.trim() || null,
   678→        phone: form.phone.trim() || null,
   679→        city: form.city.trim() || null,
   680→        country: form.country.trim() || null,
   681→        preferredChannel: form.preferredChannel,
   682→        companyId: form.companyId === NO_VALUE ? null : form.companyId,

   684→        actorRole: user?.role ?? "system",
   685→      });
   686→      toast.success("Perubahan contact tersimpan");
   687→      setEditing(false);
   688→      onSaved(res.contact);
   689→    } catch (err) {
   690→      toast.error(err instanceof Error ? err.message : "Gagal menyimpan perubahan");
   691→    } finally {
   692→      setSaving(false);
   693→    }
   694→  }
   695→
   696→  async function patchTags(next: string[]) {
   697→    setTagBusy(true);
   698→    try {
   699→      const res = await api.updateContact(contact.id, {
   700→        tags: next,
   701→        actorName: user?.name ?? "System",
   702→        actorRole: user?.role ?? "system",
   703→      });
   704→      setTags(next);
   705→      onSaved(res.contact);
   706→    } catch (err) {
   707→      toast.error(err instanceof Error ? err.message : "Gagal memperbarui tag");
   708→    } finally {
   709→      setTagBusy(false);
   710→    }
   711→  }
   712→
   713→  function addTag() {
   714→    const t = tagInput.trim();
   715→    if (!t || tags.includes(t)) {
   716→      setTagInput("");
   717→      return;
   718→    }
   719→    setTagInput("");
   720→    void patchTags([...tags, t]);
   721→  }
   722→
   723→  function removeTag(t: string) {
   724→    void patchTags(tags.filter((x) => x !== t));
   725→  }
   726→
   727→  async function handleDelete() {

   732→        headers: { "Content-Type": "application/json" },
   733→      });

   736→        throw new Error(data.error ?? `HTTP ${res.status}`);
   737→      }
   738→      toast.success(`Contact "${contact.fullName}" dihapus`);
   739→      onDeleted();
   740→    } catch (err) {
   741→      toast.error(err instanceof Error ? err.message : "Gagal menghapus contact");
   742→      setDeleting(false);
   743→    }
   744→  }
   745→
   746→  return (
   747→    <div className="flex flex-1 flex-col">
   748→      <SheetHeader className="border-b border-zinc-100">
   749→        <div className="flex items-start gap-3 pr-8">
   750→          <AvatarBubble name={contact.fullName} size="lg" />
   751→          <div className="min-w-0 flex-1">
   752→            <SheetTitle className="leading-tight">{contact.fullName}</SheetTitle>

   765→          </Button>
   766→        </div>
   767→      </SheetHeader>
   768→
   769→      <div className="flex-1 space-y-5 p-4">
   770→        {editing ? (
   771→          <form onSubmit={handleSave} className="space-y-4 rounded-xl border bg-white p-4 shadow-sm">
   772→            <p className="text-sm font-semibold text-zinc-900">Edit Contact</p>
   773→            <ContactFormFields
   774→              mode="edit"
   775→              values={form}
   776→              onChange={patchForm}
   777→              disabled={saving}
   778→              companies={companies}
   779→            />
   780→            <div className="flex justify-end gap-2">
   781→              <Button type="button" variant="outline" onClick={() => setEditing(false)} disabled={saving}>
   782→                Batal
   783→              </Button>
   784→              <Button type="submit" className="bg-zinc-900 text-white hover:bg-zinc-800" disabled={saving}>
   785→                {saving && <Loader2 className="size-4 animate-spin" />} Simpan Perubahan
   786→              </Button>
   787→            </div>
   788→          </form>
   789→        ) : (
   790→          <section aria-label="Informasi contact">
   791→            <SectionTitle>Informasi</SectionTitle>
   792→            <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   793→              <InfoRow icon={Mail} label="Email" value={contact.email} />
   794→              {contact.emailAlt ? <InfoRow icon={MailPlus} label="Email alternatif" value={contact.emailAlt} /> : null}
   795→              <InfoRow icon={MessageCircle} label="WhatsApp" value={contact.whatsapp} />
   796→              <InfoRow icon={Phone} label="Telepon" value={contact.phone} />

   800→              <InfoRow icon={PreferredIcon} label="Kanal preferensi" value={channelLabel(contact.preferredChannel)} />
   801→              {contact.linkedin ? <InfoRow icon={Linkedin} label="LinkedIn" value={contact.linkedin} /> : null}
   802→              <div className="flex items-start gap-3">
   803→                <span className="mt-0.5 flex size-8 shrink-0 items-center justify-center rounded-lg bg-zinc-100 text-zinc-500">
   804→                  <ShieldCheck className="size-4" />
   805→                </span>
   806→                <div>
   807→                  <p className="text-[11px] font-medium uppercase tracking-wide text-zinc-400">Consent</p>
   808→                  <div className="mt-0.5">
   809→                    <ConsentBadge status={contact.consentStatus} />
   810→                  </div>
   811→                </div>
   812→              </div>
   813→            </div>
   814→          </section>
   815→        )}
   816→
   817→        <section aria-label="Tag contact">
   818→          <SectionTitle>Tag</SectionTitle>
   819→          <div className="rounded-xl border bg-white p-3 shadow-sm">
   820→            <div className="flex flex-wrap gap-1.5">
   821→              {tags.length === 0 && <p className="text-xs text-zinc-400">Belum ada tag — tambahkan untuk segmentasi.</p>}

   826→                    type="button"
