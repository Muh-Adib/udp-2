    84→    84→function deltaPct(cur: number, prev?: number): number | null {
    85→    85→  if (prev === undefined || prev === 0) return null;
    86→    86→  const d = ((cur - prev) / prev) * 100;
    87→    87→  return Number.isFinite(d) ? d : null;
    88→    88→}
    89→    89→
    90→    90→function computeTrends(prev: DashboardData["kpi"] | null, cur: DashboardData["kpi"]): KpiTrendMap {
    91→    91→  const out: KpiTrendMap = {};
    92→    92→  for (const key of TREND_KEYS) {
    93→    93→    out[key] = deltaPct(cur[key], prev?.[key]);
    94→    94→  }
    95→    95→  return out;
    96→    96→}

    98→    98→function trendNode(delta: number | null, goodWhen: "up" | "down"): ReactNode {
    99→    99→  if (delta === null) return null;
   100→   100→  if (Math.abs(delta) < 0.1) {
   101→   101→    return (
   102→   102→      <span className="inline-flex items-center gap-1 text-zinc-400">
   103→   103→        <Minus className="h-3 w-3" aria-hidden /> stabil
   104→   104→      </span>
   105→   105→    );
   106→   106→  }
   107→   107→  const up = delta > 0;
   108→   108→  const good = up ? goodWhen === "up" : goodWhen === "down";
   109→   109→  const Icon = up ? TrendingUp : TrendingDown;
   110→   110→  return (
   111→   111→    <span className={`inline-flex items-center gap-1 font-medium ${good ? "text-emerald-600" : "text-rose-600"}`}>
   112→   112→      <Icon className="h-3 w-3 shrink-0" aria-hidden />
   113→   113→      {up ? "+" : ""}{delta.toFixed(1)}%
   114→   114→    </span>
   115→   115→  );
   116→   116→}

   124→   124→function formatHours(h: number): string {
   125→   125→  if (h <= 0) return "< 1 jam";

   127→   127→  return `${Math.round(h)} jam`;
