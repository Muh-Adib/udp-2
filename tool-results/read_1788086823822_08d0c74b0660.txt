    70→function escapeCsvField(value: string): string {
    71→  if (/[;"\r\n]/.test(value)) return `"${value.replace(/"/g, '""')}"`;
    72→  return value;

    75→function buildProjectsCsv(projects: ProjectDTO[]): string {
    76→  const lines: string[] = [PROJECT_CSV_HEADERS.join(";")];
    77→  for (const p of projects) {
    78→    const ms = p.milestones ?? [];
    79→    const fields: string[] = [
    80→      p.code,
    81→      p.name,
    82→      p.brand?.name ?? "",
    83→      p.company?.name ?? "",
    84→      p.serviceCategory ?? "",
    85→      statusMeta(p.status).label,
    86→      String(p.progress),
    87→      p.pmName ?? "",
    88→      p.startDate ? formatDate(p.startDate) : "",
    89→      p.dueDate ? formatDate(p.dueDate) : "",
    90→      p.contractValue != null ? String(p.contractValue) : "",
    91→      p.budgetInternal != null ? String(p.budgetInternal) : "",
    92→      String(ms.filter((m) => m.status === "done").length),
    93→      String(ms.length),
    94→    ];
    95→    lines.push(fields.map(escapeCsvField).join(";"));
    96→  }
    97→  return "\uFEFF" + lines.join("\r\n");
    98→}

   320→function TimelineView({ projects, onOpen }: { projects: ProjectDTO[]; onOpen: (p: ProjectDTO) => void }) {
   321→  const range = useMemo(() => {
   322→    const now = Date.now();
   323→    let min: number | null = null;
   324→    let max: number | null = null;
   325→    for (const p of projects) {
   326→      let contributed = false;
   327→      const track = (raw?: string | Date | number | null) => {
   328→        if (!raw) return;
   329→        const ts = new Date(raw).getTime();
   330→        if (!Number.isFinite(ts)) return;
   331→        min = min === null ? ts : Math.min(min, ts);
   332→        max = max === null ? ts : Math.max(max, ts);
   333→        contributed = true;
   334→      };
   335→      track(p.startDate);
   336→      track(p.dueDate);
   337→      for (const m of p.milestones ?? []) track(m.dueDate);
   338→      if (!contributed) track(now); // fallback: project tanpa tanggal → acuan hari ini
   339→    }
   340→    if (min === null || max === null) return { min: now - 15 * DAY_MS, max: now + 15 * DAY_MS };
   341→    const pad = 7 * DAY_MS;
   342→    return { min: min - pad, max: max + pad };
   343→  }, [projects]);
   344→
   345→  const span = Math.max(range.max - range.min, DAY_MS);
   346→  const pct = useCallback((ts: number) => Math.max(0, Math.min(100, ((ts - range.min) / span) * 100)), [range.min, span]);
   347→  const nowPct = pct(Date.now());
   348→
   349→  const gridlines = useMemo(() => {
   350→    const n = 6;
   351→    return Array.from({ length: n }, (_, i) => {
   352→      const ratio = i / (n - 1);
   353→      return { pct: ratio * 100, label: shortDate(range.min + ratio * span) };
   354→    });
   355→  }, [range.min, span]);
   356→
   357→  const todayTransform = nowPct <= 5 ? "translateX(0)" : nowPct >= 95 ? "translateX(-100%)" : "translateX(-50%)";
   358→
   359→  return (
   360→    <div className="overflow-x-auto crm-scroll rounded-xl border bg-white p-4 shadow-sm">
   361→      <div className="min-w-[760px]">
   362→        {/* Sumbu waktu (header) */}
   363→        <div className="flex">
   364→          <div className="w-[200px] shrink-0" />
   365→          <div className="relative h-7 flex-1">
   366→            {gridlines.map((g, i) => (
   367→              <span
   368→                key={i}
   369→                className="absolute top-0 whitespace-nowrap text-[10px] tabular-nums text-zinc-400"
   370→                style={{
   371→                  left: `${g.pct}%`,
   372→                  transform: i === 0 ? "translateX(0)" : i === gridlines.length - 1 ? "translateX(-100%)" : "translateX(-50%)",
   373→                }}
   374→              >
   375→                {g.label}
   376→              </span>
   377→            ))}
   378→            <span
   379→              className="absolute top-0 whitespace-nowrap text-[10px] font-semibold text-rose-500"
   380→              style={{ left: `${nowPct}%`, transform: todayTransform }}
   381→            >
   382→              Hari ini
   383→            </span>
   384→          </div>
   385→        </div>
   386→
   387→        {/* Baris project + gridline vertikal */}
   388→        <div className="relative">
   389→          <div className="pointer-events-none absolute inset-y-0 left-[200px] right-0" aria-hidden>
   390→            {gridlines.map((g, i) => (
   391→              <div key={i} className="absolute inset-y-0 border-l border-dashed border-zinc-200" style={{ left: `${g.pct}%` }} />
   392→            ))}
   393→            <div className="absolute inset-y-0 border-l-2 border-rose-400" style={{ left: `${nowPct}%` }} />
   394→          </div>
   395→          {projects.map((p) => (
   396→            <TimelineRow key={p.id} project={p} pct={pct} onOpen={onOpen} />
   397→          ))}
   398→        </div>
   399→
   400→        {/* Legenda */}
   401→        <div className="mt-3 flex flex-wrap items-center gap-x-4 gap-y-1.5 border-t border-zinc-100 pt-3 text-[11px] text-zinc-500">
   402→          <span className="flex items-center gap-1.5"><span className="h-2.5 w-5 rounded-full bg-zinc-400" aria-hidden /> Perencanaan</span>
   403→          <span className="flex items-center gap-1.5"><span className="h-2.5 w-5 rounded-full bg-zinc-900/90" aria-hidden /> Berjalan</span>
   404→          <span className="flex items-center gap-1.5"><span className="h-2.5 w-5 rounded-full bg-violet-500" aria-hidden /> Review</span>
   405→          <span className="flex items-center gap-1.5"><span className="h-2.5 w-5 rounded-full bg-emerald-500" aria-hidden /> Selesai</span>
   406→          <span className="flex items-center gap-1.5"><span className="h-2 w-2 rotate-45 rounded-[2px] bg-emerald-500" aria-hidden /> Milestone selesai</span>
   407→          <span className="flex items-center gap-1.5"><span className="h-2 w-2 rotate-45 rounded-[2px] bg-amber-500" aria-hidden /> Dikerjakan</span>
   408→          <span className="flex items-center gap-1.5"><span className="h-2 w-2 rotate-45 rounded-[2px] bg-zinc-300" aria-hidden /> Menunggu</span>

   433→/** Format Date → nilai untuk <input type="date"> (yyyy-mm-dd lokal). */
   434→function toDateInputValue(d: Date | null): string {
   435→  if (!d || Number.isNaN(d.getTime())) return "";
   436→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
   437→}

   439→/** Parse nilai <input type="date"> → Date lokal pukul 12:00 (aman dari geser tanggal). */
   440→function parseDateInputValue(v: string): Date | null {
   441→  const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(v.trim());
   442→  if (!m) return null;
   443→  const d = new Date(Number(m[1]), Number(m[2]) - 1, Number(m[3]), 12, 0, 0, 0);
   444→  return Number.isNaN(d.getTime()) ? null : d;

  1030→  /** Fase 3c / Ronde 16 — simpan dialog: ubah dueDate project ATAU dueDate+status milestone (optimis + revert bila gagal). */
  1031→  async function handleSaveRescheduleDialog() {
  1032→    if (!rescheduleTarget || !user) return;
  1033→    // Ambil data terkini dari state (snapshot dialog bisa basi setelah update optimis sebelumnya)
  1034→    const project = (projects ?? []).find((p) => p.id === rescheduleTarget.project.id) ?? rescheduleTarget.project;
  1035→    const newDate = parseDateInputValue(rescheduleDate);
  1036→    if (!newDate) {
  1037→      toast.error("Pilih tanggal deadline yang valid");
  1038→      return;
  1039→    }
  1040→
  1041→    // --- Ronde 16: varian deadline project (tanpa status) ---
  1042→    if (rescheduleTarget.kind === "project") {
  1043→      const oldDue = project.dueDate ? new Date(project.dueDate) : null;
  1044→      if (oldDue && isSameLocalDay(oldDue, newDate)) return; // guard: tidak ada perubahan
  1045→
  1046→      setRescheduleSaving(true);
  1047→      patchProjectDueDateState(project.id, newDate.toISOString()); // optimis
  1048→      try {
  1049→        await api.updateProject({
  1050→          id: project.id,
  1051→          dueDate: newDate.toISOString(),
  1052→          actorName: user.name,
  1053→          actorRole: user.role,
  1054→        });
  1055→        toast.success(`Deadline ${project.code}: ${oldDue ? formatDate(oldDue) : "tanpa tanggal"} → ${formatDate(newDate)}`, {
  1056→          description: project.name,
  1057→        });
  1058→        setRescheduleTarget(null); // tutup pada sukses
  1059→      } catch (err) {
  1060→        patchProjectDueDateState(project.id, oldDue ? oldDue.toISOString() : null); // revert
  1061→        toast.error(err instanceof Error ? err.message : "Gagal menjadwalkan ulang deadline project");
  1062→      } finally {
  1063→        setRescheduleSaving(false);
  1064→      }
  1065→      return;
  1066→    }
  1067→
  1068→    // --- Fase 3c: varian milestone (dueDate + status) ---
  1069→    const m = project.milestones?.find((x) => x.id === rescheduleTarget.milestone.id) ?? rescheduleTarget.milestone;
  1070→    const oldDue = m.dueDate ? new Date(m.dueDate) : null;
  1071→    const dateChanged = !oldDue || !isSameLocalDay(oldDue, newDate);
  1072→    const statusChanged = rescheduleStatus !== m.status;
  1073→    if (!dateChanged && !statusChanged) return; // guard: tidak ada perubahan
  1074→
  1075→    setRescheduleSaving(true);
  1076→    patchMilestoneState(project.id, m.id, {
  1077→      dueDate: dateChanged ? newDate.toISOString() : undefined,
  1078→      status: statusChanged ? rescheduleStatus : undefined,
  1079→    }); // optimis
  1080→    try {
  1081→      await api.updateMilestone({
  1082→        milestoneId: m.id,
  1083→        dueDate: dateChanged ? newDate.toISOString() : undefined,
  1084→        status: statusChanged ? rescheduleStatus : undefined,
  1085→        actorName: user.name,
  1086→        actorRole: user.role,
  1087→      });
  1088→      if (dateChanged) {
  1089→        toast.success(`${m.name}: ${oldDue ? formatDate(oldDue) : "tanpa tanggal"} → ${formatDate(newDate)}`, {
  1090→          description: statusChanged ? `Status: ${msMeta(m.status).label} → ${msMeta(rescheduleStatus).label}` : undefined,
  1091→        });
  1092→      } else {
  1093→        toast.success(`Status milestone "${m.name}" diperbarui`, {
  1094→          description: `${msMeta(m.status).label} → ${msMeta(rescheduleStatus).label}`,
  1095→        });
  1096→      }
  1097→      setRescheduleTarget(null); // tutup pada sukses
  1098→      if (statusChanged) await refreshDetail(); // sinkronkan ulang data project
  1099→    } catch (err) {
  1100→      patchMilestoneState(project.id, m.id, {
  1101→        dueDate: dateChanged ? (oldDue ? oldDue.toISOString() : null) : undefined,
  1102→        status: statusChanged ? m.status : undefined,
  1103→      }); // revert
  1104→      toast.error(err instanceof Error ? err.message : "Gagal menjadwalkan ulang milestone");
  1105→    } finally {
  1106→      setRescheduleSaving(false);
  1107→    }
  1108→  }
