    81→function escapeCsvField(value: string): string {
    82→  if (/[;"\r\n]/.test(value)) return `"${value.replace(/"/g, '""')}"`;
    83→  return value;
    84→}
    85→
    86→function buildOpportunitiesCsv(opps: OpportunityDTO[]): string {
    87→  const lines: string[] = [OPP_CSV_HEADERS.join(";")];
    88→  for (const o of opps) {
    89→    const fields: string[] = [
    90→      o.title,
    91→      o.brand?.name ?? "",
    92→      o.company?.name ?? "",
    93→      o.contact?.fullName ?? "",
    94→      o.serviceCategory ?? "",
    95→      o.serviceName ?? "",
    96→      stageLabel(o.stage),
    97→      o.temperature,
    98→      o.priority,
    99→      String(o.score ?? ""),
   100→      o.estimatedValue != null ? String(o.estimatedValue) : "",
   101→      o.currency,
   102→      String(o.probability),
   103→      o.ownerName ?? "",
   104→      o.expectedCloseDate ? formatDate(o.expectedCloseDate) : "",
   105→      o.nextAction ?? "",
   106→      formatDate(o.createdAt),
   107→    ];
   108→    lines.push(fields.map(escapeCsvField).join(";"));
   109→  }
   110→  return "\uFEFF" + lines.join("\r\n");
   111→}

   122→function startOfToday(): number {
   123→  const d = new Date();
   124→  d.setHours(0, 0, 0, 0);
   125→  return d.getTime();

   128→function isOverdue(opp: OpportunityDTO): boolean {
   129→  if (!opp.nextActionDate || TERMINAL_STAGES.includes(opp.stage)) return false;
   130→  return new Date(opp.nextActionDate).getTime() < startOfToday();
   131→}

   133→function priorityChip(priority: string): string {
   134→  if (priority === "urgent") return "bg-red-100 text-red-700";
   135→  if (priority === "high") return "bg-amber-100 text-amber-700";
   136→  return "bg-zinc-100 text-zinc-600";
   137→}

   139→function temperatureDot(temperature: string): string {
   140→  if (temperature === "hot") return "bg-red-600";

   142→  return "bg-cyan-600";

   561→                competitor: competitor.trim() || undefined,
   562→              };
   563→              setLostReason("");
   564→              setLostNotes("");
   565→              setCompetitor("");
   566→              onConfirm(extra);
   567→            }}
   568→          >
   569→            Tandai Lost
   570→          </Button>
   571→        </DialogFooter>
   572→      </DialogContent>
   573→    </Dialog>
   574→  );
   575→}

   640→  const filtered = useMemo(() => {
   641→    if (owner === "all") return opps;
   642→    if (owner === UNASSIGNED) return opps.filter((o) => !o.ownerName || !o.ownerName.trim());
   643→    return opps.filter((o) => o.ownerName === owner);
   644→  }, [opps, owner]);
   645→
   646→  const tableRows = useMemo(() => {
   647→    let base = tempFilter === "all" ? filtered : filtered.filter((o) => o.temperature === tempFilter);
   648→    if (skorMin !== "all") {
   649→      const min = Number(skorMin);
   650→      base = base.filter((o) => scoreOf(o) >= min);
   651→    }
   652→    if (scoreSortDir) {
   653→      base = [...base].sort((a, b) =>
   654→        scoreSortDir === "asc" ? scoreOf(a) - scoreOf(b) : scoreOf(b) - scoreOf(a)
   655→      );
   656→    }
   657→    if (!sortDir) return base;
   658→    return [...base].sort((a, b) => {
   659→      const av = a.estimatedValue ?? 0;
   660→      const bv = b.estimatedValue ?? 0;
   661→      return sortDir === "asc" ? av - bv : bv - av;
   662→    });
   663→  }, [filtered, tempFilter, skorMin, scoreSortDir, sortDir]);
   664→
   665→  const grouped = useMemo(() => {
   666→    const map: Record<string, OpportunityDTO[]> = {};

   669→      const target = KANBAN_KEYS.includes(o.stage) ? o.stage : null;
   670→      if (target) map[target].push(o);
   671→    }
   672→    if (sortByScore) {
   673→      for (const key of KANBAN_KEYS) {
   674→        map[key].sort((a, b) => scoreOf(b) - scoreOf(a));
   675→      }
   676→    }
