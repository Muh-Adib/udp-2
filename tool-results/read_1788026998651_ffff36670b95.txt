   247→   247→  const brands = useCrmStore((s) => s.brands);
   248→   248→  const activeBrandFilter = useCrmStore((s) => s.activeBrandFilter);
   249→   249→  const setActiveBrandFilter = useCrmStore((s) => s.setActiveBrandFilter);
   250→   250→
   251→   251→  const [leads, setLeads] = useState<InboxLead[] | null>(null);
   252→   252→  const [loading, setLoading] = useState(true);
   253→   253→  const [error, setError] = useState<string | null>(null);

   255→   255→  const [selectedId, setSelectedId] = useState<string | null>(null);

   261→   261→  const detailRef = useRef<HTMLDivElement | null>(null);
   262→   262→
   263→   263→  const loadLeads = useCallback(async () => {

   272→   272→    } finally {
   273→   273→      setLoading(false);
   274→   274→    }
   275→   275→  }, [channelFilter, activeBrandFilter]);
   276→   276→
   277→   277→  useEffect(() => {
   278→   278→    void loadLeads();
   279→   279→  }, [loadLeads]);
   280→   280→
   281→   281→  const stats = useMemo(() => {
   282→   282→    const list = leads ?? [];
   283→   283→    return {
   284→   284→      total: list.length,
   285→   285→      late: list.filter((l) => l.slaHours > 24).length,
   286→   286→      duplicate: list.filter((l) => l.candidates.length > 0).length,
   287→   287→    };
   288→   288→  }, [leads]);
   289→   289→
   290→   290→  const selectedLead = useMemo(

   292→   292→    [leads, selectedId]

   298→   298→
   299→   299→  const convertMode: "new" | "link" | null = showNewContact ? "new" : linkedContactId ? "link" : null;
   300→   300→  const canConvert =
   301→   301→    convertMode !== null &&
   302→   302→    oppForm.brandId !== "" &&
   303→   303→    (convertMode !== "new" || contactForm.firstName.trim().length > 0);

   305→   305→  function handleSelectLead(lead: InboxLead) {

   307→   307→    setShowNewContact(false);
   308→   308→    setLinkedContactId(null);
   309→   309→

   312→   312→    const looksLikeEmail = rawSender.includes("@");
   313→   313→    setContactForm({
   314→   314→      ...EMPTY_CONTACT_FORM,

   320→   320→      brandId: lead.brandId ?? lead.brand?.id ?? "",

   322→   322→      ownerName: user?.name ?? "",

   324→   324→    if (typeof window !== "undefined" && window.innerWidth < 1024) {
   325→   325→      window.setTimeout(() => {
   326→   326→        detailRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });

   330→   330→
   331→   331→  function handleToggleCandidate(contactId: string) {

   333→   333→    setLinkedContactId((prev) => (prev === contactId ? null : contactId));

   336→   336→  function handleToggleNewContact() {

   338→   338→    setShowNewContact((prev) => !prev);
   339→   339→  }
   340→   340→
   341→   341→  async function handleConvert() {
   342→   342→    if (!selectedLead || !user || !convertMode) return;
   343→   343→    if (!oppForm.brandId) {
   344→   344→      toast.error("Brand wajib dipilih untuk opportunity.");
   345→   345→      return;
   346→   346→    }
   347→   347→    if (convertMode === "new" && !contactForm.firstName.trim()) {
   348→   348→      toast.error("Nama depan contact wajib diisi.");
   349→   349→      return;
   350→   350→    }
   351→   351→
   352→   352→    const opportunity: Record<string, unknown> = {
   353→   353→      brandId: oppForm.brandId,
   354→   354→      title:
   355→   355→        oppForm.title.trim() ||
   356→   356→        `Lead ${channelLabel(selectedLead.channel)} — ${(selectedLead.senderName ?? "Tanpa nama").trim().slice(0, 48)}`,
   357→   357→      ownerName: oppForm.ownerName.trim() || user.name,
   358→   358→      priority: oppForm.priority,
   359→   359→    };
   360→   360→    if (oppForm.serviceCategory) opportunity.serviceCategory = oppForm.serviceCategory;
   361→   361→    if (oppForm.serviceName) opportunity.serviceName = oppForm.serviceName;
   362→   362→    const est = Number(oppForm.estimatedValue);
   363→   363→    if (oppForm.estimatedValue.trim() !== "" && Number.isFinite(est) && est > 0) {
   364→   364→      opportunity.estimatedValue = est;
   365→   365→    }
   366→   366→
   367→   367→    const payload: Record<string, unknown> = {
   368→   368→      interactionId: selectedLead.id,
   369→   369→      action: convertMode,
   370→   370→      opportunity,
   371→   371→      actorName: user.name,
   372→   372→      actorRole: user.role,
   373→   373→    };
   374→   374→    if (convertMode === "link" && linkedContactId) payload.contactId = linkedContactId;
   375→   375→    if (convertMode === "new") {
   376→   376→      payload.contact = {
   377→   377→        firstName: contactForm.firstName.trim(),
