    21→## API contract ringkas (response JSON)
    22→- GET /api/dashboard → { kpi{totalLeads,openLeads,pipelineValue,weightedPipeline,winRate,wonValue,avgResponseHours,overdueTasks,unassignedLeads,outstandingInvoices}, funnel[{stage,count,value}], byBrand[{brandId,name,color,leads,won,value}], byChannel[{channel,count}], byCountry[{country,count}], forecast[{bucket,value}], lostReasons[{reason,count}], marketingPerf[{name,leads,won,avgResponseHours}], recentAudit[...], pipelineTrend[{label,created,won}], projectsAtRisk, productionCapacity }
    23→- GET /api/opportunities?stage&brandId&q&owner → { opportunities: OpportunityDTO[] (include brand, contact{company}, company) }
    24→- GET /api/opportunities/:id → { opportunity (include interactions[], tasks[], notes[], projects[milestones], invoices), related[] }
    25→- PATCH /api/opportunities/:id body { ...fields, actorName, actorRole } → stage change ternormalisasi + audit log; stage=won auto-buat project+milestone+invoice DP; stage=lost wajib lostReason → 400 jika kosong
    26→- POST /api/inbox/convert body { interactionId, action:"link"|"new", contactId?, contact?, opportunity{brandId,title,...}, actorName }
    27→- GET /api/inbox → { leads: InteractionDTO & { slaHours, candidates[] } }
    28→- POST /api/identify body { email?, whatsapp?, fullName?, companyName? } → { candidates[{contactId,score,reasons[],contact}] }
    29→- POST /api/contacts/merge body { primaryId, duplicateId, actorName }
