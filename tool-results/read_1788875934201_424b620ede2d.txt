    52→function formatBytes(n: number | null | undefined): string {
    53→  if (!n || n <= 0) return "";
    54→  if (n < 1024) return `${n} B`;
    55→  if (n < 1024 * 1024) return `${Math.round(n / 1024)} KB`;
    56→  return `${(n / (1024 * 1024)).toFixed(1)} MB`;
    57→}

    59→function attachmentIconOf(name: string): LucideIcon {
    60→  const ext = (name.split(".").pop() ?? "").toLowerCase();
    61→  if (["png", "jpg", "jpeg", "gif", "webp", "svg", "heic"].includes(ext)) return FileImage;
    62→  if (["pdf", "doc", "docx", "txt", "csv", "ppt", "pptx", "xls", "xlsx"].includes(ext)) return FileText;

    68→function readFileAsDataUrl(file: File): Promise<string> {
    69→  return new Promise((resolve, reject) => {
    70→    const reader = new FileReader();
    71→    reader.onload = () => resolve(String(reader.result ?? ""));
    72→    reader.onerror = () => reject(new Error("Gagal membaca file"));
    73→    reader.readAsDataURL(file);
    74→  });
    75→}

    77→/** Cocokkan 9 digit terakhir nomor (awalan +62/0 sering beda antar sumber). */
    78→function digitsMatch(a: string, b: string): boolean {
    79→  const da = a.replace(/\D+/g, "");
    80→  const db = b.replace(/\D+/g, "");
    81→  return da.length >= 9 && db.length >= 9 && da.slice(-9) === db.slice(-9);
    82→}

    84→/**
    85→ * Ronde 34-b — pencocokan identitas best-effort lead↔kontak (thread belum ter-link contactId):
    86→ * email di sender/recipient, 9 digit nomor, handle IG, atau nama persis.
    87→ */
    88→function leadMatchesContactIdentity(lead: InboxLead, c: ContactRef): boolean {
    89→  for (const raw of [(lead.senderName ?? "").trim(), (lead.recipientName ?? "").trim()]) {
    90→    if (!raw) continue;
    91→    const lower = raw.toLowerCase();
    92→    if (c.email && lower.includes(c.email.toLowerCase())) return true;
    93→    if (c.whatsapp && digitsMatch(lower, c.whatsapp)) return true;
    94→    if (c.phone && digitsMatch(lower, c.phone)) return true;
    95→    if (c.instagram && lower === c.instagram.toLowerCase()) return true;
    96→    if (c.fullName && lower === c.fullName.toLowerCase()) return true;
    97→  }
    98→  return false;
    99→}

   154→/**
   155→ * Ronde 25 — alamat tujuan satu kanal balasan dari data kontak/pengirim lead
   156→ * (sumber sama dgn backend reachableAddress: kontak dulu, lalu fallback dari senderName).
   157→ * socialProfile dibaca defensif — ada di model DB tapi belum ada di ContactRef DTO.
   158→ */
   159→function replyAddressFor(channel: string, lead: InboxLead): string | null {
   160→  const sender = (lead.senderName ?? "").trim();
   161→  const c = (lead.contact ?? null) as {
   162→    email?: string | null;
   163→    whatsapp?: string | null;
   164→    phone?: string | null;
   165→    instagram?: string | null;
   166→    socialProfile?: string | null;
   167→  } | null;
   168→  switch (channel) {
   169→    case "email":
   170→      return c?.email ?? extractEmailFromText(sender);
   171→    case "whatsapp":
   172→      return c?.whatsapp ?? (looksLikePhoneText(sender) ? sender : null);
   173→    case "instagram":
   174→      return c?.instagram
   175→        ?? (c?.socialProfile && isSocialHandle(c.socialProfile) ? c.socialProfile : null)
   176→        ?? (isSocialHandle(sender) || sender.startsWith("@") ? sender : null);
   177→    case "phone":
   178→      return c?.phone ?? c?.whatsapp ?? (looksLikePhoneText(sender) ? sender : null);
   179→    default:
   180→      return null;
   181→  }
   182→}

   209→  const label =

   217→  const Icon =
   218→    status === "read" || status === "delivered" ? CheckCheck

   223→    status === "read" ? "text-emerald-600"
   224→    : status === "failed" ? "text-red-600"
   225→    : status === "sent" || status === "delivered" ? "text-zinc-400"
   226→    : "text-zinc-300";
   227→  return (

   235→/** Fase 3 — SLA countdown per brand: sisa = SLA brand − jam tunggu (slaHours dari API). */
   236→function slaBadgeInfo(brandSlaHours: number, waitHours: number): {
   237→  label: string; className: string; tone: SlaTone;
   238→} {
   239→  const split = (hours: number) => {
   240→    let h = Math.floor(hours);
   241→    let m = Math.round((hours - h) * 60);
   242→    if (m >= 60) { h += 1; m = 0; }
   243→    return { h, m };
   244→  };
   245→  const fmt = (hours: number, prefix: string) => {
   246→    const { h, m } = split(hours);
   247→    if (h < 1) return `${prefix}${m}m`;
   248→    if (m === 0) return `${prefix}${h}j`;
   249→    return `${prefix}${h}j ${m}m`;
   250→  };
   251→
   252→  const remaining = brandSlaHours - waitHours;
   253→  if (remaining <= 0) {
   254→    return {
   255→      label: fmt(Math.abs(remaining), "Terlambat "),
   256→      className: "border-rose-200 bg-rose-50 text-rose-700",
   257→      tone: "breach" as const,
   258→    };
   259→  }
   260→  if (remaining <= brandSlaHours * 0.5) {
   261→    return {
   262→      label: fmt(remaining, "Segera: "),
   263→      className: "border-amber-200 bg-amber-50 text-amber-700",
   264→      tone: "warning" as const,
   265→    };
   266→  }
   267→  return {
   268→    label: fmt(remaining, "Sisa "),
   269→    className: "border-emerald-200 bg-emerald-50 text-emerald-700",
   270→    tone: "ok" as const,
   271→  };
   272→}

   452→  if (lead.thread) return lead.thread;

   477→/** Kanal paling sering muncul dalam thread — warna avatar kartu thread. */
   478→function dominantChannelOf(thread: ConversationThreadDTO, fallback: string): string {
   479→  const counts = new Map<string, number>();
   480→  for (const m of thread.messages) counts.set(m.channel, (counts.get(m.channel) ?? 0) + 1);
   481→  let best = fallback;
   482→  let bestN = -1;
   483→  for (const [ch, n] of counts) {
   484→    if (n > bestN) { best = ch; bestN = n; }
   485→  }
   486→  return best;
   487→}

   769→/** Salin teks + fallback legacy; false = clipboard ditolak browser (butuh salin manual). */
   770→async function copyTextSmart(text: string, successMsg: string): Promise<boolean> {
   771→  const legacyCopy = () => {
   772→    try {
   773→      const ta = document.createElement("textarea");
   774→      ta.value = text;
   775→      ta.setAttribute("readonly", "");
   776→      ta.style.position = "fixed";
   777→      ta.style.opacity = "0";
   778→      document.body.appendChild(ta);
   779→      ta.select();
   780→      const ok = document.execCommand("copy");
   781→      document.body.removeChild(ta);
   782→      return ok;
   783→    } catch {
   784→      return false;
   785→    }
   786→  };
   787→  try {
   788→    await navigator.clipboard.writeText(text);
   789→    toast.success(successMsg);
   790→    return true;
   791→  } catch {
   792→    if (legacyCopy()) {
   793→      toast.success(successMsg);
   794→      return true;
   795→    }
   796→    return false;
   797→  }
   798→}

   800→/** Label hari utk pemisah pesan chat: Hari ini / Kemarin / nama hari / tanggal. */
   801→function dayLabelOf(iso: string): string {
   802→  const d = new Date(iso);
   803→  const startOfDay = (x: Date) => new Date(x.getFullYear(), x.getMonth(), x.getDate()).getTime();
   804→  const diff = startOfDay(new Date()) - startOfDay(d);
   805→  const DAY = 86_400_000;
   806→  if (diff === 0) return "Hari ini";
   807→  if (diff === DAY) return "Kemarin";
   808→  if (diff > 0 && diff < 7 * DAY) return d.toLocaleDateString("id-ID", { weekday: "long" });
   809→  return d.toLocaleDateString("id-ID", { day: "numeric", month: "short", year: "numeric" });
   810→}
