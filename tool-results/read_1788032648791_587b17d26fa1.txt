   208→function hashColor(name: string): string {
   209→  let h = 0;
   210→  for (let i = 0; i < name.length; i++) h = (h * 31 + name.charCodeAt(i)) >>> 0;
   211→  return AVATAR_PALETTE[h % AVATAR_PALETTE.length];
   212→}

   214→function useDebounced<T>(value: T, delay = 300): T {
   215→  const [debounced, setDebounced] = useState(value);
   216→  useEffect(() => {
   217→    const t = setTimeout(() => setDebounced(value), delay);
   218→    return () => clearTimeout(t);
   219→  }, [value, delay]);
   220→  return debounced;
   221→}

   231→function consentMeta(status: string): { label: string; className: string } {
   232→  if (status === "granted") return { label: "Consent: Disetujui", className: "bg-emerald-100 text-emerald-700" };
   233→  if (status === "pending") return { label: "Consent: Menunggu", className: "bg-amber-100 text-amber-700" };
   234→  if (status === "declined") return { label: "Consent: Ditolak", className: "bg-rose-100 text-rose-700" };
   235→  return { label: status, className: "bg-zinc-100 text-zinc-600" };
   236→}

   238→function sizeBadgeClass(size?: string | null): string {
   239→  switch (size) {
   240→    case "enterprise":
   241→      return "bg-violet-100 text-violet-700";
   242→    case "government":
   243→      return "bg-zinc-200 text-zinc-800";
   244→    case "sme":
   245→      return "bg-emerald-100 text-emerald-700";
   246→    case "startup":
   247→      return "bg-orange-100 text-orange-700";
   248→    default:
   249→      return "bg-zinc-100 text-zinc-600";
   250→  }
   251→}
