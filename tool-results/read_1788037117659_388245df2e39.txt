   104→function deltaPct(cur: number, prev?: number): number | null {
   105→  if (prev === undefined || prev === 0) return null;
   106→  const d = ((cur - prev) / prev) * 100;
   107→  return Number.isFinite(d) ? d : null;
   108→}
   109→
   110→function computeTrends(prev: DashboardData["kpi"] | null, cur: DashboardData["kpi"]): KpiTrendMap {
   111→  const out: KpiTrendMap = {};
   112→  for (const key of TREND_KEYS) {
   113→    out[key] = deltaPct(cur[key], prev?.[key]);
   114→  }
   115→  return out;
   116→}

   118→function trendNode(delta: number | null, goodWhen: "up" | "down"): ReactNode {
   119→  if (delta === null) return null;
   120→  if (Math.abs(delta) < 0.1) {
   121→    return (
   122→      <span className="inline-flex items-center gap-1 text-zinc-400">
   123→        <Minus className="h-3 w-3" aria-hidden /> stabil
   124→      </span>
   125→    );
   126→  }
   127→  const up = delta > 0;
   128→  const good = up ? goodWhen === "up" : goodWhen === "down";
   129→  const Icon = up ? TrendingUp : TrendingDown;
   130→  return (
   131→    <span className={`inline-flex items-center gap-1 font-medium ${good ? "text-emerald-600" : "text-rose-600"}`}>
   132→      <Icon className="h-3 w-3 shrink-0" aria-hidden />
   133→      {up ? "+" : ""}{delta.toFixed(1)}%
   134→    </span>
   135→  );
   136→}

   138→function slaBadgeClass(hours: number): string {
   139→  if (hours <= 8) return "bg-emerald-100 text-emerald-700";
   140→  if (hours > 24) return "bg-rose-100 text-rose-700";
   141→  return "bg-amber-100 text-amber-700";

   144→function formatHours(h: number): string {
   145→  if (h <= 0) return "< 1 jam";
   146→  if (h < 1) return `${Math.round(h * 60)} mnt`;
   147→  return `${Math.round(h)} jam`;
   148→}

   150→function channelLabel(key: string): string {
   151→  const found = CHANNELS.find((c) => c.key === key);
   152→  if (found) return found.label;
   153→  return key
   154→    .split("_")
   155→    .map((w) => (w ? w.charAt(0).toUpperCase() + w.slice(1) : w))
   156→    .join(" ");
   157→}
   158→
   159→function actionBadgeClass(action: string): string {
   160→  return ACTION_BADGE_CLASS[action] ?? "bg-zinc-100 text-zinc-600";

   355→export default function DashboardModule() {
   356→  const brands = useCrmStore((s) => s.brands);
   357→  const user = useCrmStore((s) => s.user);
   358→  const setActiveModule = useCrmStore((s) => s.setActiveModule);
   359→
   360→  const [data, setData] = useState<DashboardFullData | null>(null);
   361→  const [loading, setLoading] = useState(true);
   362→  const [fetchTime, setFetchTime] = useState<Date | null>(null);
   363→  const [trends, setTrends] = useState<KpiTrendMap>({});
   364→  const prevKpiRef = useRef<DashboardData["kpi"] | null>(null);
   365→  const [confirmState, setConfirmState] = useState<ApprovalConfirmState | null>(null);
   366→  const [decisionNote, setDecisionNote] = useState("");
   367→  const [deciding, setDeciding] = useState(false);
   368→
   369→  const load = useCallback(async (silent: boolean) => {
   370→    if (!silent) setLoading(true);
   371→    try {
   372→      const res = normalizeDashboard(await api.dashboard());
   373→      setTrends(computeTrends(prevKpiRef.current, res.kpi));
   374→      prevKpiRef.current = res.kpi;
   375→      setData(res);
   376→      setFetchTime(new Date());
   377→    } catch (err) {
   378→      toast.error("Gagal memuat data Command Center", {
   379→        description: err instanceof Error ? err.message : "Silakan coba lagi.",
   380→      });
   381→    } finally {
   382→      setLoading(false);
   383→    }
   384→  }, []);
   385→
   386→  useEffect(() => {
   387→    void load(false);
   388→    const id = setInterval(() => void load(true), 60_000);
   389→    return () => clearInterval(id);
   390→  }, [load]);
   391→
   392→  const openConfirm = useCallback((approval: ApprovalRequestDTO, decision: "approve" | "reject") => {
   393→    setDecisionNote("");
   394→    setConfirmState({ approval, decision });
   395→  }, []);
   396→
   397→  const closeConfirm = useCallback(() => {
   398→    if (deciding) return;
   399→    setConfirmState(null);
   400→    setDecisionNote("");
   401→  }, [deciding]);
   402→
   403→  const submitDecision = useCallback(async () => {
   404→    if (!confirmState || !user || deciding) return;
   405→    const isReject = confirmState.decision === "reject";
   406→    const note = decisionNote.trim();
   407→    if (isReject && !note) {
   408→      toast.error("Catatan keputusan wajib diisi saat menolak pengajuan.");
   409→      return;
   410→    }
   411→    setDeciding(true);
   412→    try {
   413→      await api.decideApproval({
   414→        id: confirmState.approval.id,
   415→        decision: confirmState.decision,
   416→        decisionNote: note || undefined,
   417→        actorName: user.name,
   418→        actorRole: user.role,
   419→      });
   420→      toast.success(isReject ? "Pengajuan ditolak" : "Pengajuan disetujui", {
   421→        description: `${confirmState.approval.entityLabel ?? "Pengajuan"} — keputusan tercatat sebagai ${user.name}.`,
   422→      });
   423→      setConfirmState(null);
   424→      setDecisionNote("");
   425→      await load(true);
   426→    } catch (err) {
   427→      const msg = err instanceof Error ? err.message : "Silakan coba lagi.";
   428→      if (/403|direktur|akses|izin/i.test(msg)) {
   429→        toast.error("Akses ditolak", { description: msg });
   430→      } else {
   431→        toast.error("Gagal memutus pengajuan", { description: msg });
   432→      }
   433→    } finally {
   434→      setDeciding(false);
   435→    }
   436→  }, [confirmState, deciding, decisionNote, load, user]);
   437→
   438→  if (!data) {
   439→    if (loading) return <DashboardSkeleton />;
   440→    return (
   441→      <section aria-label="Error memuat data" className="rounded-xl border bg-white p-10 text-center shadow-sm">
   442→        <AlertTriangle className="mx-auto h-8 w-8 text-rose-500" aria-hidden />
   443→        <p className="mt-3 text-sm font-medium text-zinc-900">Gagal memuat data Command Center</p>
   444→        <p className="mt-1 text-xs text-zinc-500">Periksa koneksi server, lalu coba muat ulang.</p>
   445→        <Button size="sm" className="mt-4" onClick={() => void load(false)}>
   446→          <RefreshCw className="h-4 w-4" aria-hidden /> Coba lagi
   447→        </Button>
   448→      </section>
   449→    );
   450→  }
   451→
   452→  const kpi = data.kpi;
   453→  const canDecide = user?.role === "director" || user?.role === "super_admin";
   454→  const pendingApprovals = data.pendingApprovals ?? [];
   455→  // Fase 3 — SLA monitoring
   456→  const slaBreaches = data.slaBreaches ?? 0;
   457→  // Fase 2 Produksi — CR menunggu persetujuan klien
   458→  const pendingCRs = data.pendingChangeRequests ?? 0;
   459→  const canOpenInbox = canAccess("inbox", user?.role);
   460→  const activeBrandCount = brands.length > 0 ? brands.filter((b) => b.active).length : data.byBrand.length;
   461→  const firstBrandColor = brands[0]?.color ?? data.byBrand[0]?.color ?? "#ea580c";
   462→  const funnelMap = new Map(data.funnel.map((f) => [f.stage, f] as const));
   463→  const funnelMax = Math.max(...data.funnel.map((f) => f.count), 1);
   464→  const wonCount = data.funnel.find((f) => f.stage === "won")?.count ?? 0;
   465→
   466→  const bestMarketerIdx = data.marketingPerf.reduce((best, m, i, arr) => {
   467→    const cur = arr[best];
   468→    if (m.won > cur.won || (m.won === cur.won && m.leads > cur.leads)) return i;
   469→    return best;
   470→  }, 0);
   471→
   472→  const lostMax = Math.max(...data.lostReasons.map((r) => r.count), 1);
   473→
   474→  const kpiCards: KpiDef[] = [
   475→    {
   476→      key: "pipelineValue", label: "Pipeline Value", value: formatCurrency(kpi.pipelineValue),
   477→      icon: Wallet, hint: `${kpi.openLeads} lead terbuka`, accent: true, goodWhen: "up",
   478→    },
   479→    {
   480→      key: "weightedPipeline", label: "Weighted Forecast", value: formatCurrency(kpi.weightedPipeline),
   481→      icon: Target, hint: "estimasi probabilitas tertimbang", accent: true, goodWhen: "up",
   482→    },
   483→    {
   484→      key: "winRate", label: "Win Rate", value: `${kpi.winRate}%`,
   485→      icon: Trophy, hint: `${wonCount} deal berhasil ditutup`, accent: true, goodWhen: "up",
   486→    },
   487→    {
   488→      key: "wonValue", label: "Won Value", value: formatCurrency(kpi.wonValue),
   489→      icon: Handshake, hint: `${wonCount} deal won`, accent: true, goodWhen: "up",
   490→    },
