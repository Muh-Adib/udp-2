    78→function startOfToday(): number {
    79→  const d = new Date();
    80→  d.setHours(0, 0, 0, 0);
    81→  return d.getTime();
    82→}
    83→
    84→function isOverdue(opp: OpportunityDTO): boolean {
    85→  if (!opp.nextActionDate || TERMINAL_STAGES.includes(opp.stage)) return false;
    86→  return new Date(opp.nextActionDate).getTime() < startOfToday();
    87→}

    89→function priorityChip(priority: string): string {
    90→  if (priority === "urgent") return "bg-red-100 text-red-700";
    91→  if (priority === "high") return "bg-amber-100 text-amber-700";
    92→  return "bg-zinc-100 text-zinc-600";
    93→}
    94→
    95→function temperatureDot(temperature: string): string {
    96→  if (temperature === "hot") return "bg-red-600";
    97→  if (temperature === "warm") return "bg-amber-500";
    98→  return "bg-cyan-600";
    99→}

   370→function LostReasonDialog({
   371→  open,
   372→  saving,
   373→  onConfirm,
   374→  onCancel,
   375→}: {
   376→  open: boolean;
   377→  saving: boolean;
   378→  onConfirm: (extra: LostExtra) => void;
   379→  onCancel: () => void;
   380→}) {
   381→  const [lostReason, setLostReason] = useState("");
   382→  const [lostNotes, setLostNotes] = useState("");
   383→  const [competitor, setCompetitor] = useState("");
   384→
   385→  function resetAndClose() {
   386→    setLostReason("");
   387→    setLostNotes("");
   388→    setCompetitor("");
   389→    onCancel();
   390→  }

   392→  return (

   415→            value={competitor}

   427→        </div>
   428→        <DialogFooter>
   429→          <Button variant="outline" onClick={resetAndClose}>
   430→            Batal
   431→          </Button>

   433→            className="bg-red-600 hover:bg-red-700"
   434→            disabled={!lostReason || saving}
   435→            onClick={() => {
   436→              const extra: LostExtra = {
   437→                lostReason,
   438→                lostNotes: lostNotes.trim() || undefined,
   439→                competitor: competitor.trim() || undefined,

   442→              setLostNotes("");
   443→              setCompetitor("");
   444→              onConfirm(extra);

   446→          >
   447→            Tandai Lost
   448→          </Button>
   449→        </DialogFooter>
   450→      </DialogContent>
   451→    </Dialog>
   452→  );
   453→}

   457→export default function PipelineModule() {
   458→  const { user, brands, activeBrandFilter, setActiveBrandFilter } = useCrmStore();
   459→
   460→  const [opps, setOpps] = useState<OpportunityDTO[]>([]);
   461→  const [loading, setLoading] = useState(true);
   462→  const [qInput, setQInput] = useState("");
   463→  const [q, setQ] = useState("");
   464→  const [owner, setOwner] = useState<string>("all");
   465→  const [view, setView] = useState<ViewMode>("kanban");
   466→  const [tempFilter, setTempFilter] = useState<string>("all");
   467→  const [sortDir, setSortDir] = useState<SortDir>(null);
   468→  const [selectedId, setSelectedId] = useState<string | null>(null);
   469→  const [drawerOpen, setDrawerOpen] = useState(false);
   470→  const [pendingLost, setPendingLost] = useState<OpportunityDTO | null>(null);
   471→  const [savingStage, setSavingStage] = useState(false);
   472→
   473→  const oppsRef = useRef<OpportunityDTO[]>([]);
   474→  const draggedRecentlyRef = useRef(false);
   475→
   476→  const sensors = useSensors(useSensor(PointerSensor, { activationConstraint: { distance: 6 } }));
   477→
   478→  // Debounce pencarian
   479→  useEffect(() => {
   480→    const t = setTimeout(() => setQ(qInput), 300);
   481→    return () => clearTimeout(t);
   482→  }, [qInput]);
   483→
   484→  const load = useCallback(
   485→    async (opts?: { silent?: boolean }) => {
   486→      if (!opts?.silent) setLoading(true);
   487→      try {
   488→        const res = await api.opportunities({
   489→          brandId: activeBrandFilter !== "all" ? activeBrandFilter : undefined,
   490→          q: q.trim() ? q.trim() : undefined,
   491→        });
   492→        oppsRef.current = res.opportunities;
   493→        setOpps(res.opportunities);
   494→      } catch (e) {
   495→        toast.error(e instanceof Error ? e.message : "Gagal memuat opportunity");
   496→      } finally {
   497→        if (!opts?.silent) setLoading(false);
   498→      }
   499→    },
   500→    [activeBrandFilter, q]
   501→  );
   502→
   503→  useEffect(() => {
   504→    void load();
   505→  }, [load]);
   506→
   507→  const ownerOptions = useMemo(() => {
   508→    const names = new Set<string>();
   509→    for (const o of opps) {
   510→      if (o.ownerName && o.ownerName.trim()) names.add(o.ownerName.trim());
   511→    }
   512→    return Array.from(names).sort((a, b) => a.localeCompare(b));
   513→  }, [opps]);
   514→
   515→  const filtered = useMemo(() => {
   516→    if (owner === "all") return opps;
   517→    if (owner === UNASSIGNED) return opps.filter((o) => !o.ownerName || !o.ownerName.trim());
   518→    return opps.filter((o) => o.ownerName === owner);
   519→  }, [opps, owner]);
   520→
   521→  const tableRows = useMemo(() => {
   522→    const base = tempFilter === "all" ? filtered : filtered.filter((o) => o.temperature === tempFilter);
   523→    if (!sortDir) return base;
   524→    return [...base].sort((a, b) => {
   525→      const av = a.estimatedValue ?? 0;
   526→      const bv = b.estimatedValue ?? 0;
   527→      return sortDir === "asc" ? av - bv : bv - av;
   528→    });
   529→  }, [filtered, tempFilter, sortDir]);
   530→
   531→  const grouped = useMemo(() => {
   532→    const map: Record<string, OpportunityDTO[]> = {};
   533→    for (const key of KANBAN_KEYS) map[key] = [];
   534→    for (const o of filtered) {
   535→      const target = KANBAN_KEYS.includes(o.stage) ? o.stage : null;
   536→      if (target) map[target].push(o);
   537→    }
   538→    return map;
   539→  }, [filtered]);
   540→
   541→  function openDetail(id: string) {
   542→    setSelectedId(id);
   543→    setDrawerOpen(true);
   544→  }
   545→
   546→  function toggleSort() {
   547→    setSortDir((prev) => (prev === null ? "desc" : prev === "desc" ? "asc" : null));
   548→  }
   549→
   550→  async function moveOpportunity(opp: OpportunityDTO, stage: string, extra?: LostExtra) {
   551→    const snapshot = oppsRef.current;
   552→    const optimistic = snapshot.map((o) => (o.id === opp.id ? { ...o, stage } : o));
   553→    oppsRef.current = optimistic;
   554→    setOpps(optimistic);
   555→    setSavingStage(true);
   556→    try {
   557→      const res = await api.updateOpportunity(opp.id, {
   558→        stage,
   559→        ...(extra ?? {}),
   560→        actorName: user?.name ?? "System",
   561→        actorRole: user?.role ?? "system",
   562→      });
   563→      if (stage === "won") {
   564→        toast.success(
   565→          res.createdProject
   566→            ? `Deal Won! Project ${res.createdProject.code} otomatis dibuat beserta invoice DP`
   567→            : "Deal ditandai Won"
   568→        );
   569→      } else {
   570→        toast.success(`Stage "${opp.title}" diubah ke ${stageLabel(stage)}`);
   571→      }
   572→      await load({ silent: true });
   573→    } catch (e) {
   574→      oppsRef.current = snapshot;
   575→      setOpps(snapshot);
   576→      toast.error(e instanceof Error ? e.message : "Gagal memperbarui stage");
   577→    } finally {
   578→      setSavingStage(false);
   579→    }
   580→  }
   581→
   582→  function handleDragStart() {
   583→    draggedRecentlyRef.current = true;
   584→  }
   585→
   586→  function handleDragEnd(event: DragEndEvent) {
   587→    const { active, over } = event;
   588→    window.setTimeout(() => {
   589→      draggedRecentlyRef.current = false;
   590→    }, 250);
   591→    if (!over) return;
   592→    const opp = oppsRef.current.find((o) => o.id === active.id);
   593→    if (!opp) return;
   594→    const targetStage = String(over.id);
   595→    if (!KANBAN_KEYS.includes(targetStage)) return;
   596→    if (targetStage === opp.stage) return;
   597→    if (targetStage === "lost") {
   598→      setPendingLost(opp);
   599→      return;
   600→    }
   601→    void moveOpportunity(opp, targetStage);
   602→  }
   603→
   604→  const isEmpty = !loading && filtered.length === 0;
   605→
   606→  return (
   607→    <div className="space-y-4">
   608→      {/* Toolbar */}
   609→      <div className="flex flex-col gap-1">
   610→        <h1 className="text-xl font-semibold text-zinc-900">Sales Pipeline</h1>
   611→        <p className="text-sm text-zinc-500">
   612→          Kelola opportunity lintas brand: kanban drag-and-drop, tabel ringkas, dan detail lengkap.
   613→        </p>
   614→      </div>
   615→
   616→      <div className="flex flex-wrap items-center gap-2">
   617→        <div className="relative w-full sm:w-64">
   618→          <Search className="pointer-events-none absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-zinc-400" aria-hidden="true" />
   619→          <Input
   620→            value={qInput}
   621→            onChange={(e) => setQInput(e.target.value)}
   622→            placeholder="Cari lead, perusahaan..."
   623→            aria-label="Cari opportunity"
   624→            className="w-full bg-white pl-8"
   625→          />
   626→        </div>
   627→
   628→        <Select value={activeBrandFilter} onValueChange={setActiveBrandFilter}>
   629→          <SelectTrigger className="w-full bg-white sm:w-44" aria-label="Filter brand">
   630→            <SelectValue placeholder="Semua Brand" />
   631→          </SelectTrigger>
   632→          <SelectContent>
   633→            <SelectItem value="all">Semua Brand</SelectItem>
   634→            {brands.map((b) => (
   635→              <SelectItem key={b.id} value={b.id}>
   636→                <span className="flex items-center gap-2">
   637→                  <span className="size-2 rounded-full" style={{ backgroundColor: b.color }} />
   638→                  {b.name}
   639→                </span>
   640→              </SelectItem>
   641→            ))}
   642→          </SelectContent>
   643→        </Select>
   644→
   645→        <Select value={owner} onValueChange={setOwner}>
   646→          <SelectTrigger className="w-full bg-white sm:w-44" aria-label="Filter owner">
   647→            <SelectValue placeholder="Semua Owner" />
   648→          </SelectTrigger>
   649→          <SelectContent>
   650→            <SelectItem value="all">Semua Owner</SelectItem>
   651→            <SelectItem value={UNASSIGNED}>Belum di-assign</SelectItem>
   652→            {ownerOptions.map((name) => (
   653→              <SelectItem key={name} value={name}>
   654→                {name}
   655→              </SelectItem>
   656→            ))}
   657→          </SelectContent>
   658→        </Select>
   659→
   660→        {view === "table" ? (
   661→          <Select value={tempFilter} onValueChange={setTempFilter}>
   662→            <SelectTrigger className="w-full bg-white sm:w-40" aria-label="Filter temperatur">
   663→              <SelectValue placeholder="Semua Temperatur" />
   664→            </SelectTrigger>
   665→            <SelectContent>
   666→              <SelectItem value="all">Semua Temperatur</SelectItem>
   667→              <SelectItem value="hot">Hot</SelectItem>
   668→              <SelectItem value="warm">Warm</SelectItem>
   669→              <SelectItem value="cold">Cold</SelectItem>
   670→            </SelectContent>
   671→          </Select>
   672→        ) : null}
   673→
   674→        <div className="ml-auto flex items-center gap-2">
   675→          <Badge variant="secondary" className="bg-zinc-200/60 text-zinc-700">
   676→            {filtered.length} opportunity
   677→          </Badge>
   678→          <ToggleGroup
   679→            type="single"
   680→            variant="outline"
   681→            value={view}
   682→            onValueChange={(v) => {
   683→              if (v === "kanban" || v === "table") setView(v);
   684→            }}
   685→          >
   686→            <ToggleGroupItem
   687→              value="kanban"
   688→              aria-label="Tampilan Kanban"
   689→              className="data-[state=on]:bg-zinc-900 data-[state=on]:text-white"
   690→            >
   691→              <KanbanSquare className="size-4" aria-hidden="true" />
   692→              <span className="hidden sm:inline">Kanban</span>
   693→            </ToggleGroupItem>
   694→            <ToggleGroupItem
   695→              value="table"
   696→              aria-label="Tampilan Tabel"
   697→              className="data-[state=on]:bg-zinc-900 data-[state=on]:text-white"
   698→            >
   699→              <Table2 className="size-4" aria-hidden="true" />
   700→              <span className="hidden sm:inline">Tabel</span>
   701→            </ToggleGroupItem>
   702→          </ToggleGroup>
   703→        </div>
   704→      </div>
   705→
   706→      {/* Konten */}
   707→      {loading ? (
   708→        <div className="flex gap-3 overflow-hidden">
   709→          {Array.from({ length: 5 }).map((_, i) => (
   710→            <Skeleton key={i} className="h-[420px] w-[280px] shrink-0 rounded-xl" />
   711→          ))}
   712→        </div>
   713→      ) : isEmpty ? (
   714→        <div className="flex flex-col items-center justify-center gap-2 rounded-xl border bg-white py-16 text-center shadow-sm">
   715→          <Inbox className="size-10 text-zinc-300" aria-hidden="true" />
   716→          <p className="font-medium text-zinc-700">Belum ada opportunity</p>
   717→          <p className="max-w-sm text-sm text-zinc-500">
   718→            Tidak ada opportunity yang cocok dengan filter saat ini. Coba ubah kata kunci, brand, atau owner.
   719→          </p>
   720→        </div>
   721→      ) : view === "kanban" ? (
   722→        <DndContext
   723→          sensors={sensors}
   724→          onDragStart={handleDragStart}
   725→          onDragEnd={handleDragEnd}
   726→          onDragCancel={() => {
   727→            window.setTimeout(() => {
   728→              draggedRecentlyRef.current = false;
   729→            }, 250);
   730→          }}
   731→        >
   732→          <div className="crm-scroll overflow-x-auto pb-2">
   733→            <div className="flex min-w-max gap-3">
   734→              {KANBAN_KEYS.map((key) => (
   735→                <StageColumn
   736→                  key={key}
   737→                  stageKey={key}
   738→                  opportunities={grouped[key] ?? []}
   739→                  onOpen={(id) => {
   740→                    if (draggedRecentlyRef.current) return;
   741→                    openDetail(id);
   742→                  }}
   743→                />
   744→              ))}
   745→            </div>
   746→          </div>
   747→        </DndContext>
   748→      ) : (
   749→        <OpportunityTable
   750→          rows={tableRows}
   751→          sortDir={sortDir}
   752→          onToggleSort={toggleSort}
   753→          onOpen={(id) => {
   754→            if (draggedRecentlyRef.current) return;
   755→            openDetail(id);
   756→          }}
   757→        />
   758→      )}
   759→
   760→      {/* Drawer detail */}
   761→      <OpportunityDetail
   762→        opportunityId={selectedId}
   763→        open={drawerOpen}
   764→        onOpenChange={setDrawerOpen}
   765→        onChanged={() => void load({ silent: true })}
   766→      />
   767→
   768→      {/* Dialog alasan lost (dari drag kanban) */}
   769→      <LostReasonDialog
   770→        open={!!pendingLost}
   771→        saving={savingStage}
   772→        onCancel={() => setPendingLost(null)}
   773→        onConfirm={(extra) => {
   774→          if (!pendingLost) return;
   775→          const opp = pendingLost;
   776→          setPendingLost(null);
   777→          void moveOpportunity(opp, "lost", extra);
   778→        }}
   779→      />
   780→    </div>
   781→  );
   782→}
