   265→function hashColor(name: string): string {
   266→  let h = 0;
   267→  for (let i = 0; i < name.length; i++) h = (h * 31 + name.charCodeAt(i)) >>> 0;
   268→  return AVATAR_PALETTE[h % AVATAR_PALETTE.length];
   269→}

   271→function useDebounced<T>(value: T, delay = 300): T {
   272→  const [debounced, setDebounced] = useState(value);
   273→  useEffect(() => {
   274→    const t = setTimeout(() => setDebounced(value), delay);
   275→    return () => clearTimeout(t);
   276→  }, [value, delay]);
   277→  return debounced;
   278→}

   288→function consentMeta(status: string): { label: string; className: string } {
   289→  if (status === "granted") return { label: "Consent: Disetujui", className: "bg-emerald-100 text-emerald-700" };
   290→  if (status === "pending") return { label: "Consent: Menunggu", className: "bg-amber-100 text-amber-700" };
   291→  if (status === "declined") return { label: "Consent: Ditolak", className: "bg-rose-100 text-rose-700" };
   292→  return { label: status, className: "bg-zinc-100 text-zinc-600" };
   293→}

   713→/** Tick status kirim pesan outbound: read/delivered emerald, sent zinc, failed rose. */
   714→function DeliveryTick({ status }: { status?: string | null }) {
   715→  if (!status) return null;
   716→  const s = status.toLowerCase();
   717→  const meta =
   718→    s === "read" || s === "delivered"
   719→      ? { Icon: CheckCheck, cls: "text-emerald-400", label: "Pesan dibaca" }
   720→      : s === "sent"
   721→        ? { Icon: Check, cls: "text-zinc-400", label: "Pesan terkirim" }
   722→        : s === "failed"
   723→          ? { Icon: AlertTriangle, cls: "text-rose-400", label: "Pesan gagal terkirim" }
   724→          : null;
   725→  if (!meta) return null;
   726→  const { Icon, cls, label } = meta;
   727→  return <Icon className={cn("size-3.5 shrink-0", cls)} role="img" aria-label={label} />;
   728→}

   804→function ConversationHistorySection({ contactId, contactName }: { contactId: string; contactName: string }) {
   805→  const [items, setItems] = useState<InteractionDTO[]>([]);
   806→  const [loading, setLoading] = useState(true);
   807→  const [error, setError] = useState<string | null>(null);
   808→  const [reloadKey, setReloadKey] = useState(0);
   809→
   810→  useEffect(() => {
   811→    // Komponen di-remount per kontak (key=contact.id) → loading awal cukup dari useState.
   812→    // Reset loading/error saat "Muat ulang" dilakukan lewat handler reload() (bukan di sini).
   813→    let alive = true; // guard race: respons fetch lama diabaikan setelah unmount/ganti kontak

   815→      .interactions({ contactId })
   816→      .then((res) => {
   817→        if (!alive) return;
   818→        setItems(res.interactions);
   819→        setLoading(false);
   820→      })

   822→        if (!alive) return;
   823→        setError(err instanceof Error ? err.message : "Gagal memuat riwayat percakapan");
   824→        setLoading(false);
   825→      });

   827→      alive = false;
   828→    };

   833→    setError(null);
   834→    setReloadKey((k) => k + 1);

   838→    () => Array.from(new Set(items.map((i) => i.channel).filter((ch): ch is string => Boolean(ch)))),

   847→        <Badge variant="secondary" className="border-transparent bg-zinc-100 text-[11px] font-normal text-zinc-600">
   848→          {loading ? "…" : `${items.length} pesan`}

   851→          const Icon = CHANNEL_ICONS[ch] ?? Globe;
   852→          return (
   853→            <Badge
   854→              key={ch}
   855→              variant="secondary"

   865→        })}
   866→        <Button
   867→          type="button"
   868→          size="sm"
   869→          variant="ghost"
   870→          className="ml-auto h-7 gap-1 px-2 text-xs text-zinc-500"
   871→          onClick={reload}
   872→          disabled={loading}
   873→          aria-label="Muat ulang riwayat percakapan"
   874→        >
   875→          <RefreshCw className={cn("size-3.5", loading && "animate-spin")} /> Muat ulang
   876→        </Button>
   877→      </div>
   878→
   879→      <div className="crm-scroll mt-2 max-h-96 overflow-y-auto rounded-xl border bg-white p-3 shadow-sm">

   893→              <RefreshCw className="size-3.5" /> Coba lagi

   898→        ) : (

  1795→/** Deteksi separator dari baris pertama: titik koma (hasil ekspor Excel locale Indonesia) atau koma. */
  1796→function detectCsvSeparator(firstLine: string): string {
  1797→  const semis = (firstLine.match(/;/g) ?? []).length;
  1798→  const commas = (firstLine.match(/,/g) ?? []).length;
  1799→  return semis > commas ? ";" : ",";

  1802→/** Parser CSV murni (tanpa dep): quoted field dengan escape "", \r\n/\n, separator koma atau titik koma. */
  1803→function parseCsv(text: string): string[][] {
  1804→  const raw = text.replace(/^\uFEFF/, "").trim();
  1805→  if (!raw) return [];
  1806→  const firstNewline = raw.search(/\r?\n/);
  1807→  const sep = detectCsvSeparator(firstNewline === -1 ? raw : raw.slice(0, firstNewline));
  1808→
  1809→  const rows: string[][] = [];
  1810→  let row: string[] = [];
  1811→  let field = "";
  1812→  let inQuotes = false;
  1813→
  1814→  for (let i = 0; i < raw.length; i++) {
  1815→    const ch = raw[i];
  1816→    if (inQuotes) {
  1817→      if (ch === '"') {
  1818→        if (raw[i + 1] === '"') {
  1819→          field += '"';
  1820→          i++;
  1821→        } else {
  1822→          inQuotes = false;
  1823→        }
  1824→      } else {
  1825→        field += ch;
  1826→      }
  1827→    } else if (ch === '"') {
  1828→      inQuotes = true;
  1829→    } else if (ch === sep) {
  1830→      row.push(field);
  1831→      field = "";
  1832→    } else if (ch === "\n" || ch === "\r") {
  1833→      if (ch === "\r" && raw[i + 1] === "\n") i++;
  1834→      row.push(field);
  1835→      field = "";
  1836→      rows.push(row);
  1837→      row = [];
  1838→    } else {
  1839→      field += ch;
  1840→    }
  1841→  }
  1842→  row.push(field);
  1843→  if (row.length > 1 || row[0] !== "") rows.push(row);
  1844→
  1845→  // Lewati baris yang seluruh selnya kosong.
  1846→  return rows.filter((r) => r.some((c) => c.trim() !== ""));
  1847→}
