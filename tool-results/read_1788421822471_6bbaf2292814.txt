    44→/** Baca file gambar kecil → data URL (dipakai logo & kop surat). */
    45→function readImageFile(file: File, maxChars = 1_500_000): Promise<string> {
    46→  return new Promise((resolve, reject) => {
    47→    if (!file.type.startsWith("image/")) {
    48→      reject(new Error("File harus berupa gambar (PNG/JPG/SVG/WebP)"));
    49→      return;
    50→    }
    51→    if (file.size > 1_100_000) {
    52→      reject(new Error("Ukuran gambar terlalu besar (maksimal ±1MB)"));
    53→      return;
    54→    }
    55→    const reader = new FileReader();
    56→    reader.onload = () => {
    57→      const result = String(reader.result ?? "");
    58→      if (result.length > maxChars) {
    59→        reject(new Error("Gambar terlalu besar setelah dibaca — pakai file lebih kecil"));
    60→        return;
    61→      }
    62→      resolve(result);
    63→    };
    64→    reader.onerror = () => reject(new Error("Gagal membaca file"));
    65→    reader.readAsDataURL(file);
    66→  });
    67→}

   865→                      </div>
   866→
   867→                      {svcOpen ? (
   868→                        <div className="mt-3 space-y-2.5 pl-6">
   869→                          {/* Workflow */}
   870→                          {svc.workflow.length > 0 ? (
   871→                            <ol className="space-y-1.5">
   872→                              {svc.workflow.map((w, idx) => (
   873→                                <li key={w.id} className="flex items-center gap-2 rounded-lg border bg-zinc-50/60 px-3 py-2">
   874→                                  <span className="flex h-5 w-5 shrink-0 items-center justify-center rounded-full bg-zinc-900 text-[10px] font-bold text-white" aria-hidden>{idx + 1}</span>
   875→                                  <div className="min-w-0 flex-1">
   876→                                    <p className="truncate text-xs font-medium text-zinc-800">
   877→                                      {w.name}
   878→                                      {w.isMilestone ? (
   879→                                        <Badge variant="outline" className="ml-1.5 border-amber-200 bg-amber-50 px-1 text-[10px] text-amber-700">Milestone</Badge>
   880→                                      ) : null}
   881→                                    </p>
   882→                                    <p className="truncate text-[11px] text-zinc-500">{w.phase}</p>
   883→                                  </div>
   884→                                  <Button type="button" variant="ghost" size="sm" className="h-7 px-2 text-[11px]"
   885→                                    disabled={busyKey === `stg:${svc.id}`}
   886→                                    onClick={() => void onToggleMilestone(w, svc.id)}>
   887→                                    {w.isMilestone ? "Lepas" : "Jadikan milestone"}
   888→                                  </Button>
   889→                                  <Button type="button" variant="ghost" size="icon" className="h-7 w-7 text-zinc-400 hover:text-rose-600"
   890→                                    aria-label={`Hapus langkah ${w.name}`} onClick={() => void onRemove("stage", w.id, `${w.phase} · ${w.name}`)}>
   891→                                    <Trash2 className="h-3.5 w-3.5" aria-hidden />
   892→                                  </Button>
   893→                                </li>
   894→                              ))}
   895→                            </ol>
   896→                          ) : (
   897→                            <p className="text-xs italic text-zinc-400">Belum ada workflow — tambahkan fase produksi di bawah.</p>
   898→                          )}
   899→
   900→                          {/* Form tambah langkah */}
   901→                          <div className="flex flex-wrap items-end gap-2 rounded-lg border border-dashed p-2.5">
   902→                            <div className="w-32 space-y-1">
   903→                              <Label className="text-[11px]">Fase</Label>
   904→                              <Input className="h-8 text-xs" value={form.phase} placeholder="mis. Production"
   905→                                onChange={(e) => setStageForms((m) => ({ ...m, [svc.id]: { ...form, phase: e.target.value } }))} />
   906→                            </div>
   907→                            <div className="min-w-[160px] flex-1 space-y-1">
   908→                              <Label className="text-[11px]">Langkah</Label>
   909→                              <Input className="h-8 text-xs" value={form.name} placeholder="mis. Creative Concept & Story Board"
   910→                                onChange={(e) => setStageForms((m) => ({ ...m, [svc.id]: { ...form, name: e.target.value } }))} />
   911→                            </div>
   912→                            <label className="flex items-center gap-1.5 pb-2 text-xs text-zinc-600">
   913→                              <input type="checkbox" className="accent-zinc-900" checked={form.milestone}
   914→                                onChange={(e) => setStageForms((m) => ({ ...m, [svc.id]: { ...form, milestone: e.target.checked } }))} />
   915→                              Milestone
   916→                            </label>
   917→                            <Button type="button" size="sm" variant="outline" className="h-8"
   918→                              disabled={busyKey === `stg:${svc.id}` || !form.name.trim()}
   919→                              onClick={() => void onAddStage(svc.id, form.phase, form.name, form.milestone)
   920→                                .then(() => setStageForms((m) => ({ ...m, [svc.id]: { phase: "", name: "", milestone: false } })))}>
   921→                              <Plus className="h-3.5 w-3.5" aria-hidden /> Langkah
   922→                            </Button>
   923→                          </div>
   924→                        </div>
   925→                      ) : null}
   926→                    </div>
   927→                  );
   928→                })}
   929→
   930→                {/* Form tambah layanan */}
   931→                <ServiceAddForm
   932→                  busy={busyKey === `svc:${cat.id}`}
   933→                  form={catForms[cat.id] ?? { name: "", unit: "", price: "" }}
   934→                  onChange={(form) => setCatForms((m) => ({ ...m, [cat.id]: form }))}
   935→                  onSubmit={(f) =>
   936→                    onAddService(cat.id, f.name, f.unit, f.price)
   937→                      .then((res) => {
   938→                        if (res) setCatForms((m) => ({ ...m, [cat.id]: { name: "", unit: "", price: "" } }));
   939→                        return res;
   940→                      })
   941→                  }
   942→                />
   943→              </div>
   944→            ) : null}
   945→          </div>
   946→        );
   947→      })}
   948→
   949→      {/* Tanpa kategori */}
   950→      {uncategorized.length > 0 ? (
   951→        <div className="rounded-xl border border-dashed p-3">
   952→          <p className="mb-2 text-xs font-semibold uppercase tracking-wider text-zinc-500">Tanpa kategori</p>
   953→          <div className="flex flex-wrap gap-1.5">
   954→            {uncategorized.map((s) => (
   955→              <span key={s.id} className="group inline-flex items-center gap-1 rounded-full border border-zinc-200 bg-zinc-50 px-2.5 py-1 text-xs text-zinc-600">
   956→                {s.name}
   957→                <button type="button" className="text-zinc-300 hover:text-rose-600" aria-label={`Hapus ${s.name}`} onClick={() => void onRemove("service", s.id, s.name)}>
   958→                  <X className="h-3 w-3" aria-hidden />
   959→                </button>
   960→              </span>
   961→            ))}
   962→          </div>
   963→        </div>
   964→      ) : null}
   965→
   966→      {/* Form kategori baru */}
   967→      <div className="flex flex-wrap items-end gap-2 rounded-xl border border-dashed p-3.5">
   968→        <div className="min-w-[180px] flex-1 space-y-1">
   969→          <Label className="text-xs">Kategori layanan baru</Label>
   970→          <Input className="h-9" value={newCatName} placeholder="mis. Animasi 3D"
   971→            onChange={(e) => setNewCatName(e.target.value)} />
   972→        </div>
   973→        <Button type="button" size="sm" className="h-9" disabled={busyKey === "cat:new" || !newCatName.trim()}
   974→          onClick={() => void onAddCategory(newCatName).then((res) => { if (res) setNewCatName(""); return res; })}>
   975→          <Plus className="h-3.5 w-3.5" aria-hidden /> Tambah kategori
   976→        </Button>
   977→      </div>
   978→    </div>
   979→  );
   980→}
