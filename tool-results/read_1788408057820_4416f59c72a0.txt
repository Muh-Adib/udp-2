    76→function slugify(name: string): string {
    77→  return name
    78→    .toLowerCase()
    79→    .replace(/[^a-z0-9]+/g, "_")

    83→function quotePrefixFrom(prefix: string): string {
    84→  const clean = prefix.trim().replace(/[^a-z0-9]/gi, "").slice(0, 3).toUpperCase();
    85→  return clean ? `Q${clean}` : "";

   451→  async function submitTemplate(e: React.FormEvent<HTMLFormElement>) {
   452→    e.preventDefault();
   453→    if (!templateDraft.name.trim()) { toast.error("Nama template wajib diisi"); return; }
   454→    if (!templateDraft.body.trim()) { toast.error("Isi template wajib diisi"); return; }
   455→    const delay = Number(templateDraft.delayDays);
   456→    if (!Number.isInteger(delay) || delay < 0 || delay > 30) { toast.error("Jeda hari harus antara 0 dan 30"); return; }
   457→    if (!user) { toast.error("Sesi berakhir — silakan login ulang"); return; }
   458→    setTemplateSubmitting(true);
   459→    try {
   460→      const payload = {
   461→        name: templateDraft.name.trim(),
   462→        channel: templateDraft.channel,
   463→        brandId: templateDraft.brandId === "all" ? null : templateDraft.brandId,
   464→        stage: templateDraft.stage.trim() || null,
   465→        delayDays: delay,
   466→        body: templateDraft.body.trim(),
   467→        approved: canApproveTemplates ? templateDraft.approved : (editingTemplate?.approved ?? false),
   468→        actorName: user.name,
   469→        actorRole: user.role,
   470→      };
   471→      if (editingTemplate) {
   472→        await api.updateTemplate(editingTemplate.id, payload);
   473→        toast.success("Template diperbarui");
   474→      } else {
   475→        await api.createTemplate(payload);
   476→        toast.success("Template dibuat");
   477→      }
   478→      setTemplateDraft(EMPTY_TEMPLATE_DRAFT);
   479→      setEditingTemplate(null);
   480→      setTemplateOpen(false);
   481→      await loadTemplates();
   482→    } catch (err) {
   483→      toast.error(err instanceof Error ? err.message : "Gagal menyimpan template");
   484→    } finally {
   485→      setTemplateSubmitting(false);
   486→    }
   487→  }
