    55→function channelMeta(channel: string) {
    56→  return CHANNEL_STYLE[channel] ?? { icon: Inbox, circle: "bg-zinc-100 text-zinc-600", badge: "border-zinc-200 bg-zinc-50 text-zinc-600" };
    57→}
    58→function channelLabel(channel: string): string {
    59→  return CHANNELS.find((c) => c.key === channel)?.label ?? channel;
    60→}
    61→function slaBadge(slaHours: number): { label: string; className: string } {
    62→  if (slaHours <= 4) return { label: "SLA OK", className: "border-emerald-200 bg-emerald-50 text-emerald-700" };
    63→  if (slaHours <= 24) return { label: "Perlu respons", className: "border-amber-200 bg-amber-50 text-amber-700" };
    64→  return { label: `Terlambat ${slaHours} jam`, className: "border-rose-200 bg-rose-50 text-rose-700" };
    65→}
    66→function scoreTone(score: number): { bar: string; text: string } {
    67→  if (score >= 75) return { bar: "bg-emerald-500", text: "text-emerald-600" };
    68→  if (score >= 50) return { bar: "bg-amber-500", text: "text-amber-600" };
    69→  return { bar: "bg-zinc-400", text: "text-zinc-500" };
    70→}

   125→function LeadCard({ lead, selected, onSelect }: { lead: InboxLead; selected: boolean; onSelect: (lead: InboxLead) => void }) {
   126→  const sender = (lead.senderName ?? "").trim() || "Tanpa nama";
   127→  const sla = slaBadge(lead.slaHours);
   128→  const isLate = lead.slaHours > 24;
   129→  const meta = channelMeta(lead.channel);
   130→  return (
   131→    <button
   132→      type="button"
   133→      onClick={() => onSelect(lead)}
   134→      aria-label={`Buka detail lead dari ${sender}`}
   135→      className={cn(
   136→        "w-full rounded-xl border bg-white p-4 text-left shadow-sm transition-colors hover:bg-zinc-50",
   137→        isLate ? "border-l-4 border-l-rose-500" : "hover:border-zinc-300",
   138→        selected && !isLate && "border-zinc-900 ring-1 ring-zinc-900",
   139→        selected && isLate && "ring-1 ring-rose-500"
   140→      )}
   141→    >
   142→      <div className="flex items-start gap-3">
   143→        <ChannelAvatar channel={lead.channel} />
   144→        <div className="min-w-0 flex-1">
   145→          <div className="flex flex-wrap items-center gap-x-2 gap-y-0.5">
   146→            <span className="truncate text-sm font-semibold text-zinc-900">{sender}</span>
   147→            <span className="text-xs text-zinc-300" aria-hidden="true">·</span>
   148→            <span className="text-xs text-zinc-500">{formatDateTime(lead.createdAt)}</span>
   149→            <span className="text-xs text-zinc-300" aria-hidden="true">·</span>
   150→            <span className="text-xs text-zinc-400">{timeAgo(lead.createdAt)}</span>
   151→          </div>
   152→          <p className="mt-1 line-clamp-2 text-sm leading-relaxed text-zinc-600">{lead.content}</p>
   153→          <div className="mt-2 flex flex-wrap items-center gap-1.5">
   154→            <Badge variant="outline" className={cn("border", meta.badge)}>{channelLabel(lead.channel)}</Badge>
   155→            {lead.brand ? <BrandChip name={lead.brand.name} color={lead.brand.color} /> : null}
   156→            <Badge variant="outline" className={cn("border", sla.className)}>{sla.label}</Badge>
   157→            {lead.candidates.length > 0 ? (
   158→              <Badge variant="outline" className="border-amber-300 bg-amber-50 text-amber-700">
   159→                <AlertTriangle aria-hidden="true" />
   160→                {`Duplikat? ${lead.candidates.length} kandidat`}
   161→              </Badge>
   162→            ) : null}
   163→          </div>
   164→        </div>
   165→      </div>
   166→    </button>
   167→  );
   168→}

   170→function CandidateCard({ candidate, selected, onToggle }: { candidate: MatchCandidateDTO; selected: boolean; onToggle: () => void }) {
   171→  const tone = scoreTone(candidate.score);
   172→  const contact = candidate.contact;
   173→  const label = contact?.fullName?.trim() || `Contact #${candidate.contactId.slice(0, 8)}`;
   174→  return (
   175→    <div className={cn("rounded-lg border p-3 transition-colors", selected ? "border-emerald-300 bg-emerald-50/60" : "border-zinc-200 bg-white")}>
   176→      <div className="flex items-center gap-2.5">
   177→        <span className="flex size-8 shrink-0 items-center justify-center rounded-full bg-zinc-100 text-[11px] font-bold text-zinc-600" aria-hidden="true">
   178→          {initials(contact?.fullName ?? label)}
   179→        </span>
   180→        <span className="min-w-0 flex-1 truncate text-sm font-semibold text-zinc-900">{label}</span>
   181→        <span className={cn("text-xs font-bold", tone.text)}>{candidate.score}%</span>
   182→      </div>
   183→      <div className="mt-2 h-1.5 w-full overflow-hidden rounded-full bg-zinc-100" role="progressbar" aria-valuenow={candidate.score} aria-valuemin={0} aria-valuemax={100} aria-label={`Skor kemiripan ${candidate.score} persen`}>
   184→        <div className={cn("h-full rounded-full transition-all", tone.bar)} style={{ width: `${Math.min(100, Math.max(0, candidate.score))}%` }} />
   185→      </div>
   186→      {candidate.reasons.length > 0 ? (
   187→        <ul className="mt-2 space-y-0.5">
   188→          {candidate.reasons.map((reason) => (
   189→            <li key={reason} className="text-xs text-zinc-500">• {reason}</li>
   190→          ))}
   191→        </ul>
   192→      ) : null}
   193→      <div className="mt-2 space-y-1 text-xs text-zinc-600">
   194→        {contact?.position ? (
   195→          <p className="flex items-center gap-1.5"><User className="size-3 shrink-0 text-zinc-400" aria-hidden="true" />{contact.position}</p>
   196→        ) : null}
   197→        {contact?.company?.name ? (
   198→          <p className="flex items-center gap-1.5"><Building2 className="size-3 shrink-0 text-zinc-400" aria-hidden="true" />{contact.company.name}</p>
   199→        ) : null}
   200→        {contact?.email ? (
   201→          <p className="flex items-center gap-1.5"><Mail className="size-3 shrink-0 text-zinc-400" aria-hidden="true" /><span className="truncate">{contact.email}</span></p>
   202→        ) : null}
   203→        {contact?.whatsapp ? (
   204→          <p className="flex items-center gap-1.5"><MessageCircle className="size-3 shrink-0 text-zinc-400" aria-hidden="true" />{contact.whatsapp}</p>
   205→        ) : null}
   206→      </div>
   207→      <Button
   208→        type="button"
   209→        size="sm"
   210→        variant={selected ? "default" : "outline"}
   211→        className="mt-3 w-full"
   212→        onClick={onToggle}
   213→      >
   214→        {selected ? (
   215→          <><Check className="size-3.5" aria-hidden="true" /> Dipilih — akan digabung ke contact ini</>
   216→        ) : (
   217→          "Gabungkan ke Contact Ini"
   218→        )}
   219→      </Button>
   220→    </div>
   221→  );
   222→}

   245→export default function InboxModule() {
   246→  const user = useCrmStore((s) => s.user);
   247→  const brands = useCrmStore((s) => s.brands);
   248→  const activeBrandFilter = useCrmStore((s) => s.activeBrandFilter);
   249→  const setActiveBrandFilter = useCrmStore((s) => s.setActiveBrandFilter);
   250→
   251→  const [leads, setLeads] = useState<InboxLead[] | null>(null);
   252→  const [loading, setLoading] = useState(true);
   253→  const [error, setError] = useState<string | null>(null);
   254→  const [channelFilter, setChannelFilter] = useState<string>("all");
   255→  const [selectedId, setSelectedId] = useState<string | null>(null);
   256→  const [converting, setConverting] = useState(false);
   257→  const [showNewContact, setShowNewContact] = useState(false);
   258→  const [linkedContactId, setLinkedContactId] = useState<string | null>(null);
   259→  const [contactForm, setContactForm] = useState<ContactFormState>(EMPTY_CONTACT_FORM);
   260→  const [oppForm, setOppForm] = useState<OpportunityFormState>(EMPTY_OPP_FORM);
   261→  const detailRef = useRef<HTMLDivElement | null>(null);
   262→
   263→  const loadLeads = useCallback(async () => {
   264→    setLoading(true);
   265→    try {
   266→      const res = await api.inbox({ channel: channelFilter, brandId: activeBrandFilter });
   267→      setLeads(res.leads);
   268→      setError(null);
   269→    } catch (err) {
   270→      setError(err instanceof Error ? err.message : "Terjadi kesalahan tak terduga");
   271→      toast.error("Gagal memuat lead inbox");
   272→    } finally {
   273→      setLoading(false);
   274→    }
   275→  }, [channelFilter, activeBrandFilter]);
   276→
   277→  useEffect(() => {
   278→    void loadLeads();
   279→  }, [loadLeads]);
   280→
   281→  const stats = useMemo(() => {
   282→    const list = leads ?? [];
   283→    return {
   284→      total: list.length,
   285→      late: list.filter((l) => l.slaHours > 24).length,
   286→      duplicate: list.filter((l) => l.candidates.length > 0).length,
   287→    };
   288→  }, [leads]);
   289→
   290→  const selectedLead = useMemo(
   291→    () => leads?.find((l) => l.id === selectedId) ?? null,
   292→    [leads, selectedId]
   293→  );
   294→
   295→  const brandOptions = brands.length > 0 ? brands : selectedLead?.brand ? [selectedLead.brand] : [];
   296→  const selectedOppBrand = brandOptions.find((b) => b.id === oppForm.brandId);
   297→  const serviceNameOptions = selectedOppBrand ? BRAND_SERVICES[selectedOppBrand.slug] ?? [] : [];
   298→
   299→  const convertMode: "new" | "link" | null = showNewContact ? "new" : linkedContactId ? "link" : null;
   300→  const canConvert =
   301→    convertMode !== null &&
   302→    oppForm.brandId !== "" &&
   303→    (convertMode !== "new" || contactForm.firstName.trim().length > 0);
   304→
   305→  function handleSelectLead(lead: InboxLead) {
   306→    setSelectedId(lead.id);
   307→    setShowNewContact(false);
   308→    setLinkedContactId(null);
   309→
   310→    const rawSender = (lead.senderName ?? "").trim();
   311→    const looksLikePhone = /^\+?\d/.test(rawSender);
   312→    const looksLikeEmail = rawSender.includes("@");
   313→    setContactForm({
   314→      ...EMPTY_CONTACT_FORM,
   315→      email: looksLikeEmail ? rawSender : "",
   316→      whatsapp: looksLikePhone ? rawSender : "",
   317→    });
   318→    setOppForm({
   319→      ...EMPTY_OPP_FORM,
   320→      brandId: lead.brandId ?? lead.brand?.id ?? "",
   321→      title: `Lead ${channelLabel(lead.channel)} — ${rawSender.slice(0, 48) || "Tanpa nama"}`,
   322→      ownerName: user?.name ?? "",
   323→    });
   324→    if (typeof window !== "undefined" && window.innerWidth < 1024) {
   325→      window.setTimeout(() => {
   326→        detailRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
   327→      }, 80);
   328→    }
   329→  }
   330→
   331→  function handleToggleCandidate(contactId: string) {
   332→    setShowNewContact(false);
   333→    setLinkedContactId((prev) => (prev === contactId ? null : contactId));
   334→  }
   335→
   336→  function handleToggleNewContact() {
   337→    setLinkedContactId(null);
   338→    setShowNewContact((prev) => !prev);
   339→  }
   340→
   341→  async function handleConvert() {
   342→    if (!selectedLead || !user || !convertMode) return;
   343→    if (!oppForm.brandId) {
   344→      toast.error("Brand wajib dipilih untuk opportunity.");
   345→      return;
   346→    }
   347→    if (convertMode === "new" && !contactForm.firstName.trim()) {
   348→      toast.error("Nama depan contact wajib diisi.");
   349→      return;
   350→    }
   351→
   352→    const opportunity: Record<string, unknown> = {
   353→      brandId: oppForm.brandId,
   354→      title:
   355→        oppForm.title.trim() ||
   356→        `Lead ${channelLabel(selectedLead.channel)} — ${(selectedLead.senderName ?? "Tanpa nama").trim().slice(0, 48)}`,
   357→      ownerName: oppForm.ownerName.trim() || user.name,
   358→      priority: oppForm.priority,
   359→    };
   360→    if (oppForm.serviceCategory) opportunity.serviceCategory = oppForm.serviceCategory;
   361→    if (oppForm.serviceName) opportunity.serviceName = oppForm.serviceName;
   362→    const est = Number(oppForm.estimatedValue);
   363→    if (oppForm.estimatedValue.trim() !== "" && Number.isFinite(est) && est > 0) {
   364→      opportunity.estimatedValue = est;
   365→    }
   366→
   367→    const payload: Record<string, unknown> = {
   368→      interactionId: selectedLead.id,
   369→      action: convertMode,
   370→      opportunity,
   371→      actorName: user.name,
   372→      actorRole: user.role,
   373→    };
   374→    if (convertMode === "link" && linkedContactId) payload.contactId = linkedContactId;
   375→    if (convertMode === "new") {
   376→      payload.contact = {
   377→        firstName: contactForm.firstName.trim(),
   378→        ...(contactForm.lastName.trim() ? { lastName: contactForm.lastName.trim() } : {}),
   379→        ...(contactForm.email.trim() ? { email: contactForm.email.trim() } : {}),
   380→        ...(contactForm.whatsapp.trim() ? { whatsapp: contactForm.whatsapp.trim() } : {}),
   381→        ...(contactForm.companyName.trim() ? { companyName: contactForm.companyName.trim() } : {}),
   382→        ...(contactForm.city.trim() ? { city: contactForm.city.trim() } : {}),
   383→      };
   384→    }
   385→
   386→    setConverting(true);
   387→    try {
   388→      const res = await api.convertLead(payload);
   389→      toast.success("Lead dikonversi", {
   390→        description: `Opportunity ${res.opportunity?.title ?? ""} berhasil dibuat beserta task follow-up otomatis.`,
   391→      });
   392→      setSelectedId(null);
   393→      setShowNewContact(false);
   394→      setLinkedContactId(null);
   395→      await loadLeads();
   396→    } catch (err) {
   397→      toast.error(err instanceof Error ? err.message : "Gagal mengonversi lead.");
   398→    } finally {
   399→      setConverting(false);
   400→    }
   401→  }
   402→
   403→  const firstLoad = loading && leads === null;
   404→
   405→  return (
   406→    <div className="space-y-4">
   407→      {/* ===== Header ===== */}
   408→      <div className="rounded-xl border bg-white p-4 shadow-sm sm:p-6">
   409→        <div className="flex flex-col gap-4 lg:flex-row lg:items-start lg:justify-between">
   410→          <div className="min-w-0">
   411→            <div className="flex flex-wrap items-center gap-2">
   412→              <h1 className="text-lg font-bold text-zinc-900 sm:text-xl">Lead Inbox</h1>
   413→              {leads ? (
   414→                <Badge className="bg-zinc-900 text-white">{`${stats.total} lead belum dikonversi`}</Badge>
   415→              ) : (
   416→                <Skeleton className="h-5 w-40" />
   417→              )}
   418→            </div>
   419→            <p className="mt-1 text-sm text-zinc-500">Semua lead baru lintas kanal — Instagram, WhatsApp, Email, Website</p>
   420→            <div className="mt-2 flex flex-wrap items-center gap-x-4 gap-y-1 text-xs text-zinc-500" aria-label="Legenda SLA">
   421→              <span className="inline-flex items-center gap-1.5"><span className="size-2 rounded-full bg-emerald-500" aria-hidden="true" />SLA OK ≤ 4 jam</span>
   422→              <span className="inline-flex items-center gap-1.5"><span className="size-2 rounded-full bg-amber-500" aria-hidden="true" />Perlu respons ≤ 24 jam</span>
   423→              <span className="inline-flex items-center gap-1.5"><span className="size-2 rounded-full bg-rose-500" aria-hidden="true" />Terlambat &gt; 24 jam</span>
   424→            </div>
   425→          </div>
   426→          <div className="flex flex-col gap-2 sm:flex-row sm:items-center">
   427→            <Select
   428→              value={channelFilter}
   429→              onValueChange={(v) => { setChannelFilter(v); setSelectedId(null); }}
   430→            >
   431→              <SelectTrigger className="w-full sm:w-[170px]" aria-label="Filter kanal">
   432→                <SelectValue placeholder="Semua Kanal" />
   433→              </SelectTrigger>
   434→              <SelectContent>
   435→                <SelectItem value="all">Semua Kanal</SelectItem>
   436→                {CHANNELS.map((c) => (
   437→                  <SelectItem key={c.key} value={c.key}>{c.label}</SelectItem>
   438→                ))}
   439→              </SelectContent>
