    55→function formatBytes(n: number | null | undefined): string {
    56→  if (!n || n <= 0) return "";
    57→  if (n < 1024) return `${n} B`;
    58→  if (n < 1024 * 1024) return `${Math.round(n / 1024)} KB`;
    59→  return `${(n / (1024 * 1024)).toFixed(1)} MB`;
    60→}

    62→function attachmentIconOf(name: string): LucideIcon {
    63→  const ext = (name.split(".").pop() ?? "").toLowerCase();
    64→  if (["png", "jpg", "jpeg", "gif", "webp", "svg", "heic"].includes(ext)) return FileImage;
    65→  if (["pdf", "doc", "docx", "txt", "csv", "ppt", "pptx", "xls", "xlsx"].includes(ext)) return FileText;
    66→  if (["zip", "rar", "7z", "tar", "gz"].includes(ext)) return FileArchive;
    67→  return File;
    68→}

    71→function readFileAsDataUrl(file: File): Promise<string> {
    72→  return new Promise((resolve, reject) => {
    73→    const reader = new FileReader();
    74→    reader.onload = () => resolve(String(reader.result ?? ""));
    75→    reader.onerror = () => reject(new Error("Gagal membaca file"));
    76→    reader.readAsDataURL(file);
    77→  });
    78→}

    80→/** Cocokkan 9 digit terakhir nomor (awalan +62/0 sering beda antar sumber). */
    81→function digitsMatch(a: string, b: string): boolean {
    82→  const da = a.replace(/\D+/g, "");
    83→  const db = b.replace(/\D+/g, "");
    84→  return da.length >= 9 && db.length >= 9 && da.slice(-9) === db.slice(-9);
    85→}

    87→/**
    88→ * Ronde 34-b — pencocokan identitas best-effort lead↔kontak (thread belum ter-link contactId):
    89→ * email di sender/recipient, 9 digit nomor, handle IG, atau nama persis.
    90→ */
    91→function leadMatchesContactIdentity(lead: InboxLead, c: ContactRef): boolean {
    92→  for (const raw of [(lead.senderName ?? "").trim(), (lead.recipientName ?? "").trim()]) {
    93→    if (!raw) continue;
    94→    const lower = raw.toLowerCase();
    95→    if (c.email && lower.includes(c.email.toLowerCase())) return true;
    96→    if (c.whatsapp && digitsMatch(lower, c.whatsapp)) return true;
    97→    if (c.phone && digitsMatch(lower, c.phone)) return true;
    98→    if (c.instagram && lower === c.instagram.toLowerCase()) return true;
    99→    if (c.fullName && lower === c.fullName.toLowerCase()) return true;
   100→  }
   101→  return false;
   102→}

   160→/**
   161→ * Ronde 25 — alamat tujuan satu kanal balasan dari data kontak/pengirim lead
   162→ * (sumber sama dgn backend reachableAddress: kontak dulu, lalu fallback dari senderName).
   163→ * socialProfile dibaca defensif — ada di model DB tapi belum ada di ContactRef DTO.
   164→ */
   165→function replyAddressFor(channel: string, lead: InboxLead): string | null {
   166→  const sender = (lead.senderName ?? "").trim();
   167→  const c = (lead.contact ?? null) as {
   168→    email?: string | null;
   169→    whatsapp?: string | null;
   170→    phone?: string | null;
   171→    instagram?: string | null;
   172→    socialProfile?: string | null;
   173→  } | null;
   174→  switch (channel) {
   175→    case "email":
   176→      return c?.email ?? extractEmailFromText(sender);
   177→    case "whatsapp":
   178→      return c?.whatsapp ?? (looksLikePhoneText(sender) ? sender : null);
   179→    case "instagram":
   180→      return c?.instagram
   181→        ?? (c?.socialProfile && isSocialHandle(c.socialProfile) ? c.socialProfile : null)
   182→        ?? (isSocialHandle(sender) || sender.startsWith("@") ? sender : null);
   183→    case "phone":
   184→      return c?.phone ?? c?.whatsapp ?? (looksLikePhoneText(sender) ? sender : null);
   185→    default:
   186→      return null;
   187→  }
   188→}

   215→  const label =

   224→    status === "read" || status === "delivered" ? CheckCheck

   229→    status === "read" ? "text-emerald-600"
   230→    : status === "failed" ? "text-red-600"
   231→    : status === "sent" || status === "delivered" ? "text-zinc-400"

   233→  return (

   241→/** Fase 3 — SLA countdown per brand: sisa = SLA brand − jam tunggu (slaHours dari API). */
   242→function slaBadgeInfo(brandSlaHours: number, waitHours: number): {
   243→  label: string; className: string; tone: SlaTone;
   244→} {
   245→  const split = (hours: number) => {
   246→    let h = Math.floor(hours);
   247→    let m = Math.round((hours - h) * 60);
   248→    if (m >= 60) { h += 1; m = 0; }
   249→    return { h, m };
   250→  };
   251→  const fmt = (hours: number, prefix: string) => {
   252→    const { h, m } = split(hours);
   253→    if (h < 1) return `${prefix}${m}m`;
   254→    if (m === 0) return `${prefix}${h}j`;
   255→    return `${prefix}${h}j ${m}m`;
   256→  };
   257→
   258→  const remaining = brandSlaHours - waitHours;
   259→  if (remaining <= 0) {
   260→    return {
   261→      label: fmt(Math.abs(remaining), "Terlambat "),
   262→      className: "border-rose-200 bg-rose-50 text-rose-700",
   263→      tone: "breach" as const,
   264→    };
   265→  }
   266→  if (remaining <= brandSlaHours * 0.5) {
   267→    return {
   268→      label: fmt(remaining, "Segera: "),
   269→      className: "border-amber-200 bg-amber-50 text-amber-700",
   270→      tone: "warning" as const,
   271→    };
   272→  }
   273→  return {
   274→    label: fmt(remaining, "Sisa "),
   275→    className: "border-emerald-200 bg-emerald-50 text-emerald-700",
   276→    tone: "ok" as const,
   277→  };
   278→}

   483→/** Kanal paling sering muncul dalam thread — warna avatar kartu thread. */
   484→function dominantChannelOf(thread: ConversationThreadDTO, fallback: string): string {
   485→  const counts = new Map<string, number>();
   486→  for (const m of thread.messages) counts.set(m.channel, (counts.get(m.channel) ?? 0) + 1);
   487→  let best = fallback;
   488→  let bestN = -1;
   489→  for (const [ch, n] of counts) {
   490→    if (n > bestN) { best = ch; bestN = n; }
   491→  }
   492→  return best;
   493→}

   777→/** Salin teks + fallback legacy; false = clipboard ditolak browser (butuh salin manual). */
   778→async function copyTextSmart(text: string, successMsg: string): Promise<boolean> {
   779→  const legacyCopy = () => {
   780→    try {
   781→      const ta = document.createElement("textarea");
   782→      ta.value = text;
   783→      ta.setAttribute("readonly", "");
   784→      ta.style.position = "fixed";
   785→      ta.style.opacity = "0";
   786→      document.body.appendChild(ta);
   787→      ta.select();
   788→      const ok = document.execCommand("copy");
   789→      document.body.removeChild(ta);
   790→      return ok;
   791→    } catch {
   792→      return false;
   793→    }
   794→  };
   795→  try {
   796→    await navigator.clipboard.writeText(text);
   797→    toast.success(successMsg);
   798→    return true;
   799→  } catch {
   800→    if (legacyCopy()) {
   801→      toast.success(successMsg);
   802→      return true;
   803→    }
   804→    return false;
   805→  }
   806→}

   808→/** Label hari utk pemisah pesan chat: Hari ini / Kemarin / nama hari / tanggal. */
   809→function dayLabelOf(iso: string): string {
   810→  const d = new Date(iso);
   811→  const startOfDay = (x: Date) => new Date(x.getFullYear(), x.getMonth(), x.getDate()).getTime();
   812→  const diff = startOfDay(new Date()) - startOfDay(d);
   813→  const DAY = 86_400_000;
   814→  if (diff === 0) return "Hari ini";
   815→  if (diff === DAY) return "Kemarin";
   816→  if (diff > 0 && diff < 7 * DAY) return d.toLocaleDateString("id-ID", { weekday: "long" });
   817→  return d.toLocaleDateString("id-ID", { day: "numeric", month: "short", year: "numeric" });
   818→}
