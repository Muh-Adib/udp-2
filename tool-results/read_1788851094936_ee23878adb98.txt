   972→function QuotationFormDialog({
   973→  open,
   974→  onOpenChange,
   975→  opportunityId,
   976→  defaultCurrency,
   977→  editing,
   978→  reviseOf,
   979→  actorName,
   980→  actorRole,
   981→  estimation,
   982→  serviceName,
   983→  onSaved,
   984→}: {
   985→  open: boolean;
   986→  onOpenChange: (v: boolean) => void;
   987→  opportunityId: string;
   988→  defaultCurrency: string;
   989→  editing: QuotationDTO | null;
   990→  /** Ronde 39 — mode revisi: buat quotation BARU menyalin isi quotation sumber (tertolak). */
   991→  reviseOf?: QuotationDTO | null;
   992→  actorName: string;
   993→  actorRole: string;
   994→  /** Ronde 35 — estimasi detail sebagai sumber prefill item quotation. */
   995→  estimation?: EstimationDTO | null;
   996→  serviceName?: string | null;
   997→  onSaved: () => void;
   998→}) {
   999→  const currency = editing?.currency ?? defaultCurrency;
  1000→  const [rows, setRows] = useState<QuotationRow[]>([{ description: "", qty: "1", unitPrice: "" }]);
  1001→  const [discountPct, setDiscountPct] = useState("0");
  1002→  const [taxPct, setTaxPct] = useState("11");
  1003→  const [notes, setNotes] = useState("");
  1004→  const [validUntil, setValidUntil] = useState(defaultValidUntil());
  1005→  const [saving, setSaving] = useState(false);
  1006→  /** Ronde 35 — true bila item terisi otomatis dari estimasi detail. */
  1007→  const prefilledFromEstimation = useRef(false);
  1008→
  1009→  // Prefill setiap kali dialog dibuka (mode buat / edit draft / revisi).
  1010→  useEffect(() => {
  1011→    if (!open) return;
  1012→    if (editing) {
  1013→      prefilledFromEstimation.current = false;
  1014→      const items = parseQuotationItems(editing.items);
  1015→      setRows(
  1016→        items.length > 0
  1017→          ? items.map((it) => ({ description: it.description, qty: String(it.qty), unitPrice: String(it.unitPrice) }))
  1018→          : [{ description: "", qty: "1", unitPrice: "" }]
  1019→      );
  1020→      setDiscountPct(String(editing.discountPct ?? 0));
  1021→      setTaxPct(String(editing.taxPct ?? 11));
  1022→      setNotes(editing.notes ?? "");
  1023→      setValidUntil(editing.validUntil ? editing.validUntil.slice(0, 10) : defaultValidUntil());
  1024→    } else if (reviseOf) {
  1025→      // Ronde 39 — revisi: salin isi quotation sumber + catatan revisi otomatis.
  1026→      prefilledFromEstimation.current = false;
  1027→      const items = parseQuotationItems(reviseOf.items);
  1028→      setRows(
  1029→        items.length > 0
  1030→          ? items.map((it) => ({ description: it.description, qty: String(it.qty), unitPrice: String(it.unitPrice) }))
  1031→          : [{ description: "", qty: "1", unitPrice: "" }]
  1032→      );
  1033→      setDiscountPct(String(reviseOf.discountPct ?? 0));
  1034→      setTaxPct(String(reviseOf.taxPct ?? 11));
  1035→      const revNote = `Revisi ke-${reviseOf.revisionNo ? reviseOf.revisionNo + 1 : 1} dari ${reviseOf.number}${reviseOf.notes ? ` — ${reviseOf.notes}` : ""}`;
  1036→      setNotes(revNote);
  1037→      setValidUntil(defaultValidUntil());
  1038→    } else {
  1039→      // Ronde 35 — estimasi detail (approved/pending) dengan revenue > 0 →
  1040→      // item quotation terisi otomatis: sales tidak mengetik ulang angka estimasi.
  1041→      const estRevenue = Number(estimation?.revenue ?? 0);
  1042→      if (estRevenue > 0) {
  1043→        const label = serviceName?.trim()
  1044→          ? `Layanan ${serviceName.trim()} — sesuai estimasi detail`
  1045→          : "Layanan utama — sesuai estimasi detail";
  1046→        setRows([{ description: label, qty: "1", unitPrice: String(estRevenue) }]);
  1047→        setDiscountPct(String(estimation?.discountPct ?? 0));
  1048→        setTaxPct(String(estimation?.taxPct ?? 11));
  1049→        prefilledFromEstimation.current = true;
  1050→      } else {
  1051→        prefilledFromEstimation.current = false;
  1052→        setRows([{ description: "", qty: "1", unitPrice: "" }]);
  1053→        setDiscountPct("0");
  1054→        setTaxPct("11");
  1055→      }
  1056→      setNotes("");
  1057→      setValidUntil(defaultValidUntil());
  1058→    }
  1059→  }, [open, editing, reviseOf, estimation, serviceName]);
  1060→
  1061→  const totals = useMemo(() => {
  1062→    const items = rows.map((r) => {
  1063→      const qty = toNum(r.qty) || 1;
  1064→      const unitPrice = toNum(r.unitPrice);
  1065→      return { description: r.description.trim(), qty, unitPrice, subtotal: qty * unitPrice };
  1066→    });
  1067→    const subtotal = items.reduce((s, it) => s + it.subtotal, 0);
  1068→    const discountAmount = Math.round((subtotal * toNum(discountPct)) / 100);
  1069→    const afterDiscount = subtotal - discountAmount;
  1070→    const taxAmount = Math.round((afterDiscount * toNum(taxPct)) / 100);
  1071→    return { items, subtotal, discountAmount, taxAmount, total: afterDiscount + taxAmount };
  1072→  }, [rows, discountPct, taxPct]);

  1074→  const filledItems = totals.items.filter((it) => it.description.length > 0);

  1076→  // Ronde 36 (audit FIX): persen dipatok 0–100 di klien juga — dulu -10 DISKON
  1077→  // bisa menaikkan total dokumen resmi, 150% pajak juga lolos (server kini clamp juga).
  1078→  const discountPctNum = Math.min(100, Math.max(0, toNum(discountPct)));

  1090→    setSaving(true);

  1094→          action: "update",
  1095→          items: filledItems,

  1098→          notes: notes.trim() || null,
  1099→          validUntil: validUntil || null,
  1100→          actorName,
  1101→          actorRole,
  1102→        });
  1103→        toast.success(`Quotation ${editing.number} diperbarui`);
  1104→      } else {

  1108→          discountPct: discountPctNum,
  1109→          taxPct: taxPctNum,

  1111→          validUntil: validUntil || undefined,
  1112→          // Ronde 39 — kaitkan quotation baru sebagai revisi dari quotation sumber

  1114→          actorName,
  1115→          actorRole,
  1116→        });
  1117→        toast.success(reviseOf ? `Revisi quotation dibuat (menyusul ${reviseOf.number})` : "Quotation draft dibuat");
  1118→      }
  1119→      onOpenChange(false);
  1120→      onSaved();
  1121→    } catch (e) {
  1122→      toast.error(e instanceof Error ? e.message : "Gagal menyimpan quotation");
  1123→    } finally {
  1124→      setSaving(false);
  1125→    }
