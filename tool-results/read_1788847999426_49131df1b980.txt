   265→   265→function hashColor(name: string): string {
   266→   266→  let h = 0;
   267→   267→  for (let i = 0; i < name.length; i++) h = (h * 31 + name.charCodeAt(i)) >>> 0;
   268→   268→  return AVATAR_PALETTE[h % AVATAR_PALETTE.length];
   269→   269→}

   271→   271→function useDebounced<T>(value: T, delay = 300): T {
   272→   272→  const [debounced, setDebounced] = useState(value);
   273→   273→  useEffect(() => {
   274→   274→    const t = setTimeout(() => setDebounced(value), delay);
   275→   275→    return () => clearTimeout(t);
   276→   276→  }, [value, delay]);
   277→   277→  return debounced;
   278→   278→}
