    72→/** Durasi pendek utk uptime layanan: "45 dtk", "12 mnt", "2 j 13 mnt", "3 hr 2 j". */
    73→function formatUptimeShort(ms: number): string {
    74→  const s = Math.max(0, Math.floor(ms / 1000));
    75→  if (s < 60) return `${s} dtk`;
    76→  const m = Math.floor(s / 60);
    77→  if (m < 60) return `${m} mnt`;
    78→  const h = Math.floor(m / 60);
    79→  if (h < 24) return `${h} j ${m % 60} mnt`;
    80→  const d = Math.floor(h / 24);
    81→  return `${d} hr ${h % 24} j`;
    82→}

    84→function normalizeStats(v: unknown): ServiceStats | null {
    85→  if (!v || typeof v !== "object") return null;
    86→  const s = v as Partial<ServiceStats>;
    87→  if (typeof s.uptimeMs !== "number" || typeof s.activeRooms !== "number") return null;
    88→  return {
    89→    startedAt: typeof s.startedAt === "string" ? s.startedAt : "",
    90→    uptimeMs: s.uptimeMs,
    91→    activeRooms: s.activeRooms,
    92→    totalEmits: typeof s.totalEmits === "number" ? s.totalEmits : 0,
    93→    pollCount: typeof s.pollCount === "number" ? s.pollCount : 0,
    94→    pollErrorCount: typeof s.pollErrorCount === "number" ? s.pollErrorCount : 0,
    95→    lastPollAt: typeof s.lastPollAt === "string" ? s.lastPollAt : null,
    96→  };
    97→}

    99→/** Minta statistik layanan via socket "stats" (ack + fallback event "service:stats").
   100→ * Koneksi sekali pakai (forceNew, tanpa reconnect) agar TIDAK mengganggu socket
   101→ * notifikasi milik useNotifSocket — hanya dipanggil saat panel preferensi dibuka. */
   102→function fetchServiceStats(): Promise<ServiceStats | null> {
   103→  return new Promise((resolve) => {
   104→    let done = false;
   105→    let socket: ReturnType<typeof io> | null = null;
   106→    const finish = (v: ServiceStats | null) => {
   107→      if (done) return;
   108→      done = true;
   109→      try {
   110→        socket?.close();
   111→      } catch {
   112→        /* abaikan */
   113→      }
   114→      resolve(v);
   115→    };
   116→    try {
   117→      socket = io("/?XTransformPort=3005", {
   118→        path: "/",
   119→        transports: ["polling", "websocket"],
   120→        forceNew: true,
   121→        reconnection: false,
   122→        timeout: 4000,
   123→      });
   124→      socket.on("connect", () => {
   125→        socket?.once("service:stats", (s: unknown) => finish(normalizeStats(s)));
   126→        socket?.emit("stats", (s: unknown) => finish(normalizeStats(s)));
   127→      });
   128→      socket.on("connect_error", () => finish(null));
   129→      window.setTimeout(() => finish(null), 4500);
   130→    } catch {
   131→      finish(null);
   132→    }
   133→  });
   134→}
