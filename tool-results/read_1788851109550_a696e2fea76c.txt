    40→function taskTypeMeta(type: string) {
    41→  return TASK_TYPES.find((t) => t.key === type) ?? { key: type, label: type, icon: ClipboardList, cls: "bg-zinc-100 text-zinc-600" };

    44→function priorityMeta(p: string): { label: string; cls: string } {
    45→  switch (p) {
    46→    case "urgent": return { label: "Urgent", cls: "bg-rose-100 text-rose-700" };
    47→    case "high": return { label: "Tinggi", cls: "bg-amber-100 text-amber-700" };
    48→    case "medium": return { label: "Sedang", cls: "bg-zinc-100 text-zinc-600" };
    49→    case "low": return { label: "Rendah", cls: "bg-zinc-100 text-zinc-400" };
    50→    default: return { label: p, cls: "bg-zinc-100 text-zinc-600" };
    51→  }
    52→}

    56→function startOfToday(): Date {
    57→  const d = new Date();
    58→  d.setHours(0, 0, 0, 0);
    59→  return d;
    60→}
    61→
    62→function endOfToday(): Date {
    63→  const d = new Date();
    64→  d.setHours(23, 59, 59, 999);
    65→  return d;
    66→}
    67→
    68→function isOverdue(t: TaskDTO): boolean {
    69→  return t.status === "open" && !!t.dueDate && new Date(t.dueDate).getTime() < startOfToday().getTime();
    70→}
    71→
    72→function isToday(t: TaskDTO): boolean {
    73→  if (!t.dueDate) return false;
    74→  const due = new Date(t.dueDate).getTime();
    75→  return due >= startOfToday().getTime() && due <= endOfToday().getTime();
    76→}

   312→/** Ganti placeholder template dengan data opportunity yang diketahui. */
   313→function renderTemplate(body: string, ctx: SendContext, senderName: string): string {
   314→  const map: Record<string, string> = {
   315→    contact_name: ctx.contact?.fullName ?? "Bapak/Ibu",
   316→    company_name: ctx.company?.name ?? "perusahaan Bapak/Ibu",
   317→    brand_name: ctx.brand?.name ?? "tim kami",
   318→    service_name: ctx.serviceName ?? ctx.title,
   319→    marketing_name: senderName,
   320→  };
   321→  return body.replace(/{{\s*(\w+)\s*}}/g, (full, key: string) => map[key] ?? full);
   322→}

   324→function channelMeta(key: string) {
   325→  return OUTBOUND_CHANNELS.find((c) => c.key === key) ?? OUTBOUND_CHANNELS[0];
   326→}

   328→function SendOutboundDialog({ task, onClose, onSent }: {
   329→  task: TaskDTO | null;
   330→  onClose: () => void;
   331→  onSent: (task: TaskDTO, updated?: TaskDTO) => void;
   332→}) {
   333→  const user = useCrmStore((s) => s.user);
   334→  const [ctx, setCtx] = useState<SendContext | null>(null);
   335→  const [templates, setTemplates] = useState<FollowUpTemplateDTO[]>([]);
   336→  const [loadingCtx, setLoadingCtx] = useState(false);
   337→  const [channel, setChannel] = useState("whatsapp");
   338→  const [templateId, setTemplateId] = useState<string>("blank");
   339→  const [body, setBody] = useState("");
   340→  const [markDone, setMarkDone] = useState(true);
   341→  const [sending, setSending] = useState(false);
   342→
   343→  // Muat konteks opportunity + template saat dialog dibuka
   344→  useEffect(() => {
   345→    if (!task?.opportunityId) return;
   346→    let alive = true;
   347→    setLoadingCtx(true);
   348→    setCtx(null);
   349→    setTemplates([]);
   350→    setTemplateId("blank");
   351→    setBody("");
   352→    setChannel("whatsapp");
   353→    setMarkDone(task.status !== "done");
   354→    (async () => {
   355→      try {
   356→        const [detail, tplRes] = await Promise.all([
   357→          api.opportunity(task.opportunityId as string),
   358→          api.followUpTemplates("all"),
   359→        ]);
   360→        if (!alive) return;
   361→        const o = detail.opportunity;
   362→        const nextCtx: SendContext = {
   363→          id: o.id,
   364→          title: o.title,
   365→          serviceName: o.serviceName,
   366→          brand: o.brand ? { id: o.brand.id, name: o.brand.name, color: o.brand.color } : null,
   367→          contact: o.contact
   368→            ? { id: o.contact.id, fullName: o.contact.fullName, email: o.contact.email, whatsapp: o.contact.whatsapp }
   369→            : null,
   370→          company: o.company ? { id: o.company.id, name: o.company.name } : null,
   371→          ownerName: o.ownerName,
   372→        };
   373→        setCtx(nextCtx);
   374→        // Template global + milik brand opportunity, urut delay
   375→        const relevant = tplRes.templates
   376→          .filter((t) => !t.brandId || t.brandId === o.brandId)
   377→          .sort((a, b) => a.delayDays - b.delayDays);
   378→        setTemplates(relevant);
   379→      } catch {
   380→        if (alive) toast.error("Gagal memuat konteks opportunity");
   381→      } finally {
   382→        if (alive) setLoadingCtx(false);
   383→      }
   384→    })();
   385→    return () => { alive = false; };
   386→  }, [task]);
   387→
   388→  const senderName = user?.name ?? ctx?.ownerName ?? "Tim Sales";
   389→
   390→  function pickTemplate(id: string) {
   391→    setTemplateId(id);
   392→    if (!ctx) return;
   393→    if (id === "blank") {
   394→      setBody("");
   395→      return;
   396→    }
   397→    const tpl = templates.find((t) => t.id === id);
   398→    if (!tpl) return;
   399→    setChannel(tpl.channel);
   400→    setBody(renderTemplate(tpl.body, ctx, senderName));
   401→  }
   402→
   403→  const unknownPlaceholders = useMemo(() => {
   404→    if (!ctx) return [] as string[];
   405→    const found = body.match(/{{\s*(\w+)\s*}}/g) ?? [];
   406→    return Array.from(new Set(found));
   407→  }, [body, ctx]);
   408→
   409→  async function handleSend() {
   410→    if (!task || !ctx) return;
   411→    const content = body.trim();
   412→    if (!content) {
   413→      toast.error("Isi pesan tidak boleh kosong");
   414→      return;
   415→    }
   416→    const ch = channelMeta(channel);
   417→    if (ch.key === "email" && !ctx.contact?.email) {
   418→      toast.error("Kontak tidak memiliki alamat email");
   419→      return;
   420→    }
   421→    if (ch.key === "whatsapp" && !ctx.contact?.whatsapp) {
   422→      toast.error("Kontak tidak memiliki nomor WhatsApp");
   423→      return;
   424→    }
   425→    setSending(true);
   426→    try {
   427→      await api.createInteraction({
   428→        channel: ch.key,
   429→        direction: "outbound",
   430→        brandId: ctx.brand?.id,
   431→        senderName,
   432→        recipientName: ctx.contact?.fullName,
   433→        subject: ch.key === "email" ? `Follow-up: ${ctx.title}` : undefined,
   434→        content,
   435→        opportunityId: ctx.id,
   436→        contactId: ctx.contact?.id,
   437→        companyId: ctx.company?.id,
   438→        externalId: `task:${task.id}`,
   439→        actorName: user?.name ?? "System",
   440→        actorRole: user?.role ?? "system",
   441→      });
   442→      let updated: TaskDTO | undefined;
   443→      if (markDone && task.status !== "done") {
   444→        const res = await api.updateTask(task.id, { status: "done" });
   445→        updated = res.task;
   446→      }
   447→      toast.success(`Pesan ${ch.label} tercatat terkirim`, {
   448→        description: `Tercatat sebagai Interaction outbound pada timeline ${ctx.title}.`,
   449→      });
   450→      onSent(task, updated);
   451→      onClose();
   452→    } catch (err) {
   453→      toast.error(err instanceof Error ? err.message : "Gagal mencatat pesan");
   454→    } finally {
   455→      setSending(false);
   456→    }
   457→  }
   458→
   459→  const ChIcon = channelMeta(channel).icon;
   460→
   461→  return (
   462→    <Dialog open={task !== null} onOpenChange={(open) => { if (!open && !sending) onClose(); }}>
   463→      <DialogContent className="max-h-[92vh] overflow-y-auto crm-scroll sm:max-w-xl" aria-label="Kirim pesan follow-up">
   464→        <DialogHeader>
   465→          <DialogTitle className="flex items-center gap-2">
   466→            <span className="flex h-7 w-7 items-center justify-center rounded-lg bg-zinc-900 text-white" aria-hidden>
   467→              <Send className="h-3.5 w-3.5" />
   468→            </span>
   469→            Kirim &amp; Catat Follow-up
   470→          </DialogTitle>
   471→          <DialogDescription>

   482→            <Skeleton className="h-16 w-full rounded-lg" />
   483→            <Skeleton className="h-20 w-full rounded-lg" />
   484→            <Skeleton className="h-28 w-full rounded-lg" />

   490→              <div className="flex flex-wrap items-center gap-x-4 gap-y-1.5 text-xs">
   491→                <span className="flex items-center gap-1.5 font-medium text-zinc-900">
   492→                  <span className="flex h-6 w-6 items-center justify-center rounded-full bg-zinc-900 text-[9px] font-bold text-white" aria-hidden>
   493→                    {initials(ctx.contact?.fullName ?? "?")}
   494→                  </span>
   495→                  {ctx.contact?.fullName ?? "Kontak tidak dikenal"}
   496→                </span>
   497→                {ctx.contact?.whatsapp ? (
   498→                  <span className="inline-flex items-center gap-1 text-zinc-600"><MessageCircle className="h-3.5 w-3.5 text-emerald-600" aria-hidden /> {ctx.contact.whatsapp}</span>
   499→                ) : null}
   500→                {ctx.contact?.email ? (
   501→                  <span className="inline-flex items-center gap-1 text-zinc-600"><Mail className="h-3.5 w-3.5 text-zinc-400" aria-hidden /> {ctx.contact.email}</span>
   502→                ) : null}
   503→                {ctx.brand ? (
   504→                  <span className="inline-flex items-center gap-1 text-zinc-600">
   505→                    <span className="h-2 w-2 rounded-full" style={{ backgroundColor: ctx.brand.color }} aria-hidden />
   506→                    {ctx.brand.name}
   507→                  </span>
   508→                ) : null}
   509→              </div>
   510→            </div>
   511→

   513→            <div className="space-y-2">

   518→                <p className="rounded-lg border border-dashed border-zinc-200 p-2.5 text-xs text-zinc-400">

   520→                </p>
   521→              ) : (
   522→                <div className="crm-scroll max-h-40 space-y-1.5 overflow-y-auto pr-1">
   523→                  <label
   524→                    className={`flex cursor-pointer items-center justify-between gap-2 rounded-lg border p-2.5 text-xs transition-colors ${templateId === "blank" ? "border-zinc-900 bg-zinc-50 ring-1 ring-zinc-900" : "border-zinc-200 hover:border-zinc-300"}`}

   527→                      <Sparkles className="h-3.5 w-3.5 text-zinc-400" aria-hidden /> Tulis pesan sendiri

   536→                        className={`flex cursor-pointer items-center justify-between gap-2 rounded-lg border p-2.5 text-xs transition-colors ${templateId === t.id ? "border-zinc-900 bg-zinc-50 ring-1 ring-zinc-900" : "border-zinc-200 hover:border-zinc-300"}`}

   542→                          </span>
   543→                          <span className="mt-0.5 flex items-center gap-2 text-[10px] text-zinc-500">

   545→                              <TplIcon className="h-3 w-3" aria-hidden />
   546→                              {channelMeta(t.channel).label}
   547→                            </span>
   548→                            <span aria-hidden>·</span>
   549→                            <span>H+{t.delayDays}</span>
   550→                          </span>
   551→                        </span>
   552→                        <input type="radio" className="sr-only" checked={templateId === t.id} onChange={() => pickTemplate(t.id)} aria-label={`Pakai template ${t.name}`} />
   553→                      </label>
   554→                    );
   555→                  })}
   556→                </div>
   557→              )}
   558→            </div>
   559→
   560→            {/* Kanal + isi */}
   561→            <div className="grid gap-3 sm:grid-cols-[170px_1fr]">
   562→              <div className="space-y-2">
   563→                <Label htmlFor="outbound-channel">Kanal</Label>
   564→                <Select value={channel} onValueChange={setChannel}>
   565→                  <SelectTrigger id="outbound-channel" className="bg-white">
   566→                    <SelectValue />
   567→                  </SelectTrigger>
   568→                  <SelectContent>
   569→                    {OUTBOUND_CHANNELS.map((c) => (
   570→                      <SelectItem key={c.key} value={c.key} disabled={c.key === "email" && !ctx.contact?.email}>
   571→                        <span className="flex items-center gap-2"><c.icon className="h-3.5 w-3.5" aria-hidden /> {c.label}</span>
   572→                      </SelectItem>
   573→                    ))}
   574→                  </SelectContent>
   575→                </Select>
   576→              </div>
   577→              <div className="space-y-2">
   578→                <Label htmlFor="outbound-body">Isi pesan</Label>
   579→                <Textarea
   580→                  id="outbound-body"
   581→                  value={body}
   582→                  onChange={(e) => setBody(e.target.value)}
   583→                  rows={7}
   584→                  placeholder="Tulis pesan follow-up…"
   585→                  className="bg-white text-sm"
   586→                />
   587→                {unknownPlaceholders.length > 0 ? (
   588→                  <p className="text-[11px] text-amber-600">
   589→                    Placeholder belum terisi: {unknownPlaceholders.join(", ")} — lengkapi manual sebelum kirim.
   590→                  </p>
   591→                ) : null}
   592→              </div>
   593→            </div>
   594→
   595→            <label className="flex cursor-pointer items-center gap-2 rounded-lg border bg-zinc-50 p-3 text-sm text-zinc-700">
   596→              <Checkbox checked={markDone} onCheckedChange={(v) => setMarkDone(v === true)} aria-label="Tandai task selesai setelah kirim" />
   597→              Tandai task selesai setelah pesan tercatat
   598→            </label>
   599→          </div>
   600→        )}
   601→
   602→        <DialogFooter>
   603→          <Button variant="outline" onClick={onClose} disabled={sending}>Batal</Button>
   604→          <Button
   605→            onClick={() => void handleSend()}
   606→            disabled={sending || loadingCtx || !ctx || !body.trim()}
   607→            aria-label="Kirim dan catat pesan follow-up"
   608→          >
   609→            {sending ? <RefreshCw className="h-4 w-4 animate-spin" aria-hidden /> : <ChIcon className="h-4 w-4" aria-hidden />}
   610→            {sending ? "Mencatat…" : `Kirim via ${channelMeta(channel).label}`}
   611→          </Button>
   612→        </DialogFooter>
   613→      </DialogContent>
   614→    </Dialog>
   615→  );
   616→}

   620→export default function FollowupsModule() {
   621→  const user = useCrmStore((s) => s.user);
   622→
   623→  const [tasks, setTasks] = useState<TaskDTO[] | null>(null);
   624→  const [loading, setLoading] = useState(true);
   625→  const [error, setError] = useState<string | null>(null);
   626→  const [busyId, setBusyId] = useState<string | null>(null);
   627→
   628→  const [statusFilter, setStatusFilter] = useState("all");
   629→  const [assigneeFilter, setAssigneeFilter] = useState("all");
   630→  const [onlyOverdue, setOnlyOverdue] = useState(false);
   631→
   632→  const [createOpen, setCreateOpen] = useState(false);
   633→  const [saving, setSaving] = useState(false);
   634→  const [form, setForm] = useState({
   635→    title: "", type: "follow_up", priority: "medium", assigneeName: "", dueDate: "", opportunityId: "",
   636→  });
   637→  // Fase 3 — kirim & catat pesan outbound dari task follow-up
   638→  const [sendTarget, setSendTarget] = useState<TaskDTO | null>(null);
   639→
   640→  const load = useCallback(async (silent = false) => {
   641→    if (!silent) setLoading(true);
   642→    setError(null);
   643→    try {
   644→      const res = await api.tasks({
   645→        status: statusFilter !== "all" ? statusFilter : undefined,
   646→        assignee: assigneeFilter !== "all" ? assigneeFilter : undefined,
   647→        overdue: onlyOverdue ? "true" : undefined,
   648→      });
   649→      setTasks(res.tasks);
   650→    } catch (err) {
   651→      setError(err instanceof Error ? err.message : "Gagal memuat task");
   652→      if (!silent) toast.error("Gagal memuat task follow-up");
   653→    } finally {
   654→      setLoading(false);
   655→    }
   656→  }, [statusFilter, assigneeFilter, onlyOverdue]);
   657→
   658→  useEffect(() => { void load(); }, [load]);
   659→
   660→  const assignees = useMemo(() => {
   661→    const set = new Set<string>();
   662→    (tasks ?? []).forEach((t) => { if (t.assigneeName) set.add(t.assigneeName); });
   663→    return Array.from(set).sort((a, b) => a.localeCompare(b));
   664→  }, [tasks]);
   665→
   666→  const stats = useMemo(() => {
   667→    const list = tasks ?? [];
   668→    const todayEnd = endOfToday().getTime();
   669→    const weekAgo = todayEnd - 7 * 24 * 60 * 60 * 1000;
   670→    return {
   671→      open: list.filter((t) => t.status === "open").length,
   672→      overdue: list.filter((t) => isOverdue(t)).length,
   673→      dueToday: list.filter((t) => t.status === "open" && isToday(t)).length,
   674→      doneThisWeek: list.filter((t) => t.status === "done" && t.completedAt
   675→        && new Date(t.completedAt).getTime() >= weekAgo
   676→        && new Date(t.completedAt).getTime() <= todayEnd).length,
   677→    };
   678→  }, [tasks]);
   679→
   680→  const groups = useMemo(() => {
   681→    const list = tasks ?? [];
   682→    return {
   683→      overdue: list.filter((t) => isOverdue(t)),
   684→      today: list.filter((t) => t.status === "open" && isToday(t)),
   685→      upcoming: list.filter((t) => t.status === "open" && !isOverdue(t) && !isToday(t)),
   686→      done: list.filter((t) => t.status === "done"),
   687→    };
   688→  }, [tasks]);
   689→
   690→  async function handleToggle(t: TaskDTO) {
   691→    const nextStatus = t.status === "done" ? "open" : "done";
   692→    setBusyId(t.id);
   693→    try {
   694→      const res = await api.updateTask(t.id, { status: nextStatus });
   695→      setTasks((prev) => (prev ?? []).map((x) => (x.id === t.id ? res.task : x)));
   696→      toast.success(nextStatus === "done" ? "Task ditandai selesai" : "Task dibuka kembali");
   697→    } catch (err) {
   698→      toast.error(err instanceof Error ? err.message : "Gagal memperbarui task");
   699→    } finally {
   700→      setBusyId(null);
   701→    }
   702→  }
   703→
   704→  async function handleCreate(e: React.FormEvent) {
   705→    e?.preventDefault();
   706→    if (!form.title.trim()) { toast.error("Judul task wajib diisi"); return; }
   707→    setSaving(true);
   708→    try {
   709→      await api.createTask({
   710→        title: form.title.trim(),
   711→        type: form.type,
   712→        priority: form.priority,
   713→        assigneeName: form.assigneeName.trim() || undefined,
   714→        dueDate: form.dueDate || undefined,
   715→        opportunityId: form.opportunityId.trim() || undefined,
   716→        actorName: user?.name ?? "System",
   717→        actorRole: user?.role ?? "system",
   718→      });
   719→      toast.success("Task follow-up dibuat");
   720→      setCreateOpen(false);
   721→      setForm({ title: "", type: "follow_up", priority: "medium", assigneeName: "", dueDate: "", opportunityId: "" });
   722→      await load(true);
   723→    } catch (err) {
   724→      toast.error(err instanceof Error ? err.message : "Gagal membuat task");
   725→    } finally {
   726→      setSaving(false);
   727→    }
   728→  }
   729→
   730→  /** Callback setelah pesan outbound tercatat: perbarui status task di list. */
   731→  function handleSent(_task: TaskDTO, updated?: TaskDTO) {
   732→    if (updated) {
   733→      setTasks((prev) => (prev ?? []).map((x) => (x.id === updated.id ? updated : x)));
   734→    }
   735→  }
   736→
   737→  if (loading && tasks === null) return <FollowupsSkeleton />;
   738→
   739→  const sectionBody = (list: TaskDTO[], emptyText: string): ReactNode =>
   740→    list.length === 0 ? (
   741→      <EmptyGroup text={emptyText} />
   742→    ) : (
   743→      <div className="space-y-3">
   744→        {list.map((t) => (
   745→          <TaskCard key={t.id} task={t} onToggle={handleToggle} busy={busyId === t.id} onSend={setSendTarget} />
   746→        ))}
   747→      </div>
   748→    );
   749→
   750→  return (
   751→    <div className="space-y-6">
   752→      {/* Header + filter */}
   753→      <div className="flex flex-col gap-3 sm:flex-row sm:items-start sm:justify-between">
   754→        <div>
   755→          <h1 className="text-xl font-bold tracking-tight text-zinc-900">Follow-up Center</h1>
   756→          <p className="text-sm text-zinc-500">Task &amp; jadwal follow-up per opportunity</p>
   757→        </div>
   758→        <div className="flex items-center gap-2">
   759→          <Button
   760→            variant="outline"
   761→            size="sm"
   762→            onClick={() => void load()}
   763→            disabled={loading}
   764→            aria-label="Muat ulang daftar task"
   765→          >
   766→            <RefreshCw className={`h-4 w-4 ${loading ? "animate-spin" : ""}`} aria-hidden /> Muat ulang
   767→          </Button>
   768→          <Button size="sm" onClick={() => setCreateOpen(true)} aria-label="Tambah task follow-up baru">
   769→            <Plus className="h-4 w-4" aria-hidden /> Tambah Task
   770→          </Button>
   771→        </div>
   772→      </div>
   773→
   774→      <div className="flex flex-col gap-3 rounded-xl border bg-white p-4 shadow-sm lg:flex-row lg:items-center lg:justify-between">
   775→        <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   776→          <div className="flex items-center gap-2">
   777→            <ListFilter className="h-4 w-4 shrink-0 text-zinc-400" aria-hidden />
   778→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   779→              <SelectTrigger className="w-full sm:w-[150px]" aria-label="Filter status task">
   780→                <SelectValue placeholder="Status" />
   781→              </SelectTrigger>

   799→        <label className="flex cursor-pointer items-center gap-2 text-sm text-zinc-700">

   801→          Hanya overdue

   883→            </div>
   884→            <div className="grid gap-4 sm:grid-cols-2">
   885→              <div className="grid gap-2">
   886→                <Label htmlFor="task-assignee">Assignee</Label>
   887→                <Input
   888→                  id="task-assignee" value={form.assigneeName}
   889→                  onChange={(e) => setForm((f) => ({ ...f, assigneeName: e.target.value }))}
   890→                  placeholder="Nama orang yang mengerjakan"
   891→                />
   892→              </div>
   893→              <div className="grid gap-2">

   895→                <Input
   896→                  id="task-due" type="date" value={form.dueDate}
   897→                  onChange={(e) => setForm((f) => ({ ...f, dueDate: e.target.value }))}
   898→                />
   899→              </div>
   900→            </div>
   901→            <div className="grid gap-2">
   902→              <Label htmlFor="task-opp">Opportunity ID (opsional)</Label>
   903→              <Input
   904→                id="task-opp" value={form.opportunityId}
   905→                onChange={(e) => setForm((f) => ({ ...f, opportunityId: e.target.value }))}
   906→                placeholder="Tempel ID opportunity bila terkait"
   907→              />
   908→            </div>
   909→            <DialogFooter className="mt-2">
   910→              <Button type="button" variant="outline" onClick={() => setCreateOpen(false)}>Batal</Button>
   911→              <Button type="submit" disabled={saving} aria-label="Simpan task baru">
   912→                {saving ? "Menyimpan…" : "Simpan Task"}
   913→              </Button>
   914→            </DialogFooter>
   915→          </form>
   916→        </DialogContent>
   917→      </Dialog>
   918→
   919→      {/* Dialog kirim & catat pesan outbound (Fase 3) */}
   920→      <SendOutboundDialog task={sendTarget} onClose={() => setSendTarget(null)} onSent={handleSent} />
   921→    </div>
   922→  );
