    46→    46→/** Baca file gambar kecil → data URL (dipakai logo & kop surat). */
    47→    47→function readImageFile(file: File, maxChars = 1_500_000): Promise<string> {
    48→    48→  return new Promise((resolve, reject) => {
    49→    49→    if (!file.type.startsWith("image/")) {
    50→    50→      reject(new Error("File harus berupa gambar (PNG/JPG/SVG/WebP)"));
    51→    51→      return;
    52→    52→    }
    53→    53→    if (file.size > 1_100_000) {
    54→    54→      reject(new Error("Ukuran gambar terlalu besar (maksimal ±1MB)"));
    55→    55→      return;
    56→    56→    }
    57→    57→    const reader = new FileReader();
    58→    58→    reader.onload = () => {
    59→    59→      const result = String(reader.result ?? "");
    60→    60→      if (result.length > maxChars) {
    61→    61→        reject(new Error("Gambar terlalu besar setelah dibaca — pakai file lebih kecil"));
    62→    62→        return;
    63→    63→      }
    64→    64→      resolve(result);
    65→    65→    };
    66→    66→    reader.onerror = () => reject(new Error("Gagal membaca file"));
    67→    67→    reader.readAsDataURL(file);
    68→    68→  });
    69→    69→}

    80→    80→/** Tanggal-waktu ringkas id-ID (mis. 02 Sep, 13.01) — "" bila tidak valid. */
    81→    81→function fmtWhen(iso?: string | null): string {
    82→    82→  if (!iso) return "";
    83→    83→  const d = new Date(iso);
    84→    84→  if (Number.isNaN(d.getTime())) return "";
    85→    85→  return new Intl.DateTimeFormat("id-ID", { day: "2-digit", month: "short", hour: "2-digit", minute: "2-digit" }).format(d);
    86→    86→}
